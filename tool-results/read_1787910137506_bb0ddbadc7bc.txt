    88→// ============== HELPERS ==============
    89→
    90→function isToday(dateStr: string): boolean {
    91→  const d = new Date(dateStr)
    92→  const now = new Date()
    93→  return (
    94→    d.getFullYear() === now.getFullYear() &&
    95→    d.getMonth() === now.getMonth() &&
    96→    d.getDate() === now.getDate()
    97→  )
    98→}
    99→
   100→function isThisWeek(dateStr: string): boolean {
   101→  const d = new Date(dateStr)
   102→  const now = new Date()
   103→  const startOfWeek = new Date(now)
   104→  startOfWeek.setDate(now.getDate() - now.getDay() + 1)
   105→  startOfWeek.setHours(0, 0, 0, 0)
   106→  return d >= startOfWeek
   107→}
   108→
   109→function isThisMonth(dateStr: string): boolean {
   110→  const d = new Date(dateStr)
   111→  const now = new Date()
   112→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   113→}
   114→
   115→function matchesDateRange(dateStr: string, range: DateRange): boolean {
   116→  switch (range) {
   117→    case 'aujourdhui':
   118→      return isToday(dateStr)
   119→    case 'semaine':
   120→      return isThisWeek(dateStr)
   121→    case 'mois':
   122→      return isThisMonth(dateStr)
   123→    case 'tous':
   124→    default:
   125→      return true
   126→  }
   127→}
   128→
   129→function formatDate(dateStr: string): string {
   130→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   131→    day: '2-digit',
   132→    month: '2-digit',
   133→    year: 'numeric',
   134→    hour: '2-digit',
   135→    minute: '2-digit',
   136→  })
   137→}

   542→                    </Button>
   543→                  )
   544→                )}
   545→            </div>
   546→            <Button
   547→              variant="outline"
   548→              size="sm"
   549→              disabled={currentPage >= totalPages}
   550→              onClick={() => setCurrentPage((p) => p + 1)}
   551→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   552→            >
   553→              Suivant
   554→              <ChevronRight className="ml-1 h-4 w-4" />
   555→            </Button>
   556→          </div>
   557→        </div>
   558→      )}
   559→
   560→      {/* ===== REJECT DIALOG ===== */}
   561→      <Dialog open={rejectDialogOpen} onOpenChange={setRejectDialogOpen}>
   562→        <DialogContent className="sm:max-w-md">
   563→          <DialogHeader>
   564→            <DialogTitle className={isDark ? 'text-slate-100' : 'text-[#333333]'}>
   565→              Rejeter l&apos;enrôlement
   566→            </DialogTitle>
   567→            <DialogDescription>
   568→              {rejectTarget && (
   569→                <>
   570→                  <span className={`font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   571→                    {rejectTarget.actorName}
   572→                  </span>{' '}
   573→                  — {rejectTarget.dossierId}
   574→                </>
   575→              )}
   576→            </DialogDescription>
   577→          </DialogHeader>
   578→
   579→          <div className="flex flex-col gap-4">
   580→            {/* Predefined reasons */}
   581→            <div className="flex flex-col gap-2">
   582→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   583→                Raison prédéfinie
   584→              </Label>
   585→              <div className="flex flex-wrap gap-2">
   586→                {PREDEFINED_REASONS.map((reason) => (
   587→                  <Button
   588→                    key={reason}
   589→                    variant={selectedPreset === reason ? 'default' : 'outline'}
   590→                    size="sm"
   591→                    onClick={() => handlePresetReason(reason)}
   592→                    className={
   593→                      selectedPreset === reason
   594→                        ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   595→                        : isDark ? 'text-slate-100' : 'text-[#333333]'
   596→                    }
   597→                  >
   598→                    {reason}
   599→                  </Button>
   600→                ))}
   601→              </div>
   602→            </div>
   603→
   604→            {/* Custom reason */}
   605→            <div className="flex flex-col gap-2">
   606→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   607→                Raison du rejet <span className="text-red-500">*</span>
   608→              </Label>
   609→              <Textarea
   610→                value={rejectReason}
   611→                onChange={(e) => setRejectReason(e.target.value)}
   612→                placeholder="Décrivez la raison du rejet..."
   613→                className="min-h-[100px] resize-none"
   614→              />
   615→            </div>
   616→          </div>
   617→
   618→          <DialogFooter>
   619→            <Button
   620→              variant="outline"
   621→              onClick={() => setRejectDialogOpen(false)}
   622→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   623→            >
   624→              Annuler
   625→            </Button>
   626→            <Button
   627→              variant="destructive"
   628→              onClick={handleConfirmReject}
   629→              disabled={!rejectReason.trim()}
   630→            >
   631→              <XCircle className="mr-1.5 h-4 w-4" />
   632→              Confirmer le rejet
   633→            </Button>
   634→          </DialogFooter>
   635→        </DialogContent>
   636→      </Dialog>
   637→        </>
   638→      )}
   639→    </div>
   640→  )
