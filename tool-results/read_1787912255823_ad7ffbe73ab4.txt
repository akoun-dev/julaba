     1→'use client'
     2→
     3→import { useState, useCallback, useEffect } from 'react'
     4→import { useAppStore } from '@/lib/stores/app-store'
     5→import { useBackofficeStore, type BoRole, ROLE_LABELS } from '@/lib/stores/backoffice-store'
     6→import { OrbitOtp } from './orbit-otp'
     7→import { ArrowLeft, Shield, Fingerprint, CheckCircle2, Lock, KeyRound } from 'lucide-react'
     8→
     9→// Types for demo accounts from API (no password exposed)
    10→interface DemoAccount {
    11→  email: string
    12→  name: string
    13→  role: string
    14→  zone: string | null
    15→}
    16→
    17→// Type for authenticated user from login API
    18→interface AuthenticatedUser {
    19→  id: string
    20→  email: string
    21→  name: string
    22→  role: string
    23→  zone: string | null
    24→  isActive: boolean
    25→  lastLogin: string | null
    26→  createdAt: string
    27→}
    28→
    29→type Step = 'credentials' | 'mfa' | 'success'
    30→
    31→export function BoAuthScreen() {
    32→  const { setUserRole, navigate, setAuth } = useAppStore()
    33→  const { setBoAuth } = useBackofficeStore()
    34→
    35→  const [step, setStep] = useState<Step>('credentials')
    36→  const [email, setEmail] = useState('')
    37→  const [password, setPassword] = useState('')
    38→  const [error, setError] = useState('')
    39→  const [loading, setLoading] = useState(false)
    40→  const [verifying, setVerifying] = useState(false)
    41→  const [matchedUser, setMatchedUser] = useState<AuthenticatedUser | null>(null)
    42→  const [showDemo, setShowDemo] = useState(false)
    43→  const [otpResetKey, setOtpResetKey] = useState(0)
    44→  const [demoAccounts, setDemoAccounts] = useState<DemoAccount[]>([])
    45→  const [demoLoading, setDemoLoading] = useState(true)
    46→
    47→  // Fetch demo accounts from DB on mount
    48→  useEffect(() => {
    49→    fetch('/api/backoffice/demo-accounts')
    50→      .then((r) => (r.ok ? r.json() : []))
    51→      .then((data: DemoAccount[]) => setDemoAccounts(Array.isArray(data) ? data : []))
    52→      .catch(() => setDemoAccounts([]))
    53→      .finally(() => setDemoLoading(false))
    54→  }, [])
    55→
    56→  const handleLogin = useCallback(() => {
    57→    setError('')
    58→    if (!email || !password) {
    59→      setError('Veuillez remplir tous les champs')
    60→      return
    61→    }
    62→
    63→    setLoading(true)
    64→    fetch('/api/backoffice/login', {
    65→      method: 'POST',
    66→      headers: { 'Content-Type': 'application/json' },
    67→      body: JSON.stringify({ email, password }),
    68→    })
    69→      .then((res) => {
    70→        if (!res.ok) return res.json().then((d) => { throw new Error(d.erreur || 'Erreur') })
    71→        return res.json()
    72→      })
    73→      .then((user: AuthenticatedUser) => {
    74→        setMatchedUser(user)
    75→        setEmail(user.email)
    76→        setStep('mfa')
    77→      })
    78→      .catch((err) => {
    79→        setError(err.message || 'Email ou mot de passe incorrect')
    80→      })
    81→      .finally(() => setLoading(false))
    82→  }, [email, password])
    83→
    84→  const handleMfaComplete = useCallback(
    85→    (code: string) => {
    86→      if (code.length !== 6) return
    87→      setError('')
    88→      setVerifying(true)
    89→
    90→      // Auto-submit after 300ms of verification animation
    91→      setTimeout(() => {
    92→        setStep('success')
    93→        setTimeout(() => {
    94→          if (!matchedUser) return
    95→          setUserRole('backoffice')
    96→          setAuth(matchedUser.email, matchedUser.name, '')
    97→          setBoAuth({
    98→            id: matchedUser.id,
    99→            email: matchedUser.email,
   100→            name: matchedUser.name,
   101→            role: matchedUser.role as BoRole,
   102→            zone: matchedUser.zone || undefined,
   103→            isActive: matchedUser.isActive,
   104→            lastLogin: matchedUser.lastLogin || undefined,
   105→            createdAt: matchedUser.createdAt,
   106→          })
   107→          navigate('bo-dashboard')
   108→        }, 600)
   109→      }, 300)
   110→    },
   111→    [matchedUser, setBoAuth, setAuth, setUserRole, navigate]
   112→  )
   113→

   115→    setError('')
   116→    setLoading(true)
   117→    // Authenticate against DB with the demo account

   119→      method: 'POST',
   120→      headers: { 'Content-Type': 'application/json' },
   121→      body: JSON.stringify({ email: account.email, password: 'admin123' }),

   137→  }, [])
   138→

   141→      setStep('credentials')

   143→      setVerifying(false)
   144→      setOtpResetKey((k) => k + 1)

   146→      navigate('auth')

   151→    (e: React.KeyboardEvent) => {
   152→      if (e.key === 'Enter' && step === 'credentials') handleLogin()
   153→    },
   154→    [step, handleLogin]
   155→  )
   156→
   157→  return (
   158→    <div className="bo-auth-root">
   159→      {/* ============ LEFT PANEL — BRANDING ============ */}
   160→      <div className="bo-auth-left">
   161→        <div className="bo-auth-left-pattern" />
   162→        <div className="bo-auth-left-content">
   163→          <div className="bo-auth-logo">
   164→            <span className="bo-auth-logo-letter">J</span>
   165→          </div>
   166→          <h1 className="bo-auth-left-title">
   167→            Jùlaba
   168→            <br />
   169→            BackOffice
   170→          </h1>
   171→          <p className="bo-auth-left-desc">
   172→            Interface d&rsquo;administration sécurisée pour la gestion des acteurs

   288→              {/* Demo toggle */}

   306→                    <div style={{ textAlign: 'center', padding: '12px', color: 'rgba(255,255,255,0.4)', fontSize: '13px' }}>
   307→                      Aucun compte disponible
   308→                    </div>
   309→                  ) : (
   310→                    demoAccounts.map((account) => (
   311→                      <button
   312→                        key={account.email}
   313→                        className="bo-auth-demo-item"
   314→                        onClick={() => handleDemoLogin(account)}
   315→                      >
   316→                        <div className="bo-auth-demo-info">
   317→                          <span className="bo-auth-demo-email">{account.name} — {account.email}</span>
   318→                          <span className="bo-auth-demo-role">{ROLE_LABELS[account.role as BoRole] || account.role}{account.zone ? ` · ${account.zone}` : ''}</span>
   319→                        </div>
   320→                        <span className="bo-auth-demo-badge">{ROLE_LABELS[account.role as BoRole] || account.role}</span>
   321→                      </button>
   322→                    ))
   323→                  )}
   324→                </div>
   325→              )}
   326→            </div>
   327→          )}
   328→
   329→          {/* ====== STEP: MFA ====== */}
   330→          {step === 'mfa' && (
   331→            <div className="bo-auth-card">
   332→              {/* Fingerprint icon */}
   333→              <div className="bo-auth-mfa-icon-wrapper">
   334→                <div className="bo-auth-mfa-icon-bg">
   335→                  <Fingerprint size={30} />
   336→                </div>
   337→              </div>
   338→
   339→              <p className="bo-auth-mfa-desc">
   340→                Authentification à deux facteurs requise après la connexion.
   341→              </p>
   342→
   343→              {/* OrbitOtp component */}
   344→              <OrbitOtp
   345→                length={6}
   346→                onComplete={handleMfaComplete}
   347→                error={error || undefined}
   348→                verifying={verifying}
   349→                resetKey={otpResetKey}
   350→                onResend={() => {
   351→                  setOtpResetKey((k) => k + 1)
   352→                  setError('')
   353→                }}
   354→              />
   355→
   356→              <p className="bo-auth-mfa-demo-hint">
   357→                Démo : entrez n&rsquo;importe quel code à 6 chiffres
   358→              </p>
   359→
   360→              {/* Back to credentials */}
   361→              <button
   362→                className="bo-auth-mfa-back-btn"
   363→                onClick={() => {
   364→                  setStep('credentials')
   365→                  setError('')
   366→                  setVerifying(false)
   367→                  setOtpResetKey((k) => k + 1)
   368→                }}
   369→              >
   370→                <ArrowLeft size={14} />
   371→                Retour aux identifiants
   372→              </button>
   373→            </div>
   374→          )}
   375→
   376→          {/* ====== STEP: SUCCESS ====== */}
   377→          {step === 'success' && (
   378→            <div className="bo-auth-success">
   379→              <div className="bo-auth-success-icon-wrapper">
   380→                <CheckCircle2 className="bo-auth-success-check" size={40} />
