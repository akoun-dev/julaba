    77→function formatDateTime(iso: string) {
    78→  const d = new Date(iso)
    79→  return d.toLocaleDateString('fr-FR', {
    80→    day: '2-digit',
    81→    month: '2-digit',
    82→    year: 'numeric',
    83→    hour: '2-digit',
    84→    minute: '2-digit',
    85→  })
    86→}
    87→
    88→function getTimeAgo(iso: string) {
    89→  const now = Date.now()
    90→  const then = new Date(iso).getTime()
    91→  const diff = now - then
    92→  const minutes = Math.floor(diff / 60000)
    93→  if (minutes < 1) return 'à l\'instant'
    94→  if (minutes < 60) return `il y a ${minutes}min`
    95→  const hours = Math.floor(minutes / 60)
    96→  if (hours < 24) return `il y a ${hours}h`
    97→  const days = Math.floor(hours / 24)
    98→  if (days < 7) return `il y a ${days}j`
    99→  return formatDateTime(iso)
   100→}

   102→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   103→  const now = new Date()
   104→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   105→  let start: Date
   106→
   107→  switch (dateRange) {
   108→    case 'aujourdhui':
   109→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   110→      break
   111→    case 'semaine': {
   112→      const dayOfWeek = now.getDay() || 7
   113→      start = new Date(now)
   114→      start.setDate(now.getDate() - dayOfWeek + 1)
   115→      start.setHours(0, 0, 0, 0)
   116→      break
   117→    }
   118→    case 'mois':
   119→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   120→      break
   121→    case 'trimestre': {
   122→      const quarter = Math.floor(now.getMonth() / 3)
   123→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   124→      break
   125→    }
   126→    default:
   127→      start = new Date(2000, 0, 1)
   128→  }
   129→
   130→  return { start, end }
   131→}

   133→function truncateString(str: string, maxLen: number) {
   134→  if (!str) return '—'
   135→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   136→}

   138→function formatJsonDetails(details: string | undefined) {
   139→  if (!details) return null
   140→  try {
   141→    const parsed = JSON.parse(details)
   142→    return JSON.stringify(parsed, null, 2)
   143→  } catch {
   144→    return details
   145→  }
   146→}

   150→function downloadCsv(entries: AuditEntry[], filename: string) {
   151→  const headers = [
   152→    'Date/Heure',
   153→    'Utilisateur',
   154→    'Email',
   155→    'Action',
   156→    'Module',
   157→    'Adresse IP',
   158→    'User-Agent',
   159→    'Détails',
   160→  ]
   161→  const rows = entries.map((e) => [
   162→    formatDateTime(e.timestamp),
   163→    e.userName,
   164→    e.userEmail,
   165→    e.action,
   166→    e.module,
   167→    e.ipAddress ?? '',
   168→    e.userAgent ?? '',
   169→    e.details ?? '',
   170→  ])
   171→
   172→  const csvContent = [
   173→    headers.join(';'),
   174→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}”`).join(';')),
   175→  ].join('\n')
   176→
   177→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   178→  const url = URL.createObjectURL(blob)
   179→  const link = document.createElement('a')
   180→  link.href = url
   181→  link.download = filename
   182→  document.body.appendChild(link)
   183→  link.click()
   184→  document.body.removeChild(link)
   185→  URL.revokeObjectURL(url)
   186→}

   523→              <Select
   524→                value={moduleFilter}
   525→                onValueChange={handleFilterChange(setModuleFilter)}
   526→              >
   527→                <SelectTrigger className="h-8 w-[150px] text-xs">
   528→                  <SelectValue />
   529→                </SelectTrigger>
   530→                <SelectContent>
   531→                  <SelectItem value="tous">Tous les modules</SelectItem>
   532→                  {uniqueModules.map((m) => (
   533→                    <SelectItem key={m} value={m}>
   534→                      {m}
   535→                    </SelectItem>
   536→                  ))}
   537→                </SelectContent>
   538→              </Select>
   539→            </div>
   540→          </div>
   541→        </CardContent>
   542→      </Card>
   543→
   544→      {/* ===== TABLE ===== */}
   545→      <Card className={`${isDark ? 'bg-slate-800 border-slate-700' : ''}`}>
   546→        <CardContent className="p-0">
   547→          <div className="overflow-x-auto">
   548→            <Table>
   549→              <TableHeader>
   550→                <TableRow
   551→                  className="hover:bg-transparent"
   552→                  style={{ backgroundColor: isDark ? '#1e293b' : '#fafafa' }}
   553→                >
   554→                  <TableHead
   555→                    className={`w-8 font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   556→                  >
   557→                    <span className="sr-only">Détails</span>
   558→                  </TableHead>
   559→                  <TableHead
   560→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   561→                    style={{ minWidth: 130 }}
   562→                  >
   563→                    Date/Heure
   564→                  </TableHead>
   565→                  <TableHead
   566→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   567→                    style={{ minWidth: 140 }}
   568→                  >
   569→                    Utilisateur
   570→                  </TableHead>
   571→                  <TableHead
   572→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   573→                    style={{ minWidth: 120 }}
   574→                  >
   575→                    Action
   576→                  </TableHead>
   577→                  <TableHead
   578→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   579→                    style={{ minWidth: 100 }}
   580→                  >
   581→                    Module
   582→                  </TableHead>
   583→                  <TableHead
   584→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   585→                    style={{ minWidth: 120 }}
   586→                  >
   587→                    IP Address
   588→                  </TableHead>
   589→                  <TableHead
   590→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   591→                    style={{ minWidth: 160 }}
   592→                  >
   593→                    User-Agent
   594→                  </TableHead>
   595→                  <TableHead
   596→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   597→                    style={{ minWidth: 80 }}
   598→                  >
   599→                    Détails
   600→                  </TableHead>
   601→                </TableRow>
   602→              </TableHeader>
   603→              <TableBody>
   604→                {paginatedEntries.length === 0 ? (
   605→                  <TableRow>
   606→                    <TableCell
   607→                      colSpan={8}
   608→                      className={`py-12 text-center text-sm ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   609→                    >
   610→                      Aucune entrée d\'audit trouvée pour les filtres sélectionnés.
   611→                    </TableCell>
   612→                  </TableRow>
   613→                ) : (
   614→                  paginatedEntries.map((entry, idx) => {
   615→                    const isExpanded = expandedRowId === entry.id
   616→                    return (
   617→                      <Fragment key={entry.id}>
   618→                        <TableRow
   619→                          className={`cursor-pointer select-none transition-colors ${isDark ? 'hover:bg-slate-700' : 'hover:bg-gray-50'} ${
   620→                            idx % 2 === 0
   621→                              ? isDark ? 'bg-slate-800' : 'bg-white'
   622→                              : isDark ? 'bg-slate-800/70' : 'bg-gray-50/40'
   623→                          } ${isExpanded ? (isDark ? 'bg-slate-700/50' : 'bg-gray-50') : ''}`}
   624→                          onClick={() => toggleRowExpand(entry.id)}
   625→                        >
   626→                          {/* Expand indicator */}
   627→                          <TableCell className="w-8 p-2">
   628→                            <div className={`flex h-5 w-5 items-center justify-center rounded ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   629→                              {isExpanded ? (
   630→                                <ChevronUp className="h-3.5 w-3.5" />
   631→                              ) : (
   632→                                <ChevronDown className="h-3.5 w-3.5" />
   633→                              )}
   634→                            </div>
   635→                          </TableCell>
   636→
   637→                          {/* Date/Time */}
   638→                          <TableCell className="text-sm">
   639→                            <div>
   640→                              <div className={isDark ? 'text-slate-100' : 'text-slate-900'}>
   641→                                {formatDateTime(entry.timestamp)}
   642→                              </div>
   643→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   644→                                {getTimeAgo(entry.timestamp)}
   645→                              </div>
   646→                            </div>
   647→                          </TableCell>
   648→
   649→                          {/* User */}
   650→                          <TableCell>
   651→                            <div>
   652→                              <div className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   653→                                {entry.userName}
   654→                              </div>
   655→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   656→                                {entry.userEmail}
   657→                              </div>
   658→                            </div>
   659→                          </TableCell>
   660→
   661→                          {/* Action */}
   662→                          <TableCell>
   663→                            <Badge
   664→                              variant="outline"
   665→                              className={`whitespace-nowrap text-[11px] font-medium ${
   666→                                ACTION_BADGE_COLORS[entry.action] ??
   667→                                (isDark ? 'bg-slate-700 text-slate-300 border-slate-600' : 'bg-gray-100 text-gray-700 border-gray-200')
   668→                              }`}
   669→                            >
   670→                              {entry.action}
   671→                            </Badge>
   672→                          </TableCell>
   673→
   674→                          {/* Module */}
   675→                          <TableCell className={`text-sm ${isDark ? 'text-slate-300' : 'text-gray-600'}`}>
   676→                            {entry.module}
   677→                          </TableCell>
   678→
   679→                          {/* IP */}
   680→                          <TableCell className={`font-mono text-xs ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>
   681→                            {entry.ipAddress ?? '—'}
   682→                          </TableCell>
   683→
   684→                          {/* User-Agent (truncated) */}
   685→                          <TableCell className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   686→                            {truncateString(entry.userAgent ?? '', 28)}
   687→                          </TableCell>
   688→
   689→                          {/* Details button */}
   690→                          <TableCell>
   691→                            {entry.details ? (
   692→                              <Button
   693→                                variant="ghost"
   694→                                size="sm"
   695→                                className={`h-7 text-[11px] ${isDark ? 'text-slate-400 hover:text-slate-300' : 'text-gray-500 hover:text-gray-700'}`}
   696→                                onClick={(e) => {
   697→                                  e.stopPropagation()
   698→                                  toggleRowExpand(entry.id)
   699→                                }}
   700→                              >
   701→                                Voir
   702→                              </Button>
   703→                            ) : (
   704→                              <span className={`text-xs ${isDark ? 'text-slate-600' : 'text-gray-300'}`}>—</span>
   705→                            )}
   706→                          </TableCell>
   707→                        </TableRow>
   708→
   709→                        {/* Expanded details */}
   710→                        {isExpanded && (
   711→                          <TableRow>
   712→                            <TableCell
   713→                              colSpan={8}
   714→                              className="p-0"
   715→                            >
   716→                              <ExpandedDetails entry={entry} />
   717→                            </TableCell>
   718→                          </TableRow>
   719→                        )}
   720→                      </Fragment>
   721→                    )
   722→                  })
   723→                )}
   724→              </TableBody>
   725→            </Table>
   726→          </div>
   727→        </CardContent>
   728→      </Card>
   729→
   730→      {/* ===== PAGINATION ===== */}
   731→      <div className="flex items-center justify-between">
   732→        <p className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   733→          Affichage {Math.min((currentPage - 1) * ITEMS_PER_PAGE + 1, filteredEntries.length)}
   734→          {'–'}
   735→          {Math.min(currentPage * ITEMS_PER_PAGE, filteredEntries.length)}
   736→          {' '}sur {filteredEntries.length}
   737→        </p>
   738→
   739→        <div className="flex items-center gap-1">
   740→          <Button
   741→            variant="outline"
   742→            size="sm"
   743→            className="h-8 w-8 p-0"
   744→            disabled={currentPage <= 1}
   745→            onClick={() => setCurrentPage((p) => p - 1)}
   746→          >
   747→            <ChevronLeft className="h-4 w-4" />
   748→          </Button>
   749→
   750→          {/* Page numbers */}
   751→          {Array.from({ length: totalPages }, (_, i) => i + 1)
   752→            .filter((page) => {
   753→              // Show first, last, current, and neighbors
   754→              if (page === 1 || page === totalPages) return true
   755→              if (Math.abs(page - currentPage) <= 1) return true
   756→              return false
   757→            })
   758→            .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   759→              if (idx > 0) {
   760→                const prev = arr[idx - 1]
   761→                if (page - prev > 1) {
   762→                  acc.push('ellipsis')
   763→                }
   764→              }
   765→              acc.push(page)
   766→              return acc
   767→            }, [])
   768→            .map((item, idx) =>
   769→              item === 'ellipsis' ? (
   770→                <span
   771→                  key={`ellipsis-${idx}`}
   772→                  className={`flex h-8 w-8 items-center justify-center text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   773→                >
   774→                  …
   775→                </span>
   776→              ) : (
   777→                <Button
   778→                  key={item}
   779→                  variant={currentPage === item ? 'default' : 'outline'}
   780→                  size="sm"
   781→                  className={`h-8 w-8 p-0 text-xs ${
   782→                    currentPage === item
   783→                      ? 'text-white'
   784→                      : ''
   785→                  }`}
   786→                  style={
   787→                    currentPage === item
   788→                      ? { backgroundColor: isDark ? '#3B82F6' : '#0F172A', color: '#fff' }
   789→                      : undefined
   790→                  }
   791→                  onClick={() => setCurrentPage(item)}
   792→                >
   793→                  {item}
   794→                </Button>
   795→              )
   796→            )}
   797→
   798→          <Button
   799→            variant="outline"
   800→            size="sm"
   801→            className="h-8 w-8 p-0"
   802→            disabled={currentPage >= totalPages}
   803→            onClick={() => setCurrentPage((p) => p + 1)}
   804→          >
   805→            <ChevronRight className="h-4 w-4" />
   806→          </Button>
   807→        </div>
   808→        </>
   809→      )}
   810→    </div>
   811→  )
