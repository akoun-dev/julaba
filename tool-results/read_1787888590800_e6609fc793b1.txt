   199→function NationalObjectives({ dashboard, isLoading }: { dashboard: DashboardData | null; isLoading: boolean }) {
   200→  const { boTheme } = useBackofficeStore()
   201→  const isDark = boTheme === 'dark'
   202→
   203→  const target = 15000
   204→  const current = dashboard?.totalActors || 0
   205→  const pct = target > 0 ? Math.min(Math.round((current / target) * 100), 100) : 0
   206→  const remaining = Math.max(0, target - current)
   207→
   208→  const [animatedProgress, setAnimatedProgress] = useState(0)
   209→
   210→  useEffect(() => {
   211→    if (!isLoading) {
   212→      const timer = setTimeout(() => setAnimatedProgress(pct), 150)
   213→      return () => clearTimeout(timer)
   214→    }
   215→  }, [pct, isLoading])
   216→
   217→  if (isLoading) {
   218→    return (
   219→      <div className={`rounded-2xl p-6 border ${isDark ? 'bg-slate-800 border-slate-700' : 'bg-white border-slate-200 shadow-[0_1px_3px_rgba(0,0,0,0.04)]'}`}>
   220→        <div className="space-y-3">
   221→          <Skeleton className="h-4 w-48" />
   222→          <Skeleton className="h-2 w-32" />
   223→          <Skeleton className="h-7 w-24" />
   224→          <Skeleton className="h-2.5 w-full" />
   225→        </div>
   226→      </div>
   227→    )
   228→  }
   229→
   230→  return (
   231→    <div className={`rounded-2xl p-6 border ${isDark ? 'bg-slate-800 border-slate-700 shadow-none' : 'bg-white border-slate-200 shadow-[0_1px_3px_rgba(0,0,0,0.04)]'}`}>
   232→      <div className="flex items-center justify-between mb-4">
   233→        <div>
   234→          <h3 className={`font-semibold text-sm ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>Objectifs Nationaux 2026</h3>
   235→          <p className={`text-xs mt-0.5 ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>Cible d'enrôlement marchands</p>
   236→        </div>
   237→        <span className={`text-xs font-semibold px-2.5 py-1 rounded-full ${isDark ? 'bg-emerald-500/15 text-emerald-400' : 'bg-emerald-50 text-emerald-700'}`}>{pct}%</span>
   238→      </div>
   239→      <div className="flex items-center justify-between mb-3">
   240→        <span className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>{current.toLocaleString('fr-FR')}</span>
   241→        <span className={`text-sm ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>sur {target.toLocaleString('fr-FR')}</span>
   242→      </div>
   243→      <Progress
   244→        value={animatedProgress}
   245→        className={`h-2.5 rounded-full ${isDark ? 'bg-slate-700 [&>div]:bg-blue-500' : 'bg-slate-100 [&>div]:bg-blue-500'}`}
   246→      />
   247→      <p className={`text-xs mt-2.5 ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>
   248→        Il reste <span className={`font-medium ${isDark ? 'text-slate-300' : 'text-slate-700'}`}>{remaining.toLocaleString('fr-FR')}</span> acteurs à enrôler
   249→      </p>
   250→    </div>
   251→  )
   252→}
