    55→} from '@/lib/stores/backoffice-store'
    56→import { Skeleton } from '@/components/ui/skeleton'
    57→
    58→// ============== CONSTANTS ==============
    59→const ITEMS_PER_PAGE = 15
    60→
    61→type ActorTypeFilter = 'tous' | 'marchand' | 'producteur' | 'cooperatif'
    62→type ActorStatusFilter = 'tous' | 'actif' | 'suspendu' | 'en_attente' | 'rejete'
    63→
    64→// ============== MAIN COMPONENT ==============
    65→
    66→export function BoActeursScreen() {
    67→  const { actors, updateActorStatus, searchQuery, setSearchQuery, boTheme, loading, fetchAllData } =
    68→    useBackofficeStore()
    69→  const isDark = boTheme === 'dark'
    70→
    71→  // Local state
    72→  const [typeFilter, setTypeFilter] = useState<ActorTypeFilter>('tous')
    73→  const [statusFilter, setStatusFilter] = useState<ActorStatusFilter>('tous')
    74→  const [zoneFilter, setZoneFilter] = useState<string>('tous')
    75→  const [currentPage, setCurrentPage] = useState(1)
    76→  const [selectedActors, setSelectedActors] = useState<Set<string>>(new Set())
    77→  const [showDetailModal, setShowDetailModal] = useState(false)
    78→  const [detailActor, setDetailActor] = useState<BoActor | null>(null)
    79→
    80→  // Unique zones from data
    81→  const zones = useMemo(() => {
    82→    const zoneSet = new Set(actors.map((a) => a.zone))
    83→    return Array.from(zoneSet).sort()
    84→  }, [actors])
    85→
    86→  // Filtered actors
    87→  const filteredActors = useMemo(() => {
    88→    return actors.filter((actor) => {
    89→      // Search filter
    90→      if (searchQuery) {
    91→        const q = searchQuery.toLowerCase()
    92→        const matchesSearch =
    93→          actor.firstName.toLowerCase().includes(q) ||
    94→          actor.lastName.toLowerCase().includes(q) ||
    95→          actor.actorId.toLowerCase().includes(q) ||
    96→          actor.phone.includes(q) ||
    97→          actor.zone.toLowerCase().includes(q)
    98→        if (!matchesSearch) return false
    99→      }
   100→
   101→      // Type filter
   102→      if (typeFilter !== 'tous' && actor.type !== typeFilter) return false
   103→
   104→      // Status filter
   105→      if (statusFilter !== 'tous' && actor.status !== statusFilter) return false
   106→
   107→      // Zone filter
   108→      if (zoneFilter !== 'tous' && actor.zone !== zoneFilter) return false
   109→
   110→      return true
   111→    })
   112→  }, [actors, searchQuery, typeFilter, statusFilter, zoneFilter])
   113→
   114→  // Counters
   115→  const counts = useMemo(() => {
   116→    return {
   117→      total: actors.length,
   118→      marchands: actors.filter((a) => a.type === 'marchand').length,
   119→      producteurs: actors.filter((a) => a.type === 'producteur').length,
   120→      cooperatives: actors.filter((a) => a.type === 'cooperatif').length,
   121→    }
   122→  }, [actors])
   123→
   124→  // Pagination
   125→  const totalPages = Math.max(1, Math.ceil(filteredActors.length / ITEMS_PER_PAGE))
   126→  const paginatedActors = useMemo(() => {
   127→    const start = (currentPage - 1) * ITEMS_PER_PAGE
   128→    return filteredActors.slice(start, start + ITEMS_PER_PAGE)
   129→  }, [filteredActors, currentPage])
   130→
   131→  // Reset page when filters change
   132→  const handleTypeFilterChange = useCallback((val: string) => {
   133→    setTypeFilter(val as ActorTypeFilter)
   134→    setCurrentPage(1)
   135→    setSelectedActors(new Set())
   136→  }, [])
   137→
   138→  const handleStatusFilterChange = useCallback((val: string) => {
   139→    setStatusFilter(val as ActorStatusFilter)
   140→    setCurrentPage(1)
   141→    setSelectedActors(new Set())
   142→  }, [])
   143→
   144→  const handleZoneFilterChange = useCallback((val: string) => {
   145→    setZoneFilter(val)
   146→    setCurrentPage(1)
   147→    setSelectedActors(new Set())
   148→  }, [])
   149→
   150→  const handleSearchChange = useCallback(
   151→    (e: React.ChangeEvent<HTMLInputElement>) => {
   152→      setSearchQuery(e.target.value)
   153→      setCurrentPage(1)
   154→      setSelectedActors(new Set())
   155→    },
   156→    [setSearchQuery]
   157→  )
   158→
   159→  // Selection
   160→  const toggleSelectAll = useCallback(() => {
   161→    if (selectedActors.size === paginatedActors.length) {
   162→      setSelectedActors(new Set())
   163→    } else {
   164→      setSelectedActors(new Set(paginatedActors.map((a) => a.id)))
   165→    }
   166→  }, [selectedActors, paginatedActors])
   167→
   168→  const toggleSelectActor = useCallback((id: string) => {
   169→    setSelectedActors((prev) => {
   170→      const next = new Set(prev)
   171→      if (next.has(id)) {
   172→        next.delete(id)
   173→      } else {
   174→        next.add(id)
   175→      }
   176→      return next
   177→    })
   178→  }, [])
   179→
   180→  const isAllSelected =
   181→    paginatedActors.length > 0 && selectedActors.size === paginatedActors.length
   182→
   183→  // Actions
   184→  const handleViewActor = useCallback((actor: BoActor) => {
   185→    setDetailActor(actor)
   186→    setShowDetailModal(true)
   187→  }, [])
   188→
   189→  const handleSuspend = useCallback(
   190→    (actor: BoActor) => {
   191→      updateActorStatus(actor.id, 'suspendu')
   192→    },
   193→    [updateActorStatus]
   194→  )
   195→
   196→  const handleReactivate = useCallback(
   197→    (actor: BoActor) => {
   198→      updateActorStatus(actor.id, 'actif')
   199→    },
   200→    [updateActorStatus]
   201→  )
   202→
   203→  // CSV Export
   204→  const handleExportCSV = useCallback(() => {
   205→    const selected = actors.filter((a) => selectedActors.has(a.id))
   206→    if (selected.length === 0) return
   207→
   208→    const headers = [
   209→      'ID Acteur',
   210→      'Prénom',
   211→      'Nom',
   212→      'Type',
   213→      'Téléphone',
   214→      'Zone',
   215→      'Statut',
   216→      'Identificateur',
   217→      'Validé par',
   218→      'Date validation',
   219→      'Date création',
   220→    ]
   221→    const rows = selected.map((a) => [
   222→      a.actorId,
   223→      a.firstName,
   224→      a.lastName,
   225→      ACTOR_TYPE_LABELS[a.type],
   226→      a.phone,
   227→      a.zone,
   228→      STATUS_LABELS[a.status],
   229→      a.identificateurName || '',
   230→      a.validatedBy || '',
   231→      a.validatedAt ? new Date(a.validatedAt).toLocaleDateString('fr-FR') : '',
   232→      new Date(a.createdAt).toLocaleDateString('fr-FR'),
   233→    ])
   234→    const csv = [headers.join(';'), ...rows.map((r) => r.join(';'))].join('\n')
   235→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   236→    const url = URL.createObjectURL(blob)
   237→    const link = document.createElement('a')
   238→    link.href = url
   239→    link.download = `acteurs_export_${new Date().toISOString().slice(0, 10)}.csv`
   240→    link.click()
   241→    URL.revokeObjectURL(url)
   242→  }, [actors, selectedActors])
   243→

   262→        <div>
   263→          <h1 className={`text-2xl font-bold tracking-tight ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>GESTION DES ACTEURS</h1>
   264→          <p className="text-sm text-muted-foreground mt-1">Consultez, filtrez et gérez l'ensemble des acteurs enregistrés sur la plateforme Jùlaba.</p>
   265→        </div>
   266→        <div className="flex flex-col items-center justify-center min-h-[300px] gap-4">
   267→          <Loader2 className="h-8 w-8 animate-spin text-blue-500" />
   268→          <p className={`text-sm ${isDark ? 'text-slate-400' : 'text-slate-500'}`}>Chargement des acteurs...</p>
   269→        </div>
   270→      </div>
   271→    )
   272→  }
   273→
   274→  if (actors.length === 0 && !loading) {
   275→    return (
   276→      <div className={'p-6 space-y-6 ' + (isDark ? 'bg-slate-900' : 'bg-[#F8FAFC]')}>

   278→          <h1 className={`text-2xl font-bold tracking-tight ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>GESTION DES ACTEURS</h1>
   279→          <p className="text-sm text-muted-foreground mt-1">Consultez, filtrez et gérez l'ensemble des acteurs enregistrés sur la plateforme Jùlaba.</p>
   280→        </div>
   281→        <div className={`flex flex-col items-center justify-center min-h-[300px] gap-4 rounded-2xl border border-dashed p-12 ${isDark ? 'border-slate-700 bg-slate-800/30' : 'border-slate-300 bg-white'}`}>
   282→          <div className={`flex h-16 w-16 items-center justify-center rounded-full ${isDark ? 'bg-slate-800' : 'bg-slate-100'}`}>
   283→            <Inbox className={`h-8 w-8 ${isDark ? 'text-slate-500' : 'text-slate-400'}`} />
   284→          </div>
   285→          <div className="text-center">
   286→            <p className={`text-lg font-semibold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>Aucun acteur</p>
   287→            <p className={`mt-1 text-sm ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>Aucun acteur n'est encore enregistré sur la plateforme.</p>
   288→          </div>
   289→          <Button variant="outline" onClick={() => fetchAllData()} className="gap-2">
   290→            <RefreshCw className="h-4 w-4" />
   291→            Réessayer
   292→          </Button>
   293→        </div>
   294→      </div>
   295→    )
   296→  }
   297→
   298→  return (
   299→    <div
   300→      className={
   301→        'p-6 space-y-6 ' + (isDark ? 'bg-slate-900' : 'bg-[#F8FAFC]')
   302→      }
   303→    >
   304→      {/* ===== HEADER ===== */}
   305→      <div>
   306→        <h1
   307→          className={`text-2xl font-bold tracking-tight ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   308→        >
   309→          GESTION DES ACTEURS
   310→        </h1>
   311→        <p className="text-sm text-muted-foreground mt-1">
   312→          Consultez, filtrez et gérez l\'ensemble des acteurs enregistrés sur la
   313→          plateforme Jùlaba.
   314→        </p>
   315→      </div>
   316→
   317→      {/* ===== SEARCH ===== */}
   318→      <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   319→        <CardContent className="p-4">
   320→          <div className="relative max-w-md">
   321→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   322→            <Input
   323→              placeholder="Rechercher par nom, téléphone, ID, zone..."
   324→              value={searchQuery}
   325→              onChange={handleSearchChange}
   326→              className="pl-9"
   327→            />
   328→          </div>
   329→        </CardContent>
   330→      </Card>
   331→
   332→      {/* ===== FILTERS ===== */}
   333→      <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   334→        <CardContent className="p-4">
   335→          <div className="flex flex-wrap items-center gap-3">
   336→            <span className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   337→              Filtres :
   338→            </span>
   339→
   340→            <Select value={typeFilter} onValueChange={handleTypeFilterChange}>
   341→              <SelectTrigger className="w-[170px]">
   342→                <SelectValue placeholder="Type" />
   343→              </SelectTrigger>
   344→              <SelectContent>
   345→                <SelectItem value="tous">Tous les types</SelectItem>
   346→                <SelectItem value="marchand">
   347→                  Marchand(e)s
   348→                </SelectItem>
   349→                <SelectItem value="producteur">
   350→                   Producteur(rice)s
   351→                </SelectItem>
   352→                <SelectItem value="cooperatif">
   353→                   Coopératives
   354→                </SelectItem>
   355→              </SelectContent>
   356→            </Select>
   357→
   358→            <Select
   359→              value={statusFilter}
   360→              onValueChange={handleStatusFilterChange}
   361→            >
   362→              <SelectTrigger className="w-[170px]">
   363→                <SelectValue placeholder="Statut" />
   364→              </SelectTrigger>
   365→              <SelectContent>
   366→                <SelectItem value="tous">Tous les statuts</SelectItem>
   367→                <SelectItem value="actif">Actif</SelectItem>
   368→                <SelectItem value="suspendu">Suspendu</SelectItem>
   369→                <SelectItem value="en_attente"><span className="flex items-center gap-1.5"><Hourglass className="w-3.5 h-3.5" /> En attente</span></SelectItem>
   370→                <SelectItem value="rejete">Rejeté</SelectItem>
   371→              </SelectContent>
   372→            </Select>
   373→
   374→            <Select value={zoneFilter} onValueChange={handleZoneFilterChange}>
   375→              <SelectTrigger className="w-[180px]">
   376→                <SelectValue placeholder="Zone" />
   377→              </SelectTrigger>
   378→              <SelectContent>
   379→                <SelectItem value="tous">Toutes les zones</SelectItem>
   380→                {zones.map((z) => (
   381→                  <SelectItem key={z} value={z}>
   382→                    {z}
   383→                  </SelectItem>
   384→                ))}
   385→              </SelectContent>
   386→            </Select>
   387→
   388→            {hasActiveFilters && (
   389→              <Button
   390→                variant="ghost"
   391→                size="sm"
   392→                onClick={resetFilters}
   393→                className="text-muted-foreground hover:text-foreground"
   394→              >
   395→                <X className="size-3.5 mr-1" />
   396→                Réinitialiser
   397→              </Button>
   398→            )}
   399→          </div>
   400→        </CardContent>
   401→      </Card>
   402→
   403→      {/* ===== COUNTER BAR ===== */}
   404→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   405→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   406→          <CardContent className="p-4 flex items-center gap-3">
   407→            <div className={`flex items-center justify-center size-10 rounded-lg ${isDark ? 'bg-slate-700' : 'bg-gray-100'}`}>
   408→              <Users className={`size-5 ${isDark ? 'text-slate-300' : 'text-gray-600'}`} />
   409→            </div>
   410→            <div>
   411→              <p className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   412→                {counts.total}
   413→              </p>
   414→              <p className="text-xs text-muted-foreground">Total acteurs</p>
   415→            </div>
   416→          </CardContent>
   417→        </Card>
   418→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   419→          <CardContent className="p-4 flex items-center gap-3">
   420→            <div className={`flex items-center justify-center size-10 rounded-lg ${isDark ? 'bg-orange-500/10' : 'bg-orange-50'}`}>
   421→              <Store className="size-5 text-orange-600" />
   422→            </div>
   423→            <div>
   424→              <p className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   425→                {counts.marchands}
   426→              </p>
   427→              <p className="text-xs text-muted-foreground">Marchands</p>
   428→            </div>
   429→          </CardContent>
   430→        </Card>
   431→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   432→          <CardContent className="p-4 flex items-center gap-3">
   433→            <div className={`flex items-center justify-center size-10 rounded-lg ${isDark ? 'bg-emerald-500/10' : 'bg-green-50'}`}>
   434→              <Wheat className="size-5 text-green-600" />
   435→            </div>
   436→            <div>
   437→              <p className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   438→                {counts.producteurs}
   439→              </p>
   440→              <p className="text-xs text-muted-foreground">Producteurs</p>
   441→            </div>
   442→          </CardContent>
   443→        </Card>
   444→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   445→          <CardContent className="p-4 flex items-center gap-3">
   446→            <div className={`flex items-center justify-center size-10 rounded-lg ${isDark ? 'bg-amber-500/10' : 'bg-amber-50'}`}>
   447→              <Handshake className="size-5 text-amber-600" />
   448→            </div>
   449→            <div>
   450→              <p className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   451→                {counts.cooperatives}
   452→              </p>
   453→              <p className="text-xs text-muted-foreground">Coopératives</p>
   454→            </div>
   455→          </CardContent>
   456→        </Card>
   457→      </div>
   458→
   459→      {/* ===== BULK ACTIONS ===== */}

   467→              <Button size="sm" variant="outline" onClick={handleExportCSV}>

   470→              </Button>

   474→              </Button>

   476→          </CardContent>

   482→        <CardContent className="p-0">
   483→          <div className="max-h-[620px] overflow-y-auto">

   488→                    <Checkbox
   489→                      checked={isAllSelected}
   490→                      onCheckedChange={toggleSelectAll}
   491→                      aria-label="Sélectionner tout"
   492→                    />
   493→                  </th>

   495→                    ID
   496→                  </th>
   497→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   498→                    Nom complet
   499→                  </th>
   500→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
