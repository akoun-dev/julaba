    77→function formatDateTime(iso: string) {
    78→  const d = new Date(iso)
    79→  return d.toLocaleDateString('fr-FR', {
    80→    day: '2-digit',
    81→    month: '2-digit',
    82→    year: 'numeric',
    83→    hour: '2-digit',
    84→    minute: '2-digit',
    85→  })
    86→}
    87→
    88→function getTimeAgo(iso: string) {
    89→  const now = Date.now()
    90→  const then = new Date(iso).getTime()
    91→  const diff = now - then
    92→  const minutes = Math.floor(diff / 60000)
    93→  if (minutes < 1) return 'à l\'instant'
    94→  if (minutes < 60) return `il y a ${minutes}min`
    95→  const hours = Math.floor(minutes / 60)
    96→  if (hours < 24) return `il y a ${hours}h`
    97→  const days = Math.floor(hours / 24)
    98→  if (days < 7) return `il y a ${days}j`
    99→  return formatDateTime(iso)
   100→}

   102→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   103→  const now = new Date()
   104→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   105→  let start: Date
   106→
   107→  switch (dateRange) {
   108→    case 'aujourdhui':
   109→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   110→      break
   111→    case 'semaine': {
   112→      const dayOfWeek = now.getDay() || 7
   113→      start = new Date(now)
   114→      start.setDate(now.getDate() - dayOfWeek + 1)
   115→      start.setHours(0, 0, 0, 0)
   116→      break
   117→    }
   118→    case 'mois':
   119→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   120→      break
   121→    case 'trimestre': {
   122→      const quarter = Math.floor(now.getMonth() / 3)
   123→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   124→      break
   125→    }
   126→    default:
   127→      start = new Date(2000, 0, 1)
   128→  }
   129→
   130→  return { start, end }
   131→}

   133→function truncateString(str: string, maxLen: number) {
   134→  if (!str) return '—'
   135→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   136→}

   138→function formatJsonDetails(details: string | undefined) {
   139→  if (!details) return null
   140→  try {
   141→    const parsed = JSON.parse(details)
   142→    return JSON.stringify(parsed, null, 2)
   143→  } catch {
   144→    return details
   145→  }
   146→}

   150→function downloadCsv(entries: AuditEntry[], filename: string) {
   151→  const headers = [
   152→    'Date/Heure',
   153→    'Utilisateur',
   154→    'Email',
   155→    'Action',
   156→    'Module',
   157→    'Adresse IP',
   158→    'User-Agent',
   159→    'Détails',
   160→  ]
   161→  const rows = entries.map((e) => [
   162→    formatDateTime(e.timestamp),
   163→    e.userName,
   164→    e.userEmail,
   165→    e.action,
   166→    e.module,
   167→    e.ipAddress ?? '',
   168→    e.userAgent ?? '',
   169→    e.details ?? '',
   170→  ])
   171→
   172→  const csvContent = [
   173→    headers.join(';'),
   174→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}”`).join(';')),
   175→  ].join('\n')
   176→
   177→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   178→  const url = URL.createObjectURL(blob)
   179→  const link = document.createElement('a')
   180→  link.href = url
   181→  link.download = filename
   182→  document.body.appendChild(link)
   183→  link.click()
   184→  document.body.removeChild(link)
   185→  URL.revokeObjectURL(url)
   186→}

   524→              </Label>
   525→              <Select
   526→                value={moduleFilter}
   527→                onValueChange={handleFilterChange(setModuleFilter)}
   528→              >
   529→                <SelectTrigger className="h-8 w-[150px] text-xs">
   530→                  <SelectValue />
   531→                </SelectTrigger>
   532→                <SelectContent>
   533→                  <SelectItem value="tous">Tous les modules</SelectItem>
   534→                  {uniqueModules.map((m) => (
   535→                    <SelectItem key={m} value={m}>
   536→                      {m}
   537→                    </SelectItem>
   538→                  ))}
   539→                </SelectContent>
   540→              </Select>
   541→            </div>
   542→          </div>
   543→        </CardContent>
   544→      </Card>
   545→
   546→      {/* ===== TABLE ===== */}
   547→      <Card className={`${isDark ? 'bg-slate-800 border-slate-700' : ''}`}>
   548→        <CardContent className="p-0">
   549→          <div className="overflow-x-auto">
   550→            <Table>
   551→              <TableHeader>
   552→                <TableRow
   553→                  className="hover:bg-transparent"
   554→                  style={{ backgroundColor: isDark ? '#1e293b' : '#fafafa' }}
   555→                >
   556→                  <TableHead
   557→                    className={`w-8 font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   558→                  >
   559→                    <span className="sr-only">Détails</span>
   560→                  </TableHead>
   561→                  <TableHead
   562→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   563→                    style={{ minWidth: 130 }}
   564→                  >
   565→                    Date/Heure
   566→                  </TableHead>
   567→                  <TableHead
   568→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   569→                    style={{ minWidth: 140 }}
   570→                  >
   571→                    Utilisateur
   572→                  </TableHead>
   573→                  <TableHead
   574→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   575→                    style={{ minWidth: 120 }}
   576→                  >
   577→                    Action
   578→                  </TableHead>
   579→                  <TableHead
   580→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   581→                    style={{ minWidth: 100 }}
   582→                  >
   583→                    Module
   584→                  </TableHead>
   585→                  <TableHead
   586→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   587→                    style={{ minWidth: 120 }}
   588→                  >
   589→                    IP Address
   590→                  </TableHead>
   591→                  <TableHead
   592→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   593→                    style={{ minWidth: 160 }}
   594→                  >
   595→                    User-Agent
   596→                  </TableHead>
   597→                  <TableHead
   598→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   599→                    style={{ minWidth: 80 }}
   600→                  >
   601→                    Détails
   602→                  </TableHead>
   603→                </TableRow>
   604→              </TableHeader>
   605→              <TableBody>
   606→                {paginatedEntries.length === 0 ? (
   607→                  <TableRow>
   608→                    <TableCell
   609→                      colSpan={8}
   610→                      className={`py-12 text-center text-sm ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   611→                    >
   612→                      Aucune entrée d\'audit trouvée pour les filtres sélectionnés.
   613→                    </TableCell>
   614→                  </TableRow>
   615→                ) : (
   616→                  paginatedEntries.map((entry, idx) => {
   617→                    const isExpanded = expandedRowId === entry.id
   618→                    return (
   619→                      <Fragment key={entry.id}>
   620→                        <TableRow
   621→                          className={`cursor-pointer select-none transition-colors ${isDark ? 'hover:bg-slate-700' : 'hover:bg-gray-50'} ${
   622→                            idx % 2 === 0
   623→                              ? isDark ? 'bg-slate-800' : 'bg-white'
   624→                              : isDark ? 'bg-slate-800/70' : 'bg-gray-50/40'
   625→                          } ${isExpanded ? (isDark ? 'bg-slate-700/50' : 'bg-gray-50') : ''}`}
   626→                          onClick={() => toggleRowExpand(entry.id)}
   627→                        >
   628→                          {/* Expand indicator */}
   629→                          <TableCell className="w-8 p-2">
   630→                            <div className={`flex h-5 w-5 items-center justify-center rounded ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   631→                              {isExpanded ? (
   632→                                <ChevronUp className="h-3.5 w-3.5" />
   633→                              ) : (
   634→                                <ChevronDown className="h-3.5 w-3.5" />
   635→                              )}
   636→                            </div>
   637→                          </TableCell>
   638→
   639→                          {/* Date/Time */}
   640→                          <TableCell className="text-sm">
   641→                            <div>
   642→                              <div className={isDark ? 'text-slate-100' : 'text-slate-900'}>
   643→                                {formatDateTime(entry.timestamp)}
   644→                              </div>
   645→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   646→                                {getTimeAgo(entry.timestamp)}
   647→                              </div>
   648→                            </div>
   649→                          </TableCell>
   650→
   651→                          {/* User */}
   652→                          <TableCell>
   653→                            <div>
   654→                              <div className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   655→                                {entry.userName}
   656→                              </div>
   657→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   658→                                {entry.userEmail}
   659→                              </div>
   660→                            </div>
   661→                          </TableCell>
   662→
   663→                          {/* Action */}
   664→                          <TableCell>
   665→                            <Badge
   666→                              variant="outline"
   667→                              className={`whitespace-nowrap text-[11px] font-medium ${
   668→                                ACTION_BADGE_COLORS[entry.action] ??
   669→                                (isDark ? 'bg-slate-700 text-slate-300 border-slate-600' : 'bg-gray-100 text-gray-700 border-gray-200')
   670→                              }`}
   671→                            >
   672→                              {entry.action}
   673→                            </Badge>
   674→                          </TableCell>
   675→
   676→                          {/* Module */}
   677→                          <TableCell className={`text-sm ${isDark ? 'text-slate-300' : 'text-gray-600'}`}>
   678→                            {entry.module}
   679→                          </TableCell>
   680→
   681→                          {/* IP */}
   682→                          <TableCell className={`font-mono text-xs ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>
   683→                            {entry.ipAddress ?? '—'}
   684→                          </TableCell>
   685→
   686→                          {/* User-Agent (truncated) */}
   687→                          <TableCell className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   688→                            {truncateString(entry.userAgent ?? '', 28)}
   689→                          </TableCell>
   690→
   691→                          {/* Details button */}
   692→                          <TableCell>
   693→                            {entry.details ? (
   694→                              <Button
   695→                                variant="ghost"
   696→                                size="sm"
   697→                                className={`h-7 text-[11px] ${isDark ? 'text-slate-400 hover:text-slate-300' : 'text-gray-500 hover:text-gray-700'}`}
   698→                                onClick={(e) => {
   699→                                  e.stopPropagation()
   700→                                  toggleRowExpand(entry.id)
   701→                                }}
   702→                              >
   703→                                Voir
   704→                              </Button>
   705→                            ) : (
   706→                              <span className={`text-xs ${isDark ? 'text-slate-600' : 'text-gray-300'}`}>—</span>
   707→                            )}
   708→                          </TableCell>
   709→                        </TableRow>
   710→
   711→                        {/* Expanded details */}
   712→                        {isExpanded && (
   713→                          <TableRow>
   714→                            <TableCell
   715→                              colSpan={8}
   716→                              className="p-0"
   717→                            >
   718→                              <ExpandedDetails entry={entry} />
   719→                            </TableCell>
   720→                          </TableRow>
   721→                        )}
   722→                      </Fragment>
   723→                    )
   724→                  })
   725→                )}
   726→              </TableBody>
   727→            </Table>
   728→          </div>
   729→        </CardContent>
   730→      </Card>
   731→
   732→      {/* ===== PAGINATION ===== */}
   733→      <div className="flex items-center justify-between">
   734→        <p className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   735→          Affichage {Math.min((currentPage - 1) * ITEMS_PER_PAGE + 1, filteredEntries.length)}
   736→          {'–'}
   737→          {Math.min(currentPage * ITEMS_PER_PAGE, filteredEntries.length)}
   738→          {' '}sur {filteredEntries.length}
   739→        </p>
   740→
   741→        <div className="flex items-center gap-1">
   742→          <Button
   743→            variant="outline"
   744→            size="sm"
   745→            className="h-8 w-8 p-0"
   746→            disabled={currentPage <= 1}
   747→            onClick={() => setCurrentPage((p) => p - 1)}
   748→          >
   749→            <ChevronLeft className="h-4 w-4" />
   750→          </Button>
   751→
   752→          {/* Page numbers */}
   753→          {Array.from({ length: totalPages }, (_, i) => i + 1)
   754→            .filter((page) => {
   755→              // Show first, last, current, and neighbors
   756→              if (page === 1 || page === totalPages) return true
   757→              if (Math.abs(page - currentPage) <= 1) return true
   758→              return false
   759→            })
   760→            .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   761→              if (idx > 0) {
   762→                const prev = arr[idx - 1]
   763→                if (page - prev > 1) {
   764→                  acc.push('ellipsis')
   765→                }
   766→              }
   767→              acc.push(page)
   768→              return acc
   769→            }, [])
   770→            .map((item, idx) =>
   771→              item === 'ellipsis' ? (
   772→                <span
   773→                  key={`ellipsis-${idx}`}
   774→                  className={`flex h-8 w-8 items-center justify-center text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   775→                >
   776→                  …
   777→                </span>
   778→              ) : (
   779→                <Button
   780→                  key={item}
   781→                  variant={currentPage === item ? 'default' : 'outline'}
   782→                  size="sm"
   783→                  className={`h-8 w-8 p-0 text-xs ${
   784→                    currentPage === item
   785→                      ? 'text-white'
   786→                      : ''
   787→                  }`}
   788→                  style={
   789→                    currentPage === item
   790→                      ? { backgroundColor: isDark ? '#3B82F6' : '#0F172A', color: '#fff' }
   791→                      : undefined
   792→                  }
   793→                  onClick={() => setCurrentPage(item)}
   794→                >
   795→                  {item}
   796→                </Button>
   797→              )
   798→            )}
   799→
   800→          <Button
   801→            variant="outline"
   802→            size="sm"
   803→            className="h-8 w-8 p-0"
   804→            disabled={currentPage >= totalPages}
   805→            onClick={() => setCurrentPage((p) => p + 1)}
   806→          >
   807→            <ChevronRight className="h-4 w-4" />
   808→          </Button>
   809→        </div>
   810→        </>
   811→      )}
   812→    </div>
   813→  )
   814→}
