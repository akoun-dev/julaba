   199→   199→function NationalObjectives({ dashboard, isLoading }: { dashboard: DashboardData | null; isLoading: boolean }) {
   200→   200→  const { boTheme } = useBackofficeStore()
   201→   201→  const isDark = boTheme === 'dark'
   202→   202→
   203→   203→  const target = 15000
   204→   204→  const current = dashboard?.totalActors || 0
   205→   205→  const pct = target > 0 ? Math.min(Math.round((current / target) * 100), 100) : 0
   206→   206→  const remaining = Math.max(0, target - current)
   207→   207→
   208→   208→  const [animatedProgress, setAnimatedProgress] = useState(0)
   209→   209→
   210→   210→  useEffect(() => {
   211→   211→    if (!isLoading) {
   212→   212→      const timer = setTimeout(() => setAnimatedProgress(pct), 150)
   213→   213→      return () => clearTimeout(timer)
   214→   214→    }
   215→   215→  }, [pct, isLoading])
   216→   216→
   217→   217→  if (isLoading) {
   218→   218→    return (
   219→   219→      <div className={`rounded-2xl p-6 border ${isDark ? 'bg-slate-800 border-slate-700' : 'bg-white border-slate-200 shadow-[0_1px_3px_rgba(0,0,0,0.04)]'}`}>
   220→   220→        <div className="space-y-3">
   221→   221→          <Skeleton className="h-4 w-48" />
   222→   222→          <Skeleton className="h-2 w-32" />
   223→   223→          <Skeleton className="h-7 w-24" />
   224→   224→          <Skeleton className="h-2.5 w-full" />
   225→   225→        </div>
   226→   226→      </div>
   227→   227→    )
   228→   228→  }
   229→   229→
   230→   230→  return (
   231→   231→    <div className={`rounded-2xl p-6 border ${isDark ? 'bg-slate-800 border-slate-700 shadow-none' : 'bg-white border-slate-200 shadow-[0_1px_3px_rgba(0,0,0,0.04)]'}`}>

   236→   236→        </div>
   237→   237→        <span className={`text-xs font-semibold px-2.5 py-1 rounded-full ${isDark ? 'bg-emerald-500/15 text-emerald-400' : 'bg-emerald-50 text-emerald-700'}`}>{pct}%</span>
   238→   238→      </div>
   239→   239→      <div className="flex items-center justify-between mb-3">
   240→   240→        <span className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>{current.toLocaleString('fr-FR')}</span>
   241→   241→        <span className={`text-sm ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>sur {target.toLocaleString('fr-FR')}</span>
   242→   242→      </div>
   243→   243→      <Progress
   244→   244→        value={animatedProgress}
   245→   245→        className={`h-2.5 rounded-full ${isDark ? 'bg-slate-700 [&>div]:bg-blue-500' : 'bg-slate-100 [&>div]:bg-blue-500'}`}
