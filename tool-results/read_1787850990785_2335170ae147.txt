    76→function formatDateTime(iso: string) {
    77→  const d = new Date(iso)
    78→  return d.toLocaleDateString('fr-FR', {
    79→    day: '2-digit',
    80→    month: '2-digit',
    81→    year: 'numeric',
    82→    hour: '2-digit',
    83→    minute: '2-digit',
    84→  })
    85→}
    86→
    87→function getTimeAgo(iso: string) {
    88→  const now = Date.now()
    89→  const then = new Date(iso).getTime()
    90→  const diff = now - then
    91→  const minutes = Math.floor(diff / 60000)
    92→  if (minutes < 1) return 'à l\'instant'
    93→  if (minutes < 60) return `il y a ${minutes}min`
    94→  const hours = Math.floor(minutes / 60)
    95→  if (hours < 24) return `il y a ${hours}h`
    96→  const days = Math.floor(hours / 24)
    97→  if (days < 7) return `il y a ${days}j`
    98→  return formatDateTime(iso)
    99→}

   101→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   102→  const now = new Date()
   103→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   104→  let start: Date
   105→
   106→  switch (dateRange) {
   107→    case 'aujourdhui':
   108→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   109→      break
   110→    case 'semaine': {
   111→      const dayOfWeek = now.getDay() || 7
   112→      start = new Date(now)
   113→      start.setDate(now.getDate() - dayOfWeek + 1)
   114→      start.setHours(0, 0, 0, 0)
   115→      break
   116→    }
   117→    case 'mois':
   118→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   119→      break
   120→    case 'trimestre': {
   121→      const quarter = Math.floor(now.getMonth() / 3)
   122→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   123→      break
   124→    }
   125→    default:
   126→      start = new Date(2000, 0, 1)
   127→  }
   128→
   129→  return { start, end }
   130→}

   132→function truncateString(str: string, maxLen: number) {
   133→  if (!str) return '—'
   134→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   135→}

   137→function formatJsonDetails(details: string | undefined) {
   138→  if (!details) return null
   139→  try {
   140→    const parsed = JSON.parse(details)
   141→    return JSON.stringify(parsed, null, 2)
   142→  } catch {
   143→    return details
   144→  }
   145→}

   149→function downloadCsv(entries: AuditEntry[], filename: string) {
   150→  const headers = [
   151→    'Date/Heure',
   152→    'Utilisateur',
   153→    'Email',
   154→    'Action',
   155→    'Module',
   156→    'Adresse IP',
   157→    'User-Agent',
   158→    'Détails',
   159→  ]
   160→  const rows = entries.map((e) => [
   161→    formatDateTime(e.timestamp),
   162→    e.userName,
   163→    e.userEmail,
   164→    e.action,
   165→    e.module,
   166→    e.ipAddress ?? '',
   167→    e.userAgent ?? '',
   168→    e.details ?? '',
   169→  ])
   170→
   171→  const csvContent = [
   172→    headers.join(';'),
   173→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(';')),
   174→  ].join('\n')
   175→
   176→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   177→  const url = URL.createObjectURL(blob)
   178→  const link = document.createElement('a')
   179→  link.href = url
   180→  link.download = filename
   181→  document.body.appendChild(link)
   182→  link.click()
   183→  document.body.removeChild(link)
   184→  URL.revokeObjectURL(url)
   185→}

   592→                    const isExpanded = expandedRowId === entry.id
   593→                    return (
   594→                      <Fragment key={entry.id}>
   595→                        <TableRow
   596→                          className={`cursor-pointer select-none transition-colors hover:bg-gray-50 ${
   597→                            idx % 2 === 0 ? 'bg-white' : 'bg-gray-50/40'
   598→                          } ${isExpanded ? 'bg-gray-50' : ''}`}
   599→                          onClick={() => toggleRowExpand(entry.id)}
   600→                        >
   601→                          {/* Expand indicator */}
   602→                          <TableCell className="w-8 p-2">
   603→                            <div className="flex h-5 w-5 items-center justify-center rounded text-gray-400">
   604→                              {isExpanded ? (
   605→                                <ChevronUp className="h-3.5 w-3.5" />
   606→                              ) : (
   607→                                <ChevronDown className="h-3.5 w-3.5" />
   608→                              )}
   609→                            </div>
   610→                          </TableCell>
   611→
   612→                          {/* Date/Time */}
   613→                          <TableCell className="text-sm">
   614→                            <div>
   615→                              <div className="text-gray-700" style={{ color: BO_COLOR }}>
   616→                                {formatDateTime(entry.timestamp)}
   617→                              </div>
   618→                              <div className="text-[11px] text-gray-400">
   619→                                {getTimeAgo(entry.timestamp)}
   620→                              </div>
   621→                            </div>
   622→                          </TableCell>
   623→
   624→                          {/* User */}
   625→                          <TableCell>
   626→                            <div>
   627→                              <div className="text-sm font-medium" style={{ color: BO_COLOR }}>
   628→                                {entry.userName}
   629→                              </div>
   630→                              <div className="text-[11px] text-gray-400">
   631→                                {entry.userEmail}
   632→                              </div>
   633→                            </div>
   634→                          </TableCell>
   635→
   636→                          {/* Action */}
   637→                          <TableCell>
   638→                            <Badge
   639→                              variant="outline"
   640→                              className={`whitespace-nowrap text-[11px] font-medium ${
   641→                                ACTION_BADGE_COLORS[entry.action] ??
   642→                                'bg-gray-100 text-gray-700 border-gray-200'
   643→                              }`}
   644→                            >
   645→                              {entry.action}
   646→                            </Badge>
   647→                          </TableCell>
   648→
   649→                          {/* Module */}
   650→                          <TableCell className="text-sm text-gray-600">
   651→                            {entry.module}
   652→                          </TableCell>
   653→
   654→                          {/* IP */}
   655→                          <TableCell className="font-mono text-xs text-gray-500">
   656→                            {entry.ipAddress ?? '—'}
   657→                          </TableCell>
   658→
   659→                          {/* User-Agent (truncated) */}
   660→                          <TableCell className="text-xs text-gray-400">
   661→                            {truncateString(entry.userAgent ?? '', 28)}
   662→                          </TableCell>
   663→
   664→                          {/* Details button */}
   665→                          <TableCell>
   666→                            {entry.details ? (
   667→                              <Button
   668→                                variant="ghost"
   669→                                size="sm"
   670→                                className="h-7 text-[11px] text-gray-500 hover:text-gray-700"
   671→                                onClick={(e) => {
   672→                                  e.stopPropagation()
   673→                                  toggleRowExpand(entry.id)
   674→                                }}
   675→                              >
   676→                                Voir
   677→                              </Button>
   678→                            ) : (
   679→                              <span className="text-xs text-gray-300">—</span>
   680→                            )}
   681→                          </TableCell>
   682→                        </TableRow>
   683→
   684→                        {/* Expanded details */}
   685→                        {isExpanded && (
   686→                          <TableRow>
   687→                            <TableCell
   688→                              colSpan={8}
   689→                              className="p-0"
   690→                            >
   691→                              <ExpandedDetails entry={entry} />
   692→                            </TableCell>
   693→                          </TableRow>
   694→                        )}
   695→                      </Fragment>
   696→                    )
   697→                  })
   698→                )}
   699→              </TableBody>
   700→            </Table>
   701→          </div>
   702→        </CardContent>
   703→      </Card>
   704→
   705→      {/* ===== PAGINATION ===== */}
   706→      <div className="flex items-center justify-between">
   707→        <p className="text-xs text-gray-400">
   708→          Affichage {Math.min((currentPage - 1) * ITEMS_PER_PAGE + 1, filteredEntries.length)}
   709→          {'–'}
   710→          {Math.min(currentPage * ITEMS_PER_PAGE, filteredEntries.length)}
   711→          {' '}sur {filteredEntries.length}
   712→        </p>
   713→
   714→        <div className="flex items-center gap-1">
   715→          <Button
   716→            variant="outline"
   717→            size="sm"
   718→            className="h-8 w-8 p-0"
   719→            disabled={currentPage <= 1}
   720→            onClick={() => setCurrentPage((p) => p - 1)}
   721→          >
   722→            <ChevronLeft className="h-4 w-4" />
   723→          </Button>
   724→
   725→          {/* Page numbers */}
   726→          {Array.from({ length: totalPages }, (_, i) => i + 1)
   727→            .filter((page) => {
   728→              // Show first, last, current, and neighbors
   729→              if (page === 1 || page === totalPages) return true
   730→              if (Math.abs(page - currentPage) <= 1) return true
   731→              return false
   732→            })
   733→            .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   734→              if (idx > 0) {
   735→                const prev = arr[idx - 1]
   736→                if (page - prev > 1) {
   737→                  acc.push('ellipsis')
   738→                }
   739→              }
   740→              acc.push(page)
   741→              return acc
   742→            }, [])
   743→            .map((item, idx) =>
   744→              item === 'ellipsis' ? (
   745→                <span
   746→                  key={`ellipsis-${idx}`}
   747→                  className="flex h-8 w-8 items-center justify-center text-xs text-gray-400"
   748→                >
   749→                  …
   750→                </span>
   751→              ) : (
   752→                <Button
   753→                  key={item}
   754→                  variant={currentPage === item ? 'default' : 'outline'}
   755→                  size="sm"
   756→                  className={`h-8 w-8 p-0 text-xs ${
   757→                    currentPage === item
   758→                      ? 'text-white'
   759→                      : ''
   760→                  }`}
   761→                  style={
   762→                    currentPage === item
   763→                      ? { backgroundColor: BO_COLOR, color: '#fff' }
   764→                      : undefined
   765→                  }
   766→                  onClick={() => setCurrentPage(item)}
   767→                >
   768→                  {item}
   769→                </Button>
   770→              )
   771→            )}
   772→
   773→          <Button
   774→            variant="outline"
   775→            size="sm"
   776→            className="h-8 w-8 p-0"
   777→            disabled={currentPage >= totalPages}
   778→            onClick={() => setCurrentPage((p) => p + 1)}
   779→          >
   780→            <ChevronRight className="h-4 w-4" />
   781→          </Button>
   782→        </div>
   783→      </div>
   784→    </div>
   785→  )
   786→}
