    63→// Simple hash utility (same as auth-screen)
    64→const simpleHash = (str: string) => {
    65→  let hash = 0
    66→  for (let i = 0; i < str.length; i++) {
    67→    const char = str.charCodeAt(i)
    68→    hash = ((hash << 5) - hash) + char
    69→    hash |= 0
    70→  }
    71→  return hash.toString()
    72→}
    73→const patternToHash = (pattern: number[]) => simpleHash(pattern.join('-'))

   923→                {!visualDone && (

   944→      {/* ======================== BOTTOM ACTION BAR ======================== */}
   945→      <div className="fixed bottom-0 left-0 right-0 z-30 bg-white border-t px-4 py-3 shadow-[0_-2px_10px_rgba(0,0,0,0.05)]">

   947→          {currentStep === 1 && (
   948→            <Button
   949→              type="button"

   962→              className={`flex-1 gap-2 ${txt}`}
   963→              style={{ borderColor: IDENT_COLOR, color: IDENT_COLOR }}

   965→            >
   966→              <ArrowLeft className="size-4" /> Précédent

   968→          )}
   969→          {currentStep < TOTAL_STEPS && (
   970→            <Button
   971→              type="button"
   972→              className={`flex-1 gap-2 text-white ${txt}`}
   973→              style={{ backgroundColor: IDENT_COLOR }}
   974→              onClick={goNext}
   975→            >
   976→              Suivant <ArrowRight className="size-4" />
   977→            </Button>
   978→          )}
   979→          {currentStep === TOTAL_STEPS && (
   980→            <Button
   981→              type="button"
   982→              className={`flex-1 gap-2 text-white ${txt}`}
   983→              style={{ backgroundColor: IDENT_COLOR }}

   985→              disabled={submitting}
   986→            >
   987→              {submitting ? <Loader2 className="size-4 animate-spin" /> : <Upload className="size-4" />}
   988→              {submitting ? 'Envoi en cours...' : '📤 Soumettre pour validation'}
   989→            </Button>
   990→          )}
   991→        </div>
   992→      </div>
   993→    </div>
   994→  )
   995→}
