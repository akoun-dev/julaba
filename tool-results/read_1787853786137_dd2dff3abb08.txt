    41→// ============== SUB COMPONENTS ==============
    42→
    43→function SectionHeader({ sectionKey, icon, title, sectionStates, toggleEdit, handleSave, handleCancel }: {
    44→  sectionKey: string
    45→  icon: React.ReactNode
    46→  title: string
    47→  sectionStates: Record<string, SectionState>
    48→  toggleEdit: (section: string) => void
    49→  handleSave: (section: string) => void
    50→  handleCancel: (section: string) => void
    51→}) {
    52→  const state = sectionStates[sectionKey]
    53→  return (
    54→    <div className="flex items-center justify-between">
    55→      <CardTitle className="text-sm font-semibold flex items-center gap-2" style={{ color: BO_COLOR }}>
    56→        {icon}
    57→        {title}
    58→      </CardTitle>
    59→      <div className="flex gap-2">
    60→        {state?.editing ? (
    61→          <>
    62→            <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={() => handleCancel(sectionKey)}>
    63→              <X className="h-3 w-3 mr-1" /> Annuler
    64→            </Button>
    65→            <Button size="sm" className="h-8 text-xs" onClick={() => handleSave(sectionKey)} disabled={state?.saving}>
    66→              {state?.saving ? (
    67→                <span className="h-3 w-3 border-2 border-white/30 border-t-white rounded-full animate-spin" />
    68→              ) : (
    69→                <><Save className="h-3 w-3 mr-1" /> Enregistrer</>
    70→              )}
    71→            </Button>
    72→          </>
    73→        ) : (
    74→          <Button variant="outline" size="sm" className="h-8 text-xs" onClick={() => toggleEdit(sectionKey)}>
    75→            <Pencil className="h-3 w-3 mr-1" /> Modifier
    76→          </Button>
    77→        )}
    78→      </div>
    79→    </div>
    80→  )
    81→}
