    29→    29→const simpleHash = (str: string) => {
    30→    30→  let hash = 0
    31→    31→  for (let i = 0; i < str.length; i++) {
    32→    32→    const char = str.charCodeAt(i)
    33→    33→    hash = ((hash << 5) - hash) + char
    34→    34→    hash |= 0
    35→    35→  }
    36→    36→  return hash.toString()
    37→    37→}
