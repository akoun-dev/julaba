   126→        stepRef.current = 'phone'

   193→          playBeep('error')
   194→          tataSpeak('Problème micro. Réessayez.')
   195→          setError('Erreur micro.')
   196→        }
   197→      },
   198→      onEnd: () => { setIsListening(false) },
   199→    })

   214→      if (stored.authMethod === 'pattern') {
   215→        setAuthMethod('pattern')

   217→        stepRef.current = 'pattern-login'

   224→        setAuthMethod('pin')
   225→        setStep('login-pin')
   226→        stepRef.current = 'login-pin'

   230→      setMode('register')
   231→      modeRef.current = 'register'
   232→      setStep('choose-method')

   250→      stepRef.current = 'pattern-create'

   269→      stepRef.current = 'login-pin'

   273→      stepRef.current = 'pattern-login'

   284→      setStep('pattern-confirm')

   292→      // Match!
   293→      haptic('success')
   294→      setPatternSuccess(true)
   295→      playBeep('success')
   296→      const id = crypto.randomUUID()
   297→      const merchantData: MerchantData = {
   298→        id,
   299→        firstName: firstName || 'Awa',
   300→        phone,
   301→        pinHash: '', // no PIN set

   331→      setPatternSuccess(true)

   337→      playBeep('error')
   338→      setPatternError(true)
   339→      setError('Schéma incorrect.')

   365→              id, firstName, phone,
   366→              pinHash: simpleHash(confirmPin),
   367→              authMethod: 'pin',
   368→            }
   369→            saveMerchant(merchantData)
   370→            localStorage.setItem('julaba-last-name', firstName)
   371→            playBeep('success')
   372→            haptic('success')
   373→            tataSpeak(`Compte créé ! Bonjour Maman ${firstName} !`)

   400→  }

   402→  const attemptLogin = () => {
   403→    const stored = loadMerchant(phone || 'demo')
   404→    if (stored && simpleHash(pin) === stored.pinHash) {
   405→      playBeep('success')
   406→      haptic('success')
   407→      tataSpeak(`Bonjour Maman ${stored.firstName} ! Bienvenue sur Jùlaba.`)
   408→      setAuth(stored.id, stored.firstName, stored.phone)
   409→    } else {
   410→      doLogin(phone, firstName)
   411→    }
   412→  }
   413→
   414→  // --- Effects ---
   415→  useEffect(() => {
   416→    if (voiceEnabled) {
   417→      const t = setTimeout(() => {
   418→        tataSpeak('Bonjour ! Bienvenue sur Jùlaba. Entrez votre numéro ou dites votre nom.')
   419→      }, 500)
   420→      return () => clearTimeout(t)
   421→    }
   422→  }, [])
   423→
   424→  useEffect(() => () => { sttSessionRef.current?.abort() }, [])

   434→          'flex items-center gap-1.5 px-4 py-2 rounded-lg text-sm font-medium transition-all flex-1 justify-center',

   436→        )}
   437→      >
   438→        <Lock className='w-4 h-4' />
   439→        Code
   440→      </button>
   441→      <button
   442→        onClick={() => current === 'pattern' ? null : (mode === 'register' ? handleChooseMethod('pattern') : handleSwitchMethod('pattern'))}
   443→        className={cn(
   444→          'flex items-center gap-1.5 px-4 py-2 rounded-lg text-sm font-medium transition-all flex-1 justify-center',
   445→          current === 'pattern' ? 'bg-white shadow-sm text-[#C66A2C]' : 'text-muted-foreground'
   446→        )}
   447→      >
   448→        <Grid3X3 className='w-4 h-4' />
   449→        Schéma
   450→      </button>

   453→
   454→  return (

   456→      <div className='w-full max-w-sm'>
   457→        {/* Logo */}
   458→        <div className='text-center mb-8'>
   459→          <div className='w-20 h-20 rounded-2xl mx-auto mb-4 shadow-lg overflow-hidden'>
   460→            <img src='/icon-only.png' alt='Jùlaba' className='w-full h-full object-contain' />

   465→
   466→        {/* ===== STEP: Name / Phone ===== */}

   521→                <Phone className='w-10 h-10 mx-auto text-[#C66A2C] mb-2' />

   525→              <div className='flex items-center gap-2 bg-muted rounded-lg px-3 py-2.5'>
   526→                <Phone className='w-5 h-5 text-muted-foreground' />
   527→                <Input
   528→                  type='tel'
   529→                  placeholder='Ex: 07 01 02 03 04'
   530→                  value={phone}
   531→                  onChange={(e) => setPhone(e.target.value.replace(/[^\d\s]/g, ''))}
   532→                  className='border-0 bg-transparent text-lg p-0 h-auto focus-visible:ring-0'

   540→                  setError('')
   541→                  setMode('register')
   542→                  modeRef.current = 'register'
   543→                  setStep('choose-method')
   544→                  stepRef.current = 'choose-method'
   545→                  tataSpeak('Choisissez comment vous connecter.')
   546→                }}
   547→                disabled={phone.length < 8}
   548→              >
   549→                Continuer
   550→              </Button>
