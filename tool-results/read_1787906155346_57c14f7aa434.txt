    51→function SectionHeader({ sectionKey, icon, title, sectionStates, toggleEdit, handleSave, handleCancel, isDark }: {
    52→  sectionKey: string
    53→  icon: React.ReactNode
    54→  title: string
    55→  sectionStates: Record<string, SectionState>
    56→  toggleEdit: (section: string) => void
    57→  handleSave: (section: string) => void
    58→  handleCancel: (section: string) => void
    59→  isDark: boolean
    60→}) {
    61→  const state = sectionStates[sectionKey]
    62→  return (
    63→    <div className="flex items-center justify-between">
    64→      <CardTitle className={`text-sm font-semibold flex items-center gap-2 ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
    65→        {icon}
    66→        {title}
    67→      </CardTitle>
    68→      <div className="flex gap-2">
    69→        {state?.editing ? (
    70→          <>
    71→            <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={() => handleCancel(sectionKey)}>
    72→              <X className="h-3 w-3 mr-1" /> Annuler
    73→            </Button>
    74→            <Button size="sm" className="h-8 text-xs" onClick={() => handleSave(sectionKey)} disabled={state?.saving}>
    75→              {state?.saving ? (
    76→                <span className="h-3 w-3 border-2 border-white/30 border-t-white rounded-full animate-spin" />
    77→              ) : (
    78→                <><Save className="h-3 w-3 mr-1" /> Enregistrer</>
    79→              )}
    80→            </Button>
    81→          </>
    82→        ) : (
    83→          <Button variant="outline" size="sm" className="h-8 text-xs" onClick={() => toggleEdit(sectionKey)}>
    84→            <Pencil className="h-3 w-3 mr-1" /> Modifier
    85→          </Button>
    86→        )}
    87→      </div>
    88→    </div>
    89→  )
    90→}
