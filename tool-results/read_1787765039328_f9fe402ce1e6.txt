    43→const IDENT_COLOR = '#9F8170'
    44→
    45→export function IdentIdentificationScreen() {
    46→  const { goBack, navigate, soleilMode, merchantId, merchantName } = useAppStore()
    47→  const {
    48→    dossiers,
    49→    addDossier,
    50→    updateDossier,
    51→    currentDraftId,
    52→    setCurrentDraftId,
    53→  } = useIdentificateurStore()
    54→  const { toast } = useToast()
    55→
    56→  const [submitting, setSubmitting] = useState(false)
    57→  const [gpsLoading, setGpsLoading] = useState(false)
    58→
    59→  // Initialize dossier
    60→  const [dossier, setDossier] = useState<Dossier | null>(() => {
    61→    if (currentDraftId) {
    62→      const existing = dossiers.find((d) => d.id === currentDraftId)
    63→      if (existing) return { ...existing }
    64→    }
    65→    if (merchantId && merchantName) return createEmptyDossier(merchantId, merchantName)
    66→    return null
    67→  })
    68→  const [isNew, setIsNew] = useState(() => {
    69→    if (currentDraftId) {
    70→      return !dossiers.find((d) => d.id === currentDraftId)
    71→    }
    72→    return true
    73→  })
    74→  const photoInputRef = useRef<HTMLInputElement>(null)
    75→  const etalInputRef = useRef<HTMLInputElement>(null)
    76→  const docInputRef = useRef<HTMLInputElement>(null)
    77→
    78→  // Text size helper
    79→  const txt = soleilMode ? 'text-base' : 'text-sm'
    80→  const txtLabel = soleilMode ? 'text-base font-medium' : 'text-sm font-medium'
    81→
    82→  const autoSave = useCallback(() => {
    83→    if (!dossier) return
    84→    const now = Date.now()
    85→    if (isNew) {
    86→      addDossier({ ...dossier, updatedAt: now, status: 'brouillon' })
    87→      setIsNew(false)
    88→      setCurrentDraftId(dossier.id)
    89→    } else {
    90→      updateDossier(dossier.id, { ...dossier, updatedAt: now, status: 'brouillon' })
    91→    }
    92→  }, [dossier, isNew, addDossier, updateDossier, setCurrentDraftId])
    93→
    94→  // Auto-save every 30 seconds
    95→  useEffect(() => {
    96→    const interval = setInterval(() => {
    97→      if (dossier) {
    98→        autoSave()
    99→      }
   100→    }, 30000)
   101→    return () => clearInterval(interval)
   102→  }, [dossier, autoSave])
   103→
   104→  const saveToStore = useCallback(
   105→    (status: Dossier['status']) => {
   106→      if (!dossier) return
   107→      const now = Date.now()
   108→      const updates: Partial<Dossier> = {
   109→        ...dossier,
   110→        updatedAt: now,
   111→        status,
   112→      }
   113→      if (status === 'en_attente') {
   114→        updates.submittedAt = now
   115→      }
   116→      if (isNew) {
   117→        addDossier(updates as Dossier)
   118→        setIsNew(false)
   119→        setCurrentDraftId(dossier.id)
   120→      } else {
   121→        updateDossier(dossier.id, updates)
   122→      }
   123→    },
   124→    [dossier, isNew, addDossier, updateDossier, setCurrentDraftId]
   125→  )
   126→
   127→  const updateField = <K extends keyof Dossier>(key: K, value: Dossier[K]) => {
   128→    setDossier((prev) => (prev ? { ...prev, [key]: value } : prev))
   129→  }
   130→
   131→  // Photo handling
   132→  const handlePhotoCapture = (e: React.ChangeEvent<HTMLInputElement>) => {
   133→    const file = e.target.files?.[0]
   134→    if (!file) return
   135→    const reader = new FileReader()
   136→    reader.onloadend = () => {
   137→      updateField('photoBase64', reader.result as string)
   138→    }
   139→    reader.readAsDataURL(file)
   140→    e.target.value = ''
   141→  }
   142→
   143→  const handleEtalPhoto = (e: React.ChangeEvent<HTMLInputElement>) => {
   144→    const file = e.target.files?.[0]
   145→    if (!file) return
   146→    const reader = new FileReader()
   147→    reader.onloadend = () => {
   148→      updateField('photoEtal', reader.result as string)
   149→    }
   150→    reader.readAsDataURL(file)
   151→    e.target.value = ''
   152→  }
   153→
   154→  // Document handling
   155→  const handleDocumentAdd = (e: React.ChangeEvent<HTMLInputElement>) => {
   156→    const files = e.target.files
   157→    if (!files || !dossier) return
   158→    const newDocs = [...(dossier.documents || [])]
   159→    Array.from(files).forEach((file) => {
   160→      if (newDocs.length >= 10) return
   161→      const reader = new FileReader()
   162→      reader.onloadend = () => {
   163→        const docEntry = {
   164→          name: file.name,
   165→          base64: reader.result as string,
   166→          type: file.type,
   167→        }
   168→        setDossier((prev) => {
   169→          if (!prev) return prev
   170→          const docs = [...(prev.documents || []), docEntry].slice(0, 10)
   171→          return { ...prev, documents: docs }
   172→        })
   173→      }
   174→      reader.readAsDataURL(file)
   175→    })
   176→    e.target.value = ''
   177→  }
   178→
   179→  const removeDocument = (index: number) => {
   180→    if (!dossier) return
   181→    const docs = [...(dossier.documents || [])]
   182→    docs.splice(index, 1)
   183→    updateField('documents', docs)
   184→  }
   185→
   186→  // GPS capture
   187→  const captureGPS = () => {
   188→    if (!navigator.geolocation) {
   189→      toast({
   190→        title: 'Erè',
   191→        description: 'Géolokalizasyon pa disponib sou sa aparèy la',
   192→      })
   193→      return
   194→    }
   195→    setGpsLoading(true)
   196→    navigator.geolocation.getCurrentPosition(
   197→      (position) => {
   198→        updateField('gps', {
   199→          lat: position.coords.latitude,
   200→          lon: position.coords.longitude,
   201→          accuracy: position.coords.accuracy,
   202→          timestamp: Date.now(),
   203→        })
   204→        setGpsLoading(false)
   205→        toast({
   206→          title: 'Pozisyon kaptire',
   207→          description: `Presizyon: ${Math.round(position.coords.accuracy)}m`,
   208→        })
   209→      },
   210→      (error) => {
   211→        setGpsLoading(false)
   212→        let msg = 'Erè lè kaptire pozisyon'
   213→        if (error.code === 1) msg = 'Pèmisyon lokasyon refize'
   214→        if (error.code === 2) msg = 'Pozisyon pa disponib'
   215→        if (error.code === 3) msg = 'Tan ekspire pou lokasyon'
   216→        toast({
   217→          title: 'Erè',
   218→          description: msg,
   219→        })
   220→      },
   221→      { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
   222→    )
   223→  }
   224→

   226→  const toggleProduit = (produit: string, field: 'produitsPrincipaux' | 'principalesCultures' | 'domainesActivite') => {
   227→    if (!dossier) return
   228→    const current = (dossier[field] as string[]) || []
   229→    if (current.includes(produit)) {
   230→      updateField(field, current.filter((p) => p !== produit))
   231→    } else {
   232→      if (current.length >= 5) {
   233→        toast({
   234→          title: 'Limite atenn',
   235→          description: 'Ou ka chwa maksimòm 5 eleman',
   236→        })
   237→        return
   238→      }
   239→      updateField(field, [...current, produit])
   240→    }
   241→  }
   242→
   243→  // Validation
   244→  const validateRequired = (): string | null => {
   245→    if (!dossier) return 'Dossier pa disponib'
   246→    if (!dossier.photoBase64) return 'Fotò aktè obligatwa'
   247→    if (!dossier.actorType) return 'Tip aktè obligatwa'
   248→    if (!dossier.firstName.trim()) return 'Non obligatwa'
   249→    if (!dossier.lastName.trim()) return 'Prenon obligatwa'
   250→    if (!dossier.phone.trim()) return 'Telefòn obligatwa'
   251→    if (!dossier.activite) return 'Aktivite obligatwa'
   252→    if (!dossier.zone) return 'Zòn / Maché obligatwa'
   253→    if (!dossier.gps) return 'Géolokalizasyon obligatwa'
   254→    return null
   255→  }
   256→

   258→  const handleSaveDraft = () => {

   260→    toast({
   261→      title: 'Brouyon anrejistre',
   262→      description: 'Dossier sovgade kòm brouyon',

   264→    navigate('ident-brouillons')
   265→  }
   266→
   267→  // Submit for validation
   268→  const handleSubmit = async () => {
   269→    const error = validateRequired()

   271→      toast({
   272→        title: 'Champ obligatwa manke',
   273→        description: error,
   274→      })
   275→      return

   278→    // Small delay for UX feedback
   279→    await new Promise((r) => setTimeout(r, 500))

   281→    setSubmitting(false)
   282→    toast({
   283→      title: 'Dossier soumè',
   284→      description: 'Dossier a voye pou validasyon',
   285→    })
   286→    navigate('ident-suivi')
   287→  }
   288→
   289→  if (!dossier) {
   290→    return (
   291→      <div className="flex items-center justify-center min-h-screen">
   292→        <Loader2 className="size-8 animate-spin" style={{ color: IDENT_COLOR }} />
   293→      </div>
   294→    )
   295→  }
   296→
   297→  const actorTypeLabels: Record<ActorType, string> = {
   298→    marchand: 'Machan',
   299→    producteur: 'Prodiktè',
   300→    cooperative: 'Kopérativ',
   301→  }
   302→

   305→    feminin: 'Femenen',
   306→    autre: 'Lòt',
   307→  }
   308→
   309→  const typeCommerceLabels: Record<string, string> = {
   310→    marche: 'Maché',
   311→    boutique: 'Boutik',
   312→    ambulant: 'Ambilan',
   313→  }
   314→
   315→  const typeProductionLabels: Record<string, string> = {
   316→    culture: 'Kilti',

   319→  }
   320→
   321→  const modeExploitationLabels: Record<string, string> = {
   322→    familial: 'Familyal',
   323→    cooperatif: 'Kopératif',
   324→    individuel: 'Endividyèl',
   325→  }
   326→
   327→  return (
   328→    <div className="flex flex-col min-h-screen bg-background">
   329→      {/* Top Bar */}
   330→      <header
   331→        className="sticky top-0 z-30 flex items-center justify-between px-4 py-3 bg-white border-b shadow-sm"

   334→          onClick={goBack}
   335→          className="p-2 rounded-full hover:bg-gray-100 transition-colors"
   336→          aria-label="Retour"

   338→          <ArrowLeft className="size-5" style={{ color: IDENT_COLOR }} />

   340→        <h1

   342→          style={{ color: IDENT_COLOR }}
   343→        >
   344→          NOUVEAU DOSSIER
   345→        </h1>
   346→        <Button
   347→          variant="ghost"
   348→          size="sm"
   349→          onClick={handleSaveDraft}
   350→          className="gap-1"
   351→          style={{ color: IDENT_COLOR }}
   352→        >
   353→          💾 Brouyon
   354→        </Button>
   355→      </header>
   356→
   357→      {/* Scrollable Form */}
   358→      <main className="flex-1 overflow-y-auto pb-24">
   359→        <div className="max-w-lg mx-auto px-4 py-4 space-y-6">
   360→          {/* 1. PHOTO DE L'ACTEUR */}
   361→          <section>
   362→            <div
   363→              className="flex items-center gap-2 pb-2 border-b"
   364→              style={{ borderColor: IDENT_COLOR }}

   366→              <Camera className="size-4" style={{ color: IDENT_COLOR }} />

   368→                className={`${txtLabel} uppercase tracking-wider`}
   369→                style={{ color: IDENT_COLOR }}
   370→              >
   371→                FOTÒ AKTÈ <span className="text-red-500">*</span>
   372→              </h2>
   373→            </div>
   374→            <div className="mt-3">
   375→              {dossier.photoBase64 ? (
   376→                <div className="relative inline-block">
   377→                  <img
   378→                    src={dossier.photoBase64}
   379→                    alt="Foto aktè"
   380→                    className="w-32 h-32 rounded-lg object-cover border-2"
   381→                    style={{ borderColor: IDENT_COLOR }}
   382→                  />
   383→                  <button
   384→                    onClick={() => photoInputRef.current?.click()}
   385→                    className="absolute -bottom-2 -right-2 p-1.5 rounded-full bg-white shadow-md border hover:bg-gray-50 transition-colors"
   386→                    style={{ borderColor: IDENT_COLOR }}
   387→                    aria-label="Repran foto"
   388→                  >
   389→                    <RotateCcw className="size-3.5" style={{ color: IDENT_COLOR }} />
   390→                  </button>
   391→                </div>

   393→                <button
   394→                  onClick={() => photoInputRef.current?.click()}
   395→                  className="flex flex-col items-center justify-center w-32 h-32 rounded-lg border-2 border-dashed hover:bg-gray-50 transition-colors cursor-pointer"
   396→                  style={{ borderColor: IDENT_COLOR }}
   397→                >
   398→                  <Camera className="size-8 mb-1" style={{ color: IDENT_COLOR, opacity: 0.6 }} />
   399→                  <span className={`${txt} text-muted-foreground`}>Pran foto</span>
   400→                </button>
   401→              )}
   402→              <input
   403→                ref={photoInputRef}
   404→                type="file"
   405→                accept="image/*"
   406→                capture="environment"
   407→                onChange={handlePhotoCapture}
   408→                className="hidden"
   409→              />
   410→            </div>
   411→          </section>
   412→
   413→          {/* 2. TYPE D'ACTEUR */}
   414→          <section>
   415→            <div
   416→              className="flex items-center gap-2 pb-2 border-b"
   417→              style={{ borderColor: IDENT_COLOR }}
   418→            >
   419→              <span style={{ color: IDENT_COLOR }}>👤</span>
   420→              <h2
   421→                className={`${txtLabel} uppercase tracking-wider`}
   422→                style={{ color: IDENT_COLOR }}
   423→              >
   424→                TIP AKTÈ <span className="text-red-500">*</span>
   425→              </h2>
   426→            </div>
   427→            <div className="mt-3 grid grid-cols-3 gap-2">
   428→              {(
   429→                [
   430→                  { type: 'marchand' as ActorType, icon: '🏪', label: 'Machan' },
   431→                  { type: 'producteur' as ActorType, icon: '🌾', label: 'Prodiktè' },
   432→                  { type: 'cooperative' as ActorType, icon: '🤝', label: 'Kopérativ' },
   433→                ] as const
   434→              ).map((item) => (
   435→                <button
   436→                  key={item.type}
   437→                  type="button"
   438→                  onClick={() => updateField('actorType', item.type)}
   439→                  className={`flex flex-col items-center gap-1 p-3 rounded-lg border-2 transition-all ${
   440→                    dossier.actorType === item.type
   441→                      ? 'border-current shadow-sm'
   442→                      : 'border-gray-200 hover:border-gray-300'
   443→                  }`}
   444→                  style={
   445→                    dossier.actorType === item.type
   446→                      ? { borderColor: IDENT_COLOR, backgroundColor: `${IDENT_COLOR}10` }
   447→                      : undefined
   448→                  }
   449→                >
   450→                  <span className="text-2xl">{item.icon}</span>
   451→                  <span

   453→                    style={{ color: dossier.actorType === item.type ? IDENT_COLOR : undefined }}
   454→                  >
   455→                    {item.label}
   456→                  </span>
   457→                </button>
   458→              ))}
   459→            </div>
   460→          </section>

   463→          <section>
   464→            <div

   466→              style={{ borderColor: IDENT_COLOR }}
   467→            >
   468→              <span style={{ color: IDENT_COLOR }}>📋</span>
   469→              <h2
   470→                className={`${txtLabel} uppercase tracking-wider`}
   471→                style={{ color: IDENT_COLOR }}
   472→              >
   473→                ENFÒMASYON OBLIGATWA <span className="text-red-500">*</span>
   474→              </h2>
   475→            </div>
   476→            <div className="mt-3 space-y-4">
   477→              {/* Prénom */}
   478→              <div className="space-y-1.5">
   479→                <Label className={txtLabel}>
   480→                  Prenon <span className="text-red-500">*</span>

   485→                  value={dossier.firstName}
   486→                  onChange={(e) => updateField('firstName', e.target.value)}
   487→                />
   488→              </div>

   490→              {/* Nom */}
