   376→function mapDashboardFromApi(d: Record<string, unknown>): DashboardData {
   377→  const actorCountsByRegionRaw = d.actorCountsByRegion as Record<string, number> | undefined
   378→  const actorCountsByRegion = actorCountsByRegionRaw
   379→    ? Object.entries(actorCountsByRegionRaw).map(([name, count]) => ({ name, count }))
   380→    : []
   381→
   382→  const dailyEnrolmentTrendRaw = d.dailyEnrolmentTrend as { date?: string; day?: string; count: number }[] | undefined
   383→  const dailyEnrolmentTrend = (dailyEnrolmentTrendRaw || []).map((item) => ({
   384→    day: item.day || item.date || '',
   385→    count: item.count,
   386→  }))
   387→
   388→  const topIdentificateursRaw = d.topIdentificateurs as { name: string; zone?: string; count: number }[] | undefined
   389→  const topIdentificateurs = (topIdentificateursRaw || []).map((item) => ({
   390→    name: item.name,
   391→    zone: item.zone || '',
   392→    count: item.count,
   393→  }))
   394→
   395→  const systemHealthRaw = d.systemHealth as Record<string, unknown> | undefined
   396→  const systemHealth = systemHealthRaw
   397→    ? [{
   398→        name: 'Plateforme',
   399→        status: (systemHealthRaw.status as string) || 'operationnel',
   400→        latency: (systemHealthRaw.responseTime as number) || 0,
   401→      }]
   402→    : []
   403→
   404→  const unacknowledgedAlertsRaw = d.unacknowledgedAlerts
   405→  const unacknowledgedAlerts = Array.isArray(unacknowledgedAlertsRaw)
   406→    ? unacknowledgedAlertsRaw.length
   407→    : (unacknowledgedAlertsRaw as number) || 0
   408→
   409→  return {
   410→    totalActors: (d.totalActors as number) || 0,
   411→    activeActors: (d.activeActors as number) || 0,
   412→    suspendedActors: (d.suspendedActors as number) || 0,
   413→    totalEnrolments: (d.totalEnrolments as number) || 0,
   414→    pendingEnrolments: (d.pendingEnrolments as number) || 0,
   415→    totalZones: (d.totalZones as number) || 0,
   416→    totalMissions: (d.totalMissions as number) || 0,
   417→    activeMissions: (d.activeMissions as number) || 0,
   418→    actorCountsByRegion,
   419→    dailyEnrolmentTrend,
   420→    topIdentificateurs,
   421→    dataQuality: { photos: 0, gps: 0, phones: 0 },
   422→    systemHealth,
   423→    unacknowledgedAlerts,
   424→  }
   425→}

   574→        set({ loading: true, error: null })
   575→        try {
   576→          const res = await fetch('/api/backoffice')
   577→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   578→          const data = await res.json()
   579→          if (data.erreur) throw new Error(data.erreur)
   580→          const dashboard = mapDashboardFromApi(data)
   581→          set({
   582→            dashboard,
   583→            ticker: {
   584→              transactionsPerMin: 0,
   585→              enrolmentsPerHour: data.pendingEnrolments || 0,
   586→              uptime: data.systemHealth?.uptime || 0,
   587→              activeUsers: data.activeActors || 0,
   588→            },
   589→          })
   590→          return dashboard
   591→        } catch (err) {
   592→          set({ error: err instanceof Error ? err.message : 'Erreur de chargement du tableau de bord' })
   593→          return null
   594→        } finally {
   595→          set({ loading: false })
   596→        }
   597→      },
   598→
   599→      fetchAllData: async () => {
   600→        set({ loading: true, error: null })
   601→        const store = get()
   602→        const results = await Promise.allSettled([
   603→          store.fetchUsers(),
   604→          store.fetchActors(),
   605→          store.fetchEnrolments(),
   606→          store.fetchZones(),
   607→          store.fetchMissions(),
   608→          store.fetchAuditLog(),
   609→          store.fetchAlerts(),
   610→          store.fetchDashboard(),
   611→        ])
   612→        const errors = results
   613→          .filter((r): r is PromiseRejectedResult => r.status === 'rejected')
   614→          .map((r) => r.reason?.message || r.reason || 'Erreur inconnue')
   615→        if (errors.length > 0) {
   616→          set({ error: errors.join('; ') })
   617→        }
   618→        set({ loading: false })
   619→      },
   620→
   621→      // ============== MUTATION ACTIONS ==============
   622→
   623→      updateActorStatus: async (actorId, status) => {
   624→        // Optimistic update
   625→        set((s) => ({
   626→          actors: s.actors.map((a) => (a.id === actorId ? { ...a, status } : a)),
   627→        }))
   628→        try {
   629→          const res = await fetch('/api/backoffice/actors', {
   630→            method: 'PATCH',
   631→            headers: { 'Content-Type': 'application/json' },
   632→            body: JSON.stringify({ id: actorId, status }),
   633→          })
   634→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   635→        } catch (err) {
   636→          // Rollback on error
   637→          set((s) => ({
   638→            actors: s.actors.map((a) => (a.id === actorId ? { ...a, status: 'actif' as const } : a)),
   639→            error: err instanceof Error ? err.message : 'Erreur de mise à jour du statut',
   640→          }))
   641→        }
   642→      },
   643→
   644→      validateEnrolment: async (enrolmentId, userId) => {
   645→        const now = new Date().toISOString()
   646→        // Optimistic update
   647→        set((s) => ({
   648→          enrolments: s.enrolments.map((e) =>
   649→            e.id === enrolmentId
   650→              ? { ...e, status: 'valide' as const, validatedBy: userId, validatedAt: now }
   651→              : e
   652→          ),
   653→        }))
   654→        try {
   655→          const res = await fetch('/api/backoffice/enrolments', {
   656→            method: 'PATCH',
   657→            headers: { 'Content-Type': 'application/json' },
   658→            body: JSON.stringify({ id: enrolmentId, action: 'valider', validatedBy: userId }),
   659→          })
   660→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   661→        } catch (err) {
   662→          // Rollback
   663→          set((s) => ({
   664→            enrolments: s.enrolments.map((e) =>
   665→              e.id === enrolmentId
   666→                ? { ...e, status: 'en_attente' as const, validatedBy: undefined, validatedAt: undefined }
   667→                : e
   668→            ),
   669→            error: err instanceof Error ? err.message : 'Erreur de validation',
   670→          }))
   671→        }
   672→      },
   673→
   674→      rejectEnrolment: async (enrolmentId, reason, userId) => {
   675→        const now = new Date().toISOString()
   676→        // Optimistic update
   677→        set((s) => ({
   678→          enrolments: s.enrolments.map((e) =>
   679→            e.id === enrolmentId
   680→              ? { ...e, status: 'rejete' as const, validatedBy: userId, validatedAt: now, rejectReason: reason }
   681→              : e
   682→          ),
   683→        }))
   684→        try {
   685→          const res = await fetch('/api/backoffice/enrolments', {
   686→            method: 'PATCH',
   687→            headers: { 'Content-Type': 'application/json' },
   688→            body: JSON.stringify({ id: enrolmentId, action: 'rejeter', validatedBy: userId, rejectReason: reason }),
   689→          })
   690→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   691→        } catch (err) {
   692→          // Rollback
   693→          set((s) => ({
   694→            enrolments: s.enrolments.map((e) =>
   695→              e.id === enrolmentId
   696→                ? { ...e, status: 'en_attente' as const, validatedBy: undefined, validatedAt: undefined, rejectReason: undefined }
   697→                : e
   698→            ),
   699→            error: err instanceof Error ? err.message : 'Erreur de rejet',
   700→          }))
   701→        }
   702→      },
   703→
   704→      acknowledgeAlert: async (alertId) => {
   705→        // Optimistic update
   706→        set((s) => ({
   707→          alerts: s.alerts.map((a) => (a.id === alertId ? { ...a, acknowledged: true } : a)),
   708→        }))
   709→        try {
   710→          const res = await fetch('/api/backoffice/alerts', {
   711→            method: 'PATCH',
   712→            headers: { 'Content-Type': 'application/json' },
   713→            body: JSON.stringify({ id: alertId, acknowledged: true }),
   714→          })
   715→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   716→        } catch (err) {
   717→          // Rollback
   718→          set((s) => ({
   719→            alerts: s.alerts.map((a) => (a.id === alertId ? { ...a, acknowledged: false } : a)),
   720→            error: err instanceof Error ? err.message : 'Erreur d\'acquittement',
   721→          }))
   722→        }
   723→      },
   724→
   725→      addAuditEntry: (entry) =>
   726→        set((s) => ({
   727→          auditLog: [
   728→            { ...entry, id: `a-${Date.now()}`, timestamp: new Date().toISOString() },
   729→            ...s.auditLog,
   730→          ],
   731→        })),
   732→
   733→      updateUser: async (userId, updates) => {
   734→        const previous = get().users.find((u) => u.id === userId)
   735→        // Optimistic update
   736→        set((s) => ({
   737→          users: s.users.map((u) => (u.id === userId ? { ...u, ...updates } : u)),
   738→        }))
   739→        try {
   740→          const body: Record<string, unknown> = { id: userId }
   741→          if (updates.role) body.role = updates.role
   742→          if (updates.isActive !== undefined) body.isActive = updates.isActive
   743→          if (updates.zone !== undefined) body.zone = updates.zone
   744→          if (updates.name) body.name = updates.name
   745→          const res = await fetch('/api/backoffice/users', {
   746→            method: 'PATCH',
   747→            headers: { 'Content-Type': 'application/json' },
   748→            body: JSON.stringify(body),
   749→          })
   750→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   751→        } catch (err) {
   752→          // Rollback
   753→          if (previous) {
   754→            set((s) => ({
   755→              users: s.users.map((u) => (u.id === userId ? previous : u)),
   756→            }))
   757→          }
   758→          set({ error: err instanceof Error ? err.message : 'Erreur de mise à jour de l\'utilisateur' })
   759→        }
   760→      },
   761→
   762→      createUser: async (user) => {
   763→        try {
   764→          const res = await fetch('/api/backoffice/users', {
   765→            method: 'POST',
   766→            headers: { 'Content-Type': 'application/json' },
   767→            body: JSON.stringify({
   768→              email: user.email,
   769→              name: user.name,
   770→              role: user.role,
   771→              zone: user.zone || null,
   772→              passwordHash: 'admin123',
   773→            }),
   774→          })
   775→          if (!res.ok) {
   776→            const data = await res.json().catch(() => ({}))
   777→            throw new Error((data as Record<string, string>).erreur || `Erreur ${res.status}`)
   778→          }
   779→          const created = await res.json()
   780→          const newUser = mapUserFromApi(created)
   781→          set((s) => ({ users: [...s.users, newUser] }))
   782→        } catch (err) {
   783→          set({ error: err instanceof Error ? err.message : 'Erreur de création de l\'utilisateur' })
   784→        }
   785→      },
   786→
   787→      // Theme
   788→      boTheme: 'light' as const,
   789→      toggleBoTheme: () => set((s) => ({ boTheme: s.boTheme === 'light' ? 'dark' : 'light' })),
   790→
   791→      // Search
   792→      searchQuery: '',
   793→      setSearchQuery: (q) => set({ searchQuery: q }),
   794→    }),
   795→    {
   796→      name: 'julaba-backoffice-store',
   797→      partialize: (state) => ({
   798→        boUserRole: state.boUserRole,
   799→        sidebarCollapsed: state.sidebarCollapsed,
   800→        boCurrentScreen: state.boCurrentScreen,
   801→        boTheme: state.boTheme,
   802→      }),
   803→      onRehydrateStorage: () => {
   804→        return (state, error) => {
   805→          if (!error && state) {
   806→            // Auto-fetch data from API after rehydration
   807→            state.fetchAllData()
   808→          }
   809→        }
   810→      },
   811→    }
   812→  )
   813→)
