    29→const simpleHash = (str: string) => {
    30→  let hash = 0
    31→  for (let i = 0; i < str.length; i++) {
    32→    const char = str.charCodeAt(i)
    33→    hash = ((hash << 5) - hash) + char
    34→    hash |= 0
    35→  }
    36→  return hash.toString()
    37→}

    46→    return raw ? JSON.parse(raw) : null
    47→  } catch {
    48→    return null
    49→  }
    50→}
