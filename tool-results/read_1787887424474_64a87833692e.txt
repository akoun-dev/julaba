    43→    43→function SectionHeader({ sectionKey, icon, title, sectionStates, toggleEdit, handleSave, handleCancel, isDark }: {
    44→    44→  sectionKey: string
    45→    45→  icon: React.ReactNode
    46→    46→  title: string
    47→    47→  sectionStates: Record<string, SectionState>
    48→    48→  toggleEdit: (section: string) => void
    49→    49→  handleSave: (section: string) => void
    50→    50→  handleCancel: (section: string) => void
    51→    51→  isDark: boolean
    52→    52→}) {
    53→    53→  const state = sectionStates[sectionKey]
    54→    54→  return (
    55→    55→    <div className="flex items-center justify-between">
    56→    56→      <CardTitle className={`text-sm font-semibold flex items-center gap-2 ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
    57→    57→        {icon}
    58→    58→        {title}
    59→    59→      </CardTitle>
    60→    60→      <div className="flex gap-2">
    61→    61→        {state?.editing ? (
    62→    62→          <>
    63→    63→            <Button variant="ghost" size="sm" className="h-8 text-xs" onClick={() => handleCancel(sectionKey)}>
    64→    64→              <X className="h-3 w-3 mr-1" /> Annuler
    65→    65→            </Button>
    66→    66→            <Button size="sm" className="h-8 text-xs" onClick={() => handleSave(sectionKey)} disabled={state?.saving}>
    67→    67→              {state?.saving ? (
    68→    68→                <span className="h-3 w-3 border-2 border-white/30 border-t-white rounded-full animate-spin" />
    69→    69→              ) : (
    70→    70→                <><Save className="h-3 w-3 mr-1" /> Enregistrer</>
    71→    71→              )}
    72→    72→            </Button>
    73→    73→          </>
    74→    74→        ) : (
    75→    75→          <Button variant="outline" size="sm" className="h-8 text-xs" onClick={() => toggleEdit(sectionKey)}>
    76→    76→            <Pencil className="h-3 w-3 mr-1" /> Modifier
    77→    77→          </Button>
    78→    78→        )}
    79→    79→      </div>
    80→    80→    </div>
    81→    81→  )
    82→    82→}
