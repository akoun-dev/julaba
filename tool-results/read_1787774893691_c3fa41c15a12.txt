    62→    62→// Simple hash utility (same as auth-screen)
    63→    63→const simpleHash = (str: string) => {
    64→    64→  let hash = 0
    65→    65→  for (let i = 0; i < str.length; i++) {
    66→    66→    const char = str.charCodeAt(i)
    67→    67→    hash = ((hash << 5) - hash) + char
    68→    68→    hash |= 0
    69→    69→  }
    70→    70→  return hash.toString()
    71→    71→}
    72→    72→const patternToHash = (pattern: number[]) => simpleHash(pattern.join('-'))
