    27→    27→const simpleHash = (str: string) => {
    28→    28→  let hash = 0
    29→    29→  for (let i = 0; i < str.length; i++) {
    30→    30→    const char = str.charCodeAt(i)
    31→    31→    hash = ((hash << 5) - hash) + char
    32→    32→    hash |= 0
    33→    33→  }
    34→    34→  return hash.toString()
    35→    35→}
    36→    36→
    37→    37→const patternToHash = (pattern: number[]) => simpleHash(pattern.join('-'))
    38→    38→const loadMerchant = (phone: string): MerchantData | null => {
    39→    39→  const raw = localStorage.getItem(`julaba-merchant-${phone}`)

    41→    41→}

   268→   268→      setStep('login-pin')
   269→   269→      stepRef.current = 'login-pin'
   270→   270→      tataSpeak('Entrez votre code à 4 chiffres.')
   271→   271→    } else {
   272→   272→      setStep('pattern-login')

   299→   299→        firstName: firstName || 'Awa',
   300→   300→        phone,

   303→   303→        authMethod: 'pattern',
   304→   304→      }
   305→   305→      saveMerchant(merchantData)
   306→   306→      localStorage.setItem('julaba-last-name', merchantData.firstName)

   444→   444→          'flex items-center gap-1.5 px-4 py-2 rounded-lg text-sm font-medium transition-all flex-1 justify-center',
