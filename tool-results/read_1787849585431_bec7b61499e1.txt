    86→// ============== HELPERS ==============
    87→
    88→function isToday(dateStr: string): boolean {
    89→  const d = new Date(dateStr)
    90→  const now = new Date()
    91→  return (
    92→    d.getFullYear() === now.getFullYear() &&
    93→    d.getMonth() === now.getMonth() &&
    94→    d.getDate() === now.getDate()
    95→  )
    96→}
    97→
    98→function isThisWeek(dateStr: string): boolean {
    99→  const d = new Date(dateStr)
   100→  const now = new Date()
   101→  const startOfWeek = new Date(now)
   102→  startOfWeek.setDate(now.getDate() - now.getDay() + 1)
   103→  startOfWeek.setHours(0, 0, 0, 0)
   104→  return d >= startOfWeek
   105→}
   106→
   107→function isThisMonth(dateStr: string): boolean {
   108→  const d = new Date(dateStr)
   109→  const now = new Date()
   110→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   111→}
   112→
   113→function matchesDateRange(dateStr: string, range: DateRange): boolean {
   114→  switch (range) {
   115→    case 'aujourdhui':
   116→      return isToday(dateStr)
   117→    case 'semaine':
   118→      return isThisWeek(dateStr)
   119→    case 'mois':
   120→      return isThisMonth(dateStr)
   121→    case 'tous':
   122→    default:
   123→      return true
   124→  }
   125→}
   126→
   127→function formatDate(dateStr: string): string {
   128→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   129→    day: '2-digit',
   130→    month: '2-digit',
   131→    year: 'numeric',
   132→    hour: '2-digit',
   133→    minute: '2-digit',
   134→  })
   135→}

   206→      if (!matchesDateRange(e.submittedAt, dateRange)) return false

   231→    setCurrentPage(1)
   232→  }, [])

   298→          className="text-2xl font-bold tracking-tight text-slate-900"
   299→        >
   300→          VALIDATION DES ENRÔLEMENTS

   314→              key={tab.key}
   315→              variant={isActive ? 'default' : 'outline'}
   316→              size="sm"

   319→                isActive
   320→                  ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   321→                  : 'text-[#333333] hover:bg-[#333333]/5'
   322→              }
   323→            >
   324→              {tab.label}
   325→              <Badge
   326→                variant="secondary"
   327→                className={
   328→                  isActive

   339→
   340→      {/* ===== ADDITIONAL FILTERS ===== */}
   341→      <div className="flex flex-wrap items-center gap-3">
   342→        <div className="flex items-center gap-2">
   343→          <span className="text-sm font-medium text-[#333333]/70">
   344→            Zone :
   345→          </span>
   346→          <Select value={zoneFilter} onValueChange={handleZoneChange}>
   347→            <SelectTrigger className="w-[180px]">
   348→              <SelectValue placeholder="Toutes les zones" />
   349→            </SelectTrigger>
   350→            <SelectContent>
   351→              <SelectItem value="all">Toutes les zones</SelectItem>
   352→              {zones.map((z) => (
   353→                <SelectItem key={z} value={z}>

   358→          </Select>

   361→        <div className="flex items-center gap-2">
   362→          <span className="text-sm font-medium text-[#333333]/70">
   363→            Période :
   364→          </span>
   365→          <Select value={dateRange} onValueChange={handleDateRangeChange}>
   366→            <SelectTrigger className="w-[180px]">
   367→              <SelectValue placeholder="Tous" />
   368→            </SelectTrigger>
   369→            <SelectContent>
   370→              {DATE_RANGE_OPTIONS.map((opt) => (
   371→                <SelectItem key={opt.value} value={opt.value}>
   372→                  {opt.label}
   373→                </SelectItem>
   374→              ))}
   375→            </SelectContent>
   376→          </Select>
   377→        </div>
   378→      </div>
   379→
   380→      {/* ===== STATS BAR ===== */}
   381→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
   382→        <Card className="gap-0 py-4">
   383→          <CardContent className="flex items-center gap-3 px-4 py-0">
   384→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-50">
   385→              <CheckCircle2 className="h-5 w-5 text-emerald-600" />
   386→            </div>
   387→            <div>
   388→              <p className="text-xs font-medium text-[#333333]/60">
   389→                Validés aujourd&apos;hui
   390→              </p>
   391→              <p className="text-xl font-bold text-[#333333]">
   392→                {todayStats.validated}
   393→              </p>
   394→            </div>
   395→          </CardContent>
   396→        </Card>
   397→
   398→        <Card className="gap-0 py-4">
   399→          <CardContent className="flex items-center gap-3 px-4 py-0">
   400→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-red-50">
   401→              <XCircle className="h-5 w-5 text-red-600" />
   402→            </div>
   403→            <div>
   404→              <p className="text-xs font-medium text-[#333333]/60">
   405→                Rejetés aujourd&apos;hui
   406→              </p>
   407→              <p className="text-xl font-bold text-[#333333]">
   408→                {todayStats.rejected}
   409→              </p>
   410→            </div>
   411→          </CardContent>
   412→        </Card>
   413→
   414→        <Card className="gap-0 py-4">
   415→          <CardContent className="flex items-center gap-3 px-4 py-0">
   416→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-amber-50">
   417→              <Clock className="h-5 w-5 text-amber-600" />
   418→            </div>
   419→            <div>
   420→              <p className="text-xs font-medium text-[#333333]/60">
   421→                <span className="inline-flex items-center gap-1"><Timer className="h-3.5 w-3.5" /> Temps moyen</span>
   422→              </p>
   423→              <p className="text-xl font-bold text-[#333333]">
   424→                {todayStats.avgTime}
   425→              </p>
   426→            </div>
   427→          </CardContent>
   428→        </Card>
   429→
   430→        <Card className="gap-0 py-4">
   431→          <CardContent className="flex items-center gap-3 px-4 py-0">
   432→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-50">
   433→              <TrendingUp className="h-5 w-5 text-emerald-600" />
   434→            </div>
   435→            <div>
   436→              <p className="text-xs font-medium text-[#333333]/60">
   437→                Taux validation
   438→              </p>
   439→              <p className="text-xl font-bold text-[#333333]">
   440→                {todayStats.rate}%
   441→              </p>
   442→            </div>
   443→          </CardContent>
   444→        </Card>
   445→      </div>
   446→
   447→      <Separator />
   448→
   449→      {/* ===== ENROLMENT CARDS LIST ===== */}

   452→          /* Empty State */
   453→          <div className="flex flex-col items-center justify-center gap-4 rounded-xl border border-dashed py-16">
   454→            <div className="flex h-16 w-16 items-center justify-center rounded-full bg-[#333333]/5">
   455→              <Inbox className="h-8 w-8 text-[#333333]/40" />
   456→            </div>
   457→            <div className="text-center">
   458→              <p className="text-lg font-semibold text-[#333333]">
   459→                Aucun enrôlement trouvé
   460→              </p>
   461→              <p className="mt-1 text-sm text-[#333333]/50">
   462→                Aucun enrôlement ne correspond aux filtres sélectionnés.
   463→              </p>
   464→            </div>
   465→          </div>
   466→        ) : (
   467→          paginatedEnrolments.map((enrolment) => (
   468→            <EnrolmentCard

   470→              enrolment={enrolment}
   471→              onValidate={handleValidate}
   472→              onReject={handleOpenReject}
   473→              onRequestInfo={handleRequestInfo}
   474→            />
   475→          ))
   476→        )}
   477→      </div>
   478→
   479→      {/* ===== PAGINATION ===== */}
   480→      {filteredEnrolments.length > ITEMS_PER_PAGE && (
   481→        <div className="flex items-center justify-between pt-2">
   482→          <p className="text-sm text-[#333333]/60">
   483→            Affichage {(currentPage - 1) * ITEMS_PER_PAGE + 1}–
   484→            {Math.min(currentPage * ITEMS_PER_PAGE, filteredEnrolments.length)}
   485→            {' '}sur {filteredEnrolments.length} enrôlements
   486→          </p>
   487→          <div className="flex items-center gap-2">
   488→            <Button
   489→              variant="outline"
   490→              size="sm"

   492→              onClick={() => setCurrentPage((p) => p - 1)}
   493→              className="text-[#333333]"
   494→            >
   495→              <ChevronLeft className="mr-1 h-4 w-4" />
   496→              Précédent

   498→            <div className="flex items-center gap-1">

   500→                .filter((page) => {
   501→                  // Show first, last, and pages around current

   503→                  if (Math.abs(page - currentPage) <= 1) return true

   522→                      …
   523→                    </span>
   524→                  ) : (
   525→                    <Button
   526→                      key={item}
   527→                      variant={currentPage === item ? 'default' : 'outline'}
   528→                      size="sm"
   529→                      className={
   530→                        currentPage === item
   531→                          ? 'h-8 w-8 bg-[#333333] text-white hover:bg-[#333333]/90 p-0'
   532→                          : 'h-8 w-8 p-0 text-[#333333]'
   533→                      }
   534→                      onClick={() => setCurrentPage(item)}
   535→                    >
   536→                      {item}
   537→                    </Button>
   538→                  )
   539→                )}
   540→            </div>
   541→            <Button
   542→              variant="outline"
   543→              size="sm"
   544→              disabled={currentPage >= totalPages}
   545→              onClick={() => setCurrentPage((p) => p + 1)}
   546→              className="text-[#333333]"
   547→            >
   548→              Suivant
   549→              <ChevronRight className="ml-1 h-4 w-4" />
   550→            </Button>
   551→          </div>
   552→        </div>
   553→      )}
   554→
   555→      {/* ===== REJECT DIALOG ===== */}
   556→      <Dialog open={rejectDialogOpen} onOpenChange={setRejectDialogOpen}>
   557→        <DialogContent className="sm:max-w-md">
   558→          <DialogHeader>
   559→            <DialogTitle className="text-[#333333]">
   560→              Rejeter l&apos;enrôlement
   561→            </DialogTitle>
   562→            <DialogDescription>
   563→              {rejectTarget && (
   564→                <>
   565→                  <span className="font-medium text-[#333333]">
   566→                    {rejectTarget.actorName}
   567→                  </span>{' '}
   568→                  — {rejectTarget.dossierId}
   569→                </>
   570→              )}
   571→            </DialogDescription>
   572→          </DialogHeader>
   573→
   574→          <div className="flex flex-col gap-4">
   575→            {/* Predefined reasons */}
   576→            <div className="flex flex-col gap-2">
   577→              <Label className="text-sm font-medium text-[#333333]">
   578→                Raison prédéfinie
   579→              </Label>
   580→              <div className="flex flex-wrap gap-2">
   581→                {PREDEFINED_REASONS.map((reason) => (
   582→                  <Button
   583→                    key={reason}
   584→                    variant={selectedPreset === reason ? 'default' : 'outline'}
   585→                    size="sm"
   586→                    onClick={() => handlePresetReason(reason)}
   587→                    className={
   588→                      selectedPreset === reason
   589→                        ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   590→                        : 'text-[#333333]'
   591→                    }
   592→                  >

   594→                  </Button>
   595→                ))}
   596→              </div>
