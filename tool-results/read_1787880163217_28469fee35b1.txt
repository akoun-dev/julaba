   103→   103→      // Zone filter

   140→   140→  const handleZoneFilterChange = useCallback((val: string) => {
   141→   141→    setZoneFilter(val)

   209→   209→      'Téléphone',

   286→   286→        </CardContent>
   287→   287→      </Card>
   288→   288→
   289→   289→      {/* ===== FILTERS ===== */}

   292→   292→          <div className="flex flex-wrap items-center gap-3">
   293→   293→            <span className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   294→   294→              Filtres :
   295→   295→            </span>
   296→   296→
   297→   297→            <Select value={typeFilter} onValueChange={handleTypeFilterChange}>

   301→   301→              <SelectContent>
   302→   302→                <SelectItem value="tous">Tous les types</SelectItem>

   306→   306→                <SelectItem value="producteur">
   307→   307→                   Producteur(rice)s

   309→   309→                <SelectItem value="cooperatif">
   310→   310→                   Coopératives

   315→   315→            <Select

   346→   346→              <Button
   347→   347→                variant="ghost"
   348→   348→                size="sm"
   349→   349→                onClick={resetFilters}
   350→   350→                className="text-muted-foreground hover:text-foreground"

   353→   353→                Réinitialiser

   356→   356→          </div>
   357→   357→        </CardContent>
   358→   358→      </Card>
   359→   359→
   360→   360→      {/* ===== COUNTER BAR ===== */}
   361→   361→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

   364→   364→            <div className={`flex items-center justify-center size-10 rounded-lg ${isDark ? 'bg-slate-700' : 'bg-gray-100'}`}>

   382→   382→                {counts.marchands}
   383→   383→              </p>

   385→   385→            </div>
