    88→// ============== HELPERS ==============
    89→
    90→function isToday(dateStr: string): boolean {
    91→  const d = new Date(dateStr)
    92→  const now = new Date()
    93→  return (
    94→    d.getFullYear() === now.getFullYear() &&
    95→    d.getMonth() === now.getMonth() &&
    96→    d.getDate() === now.getDate()
    97→  )
    98→}
    99→
   100→function isThisWeek(dateStr: string): boolean {
   101→  const d = new Date(dateStr)
   102→  const now = new Date()
   103→  const startOfWeek = new Date(now)
   104→  startOfWeek.setDate(now.getDate() - now.getDay() + 1)
   105→  startOfWeek.setHours(0, 0, 0, 0)
   106→  return d >= startOfWeek
   107→}
   108→
   109→function isThisMonth(dateStr: string): boolean {
   110→  const d = new Date(dateStr)
   111→  const now = new Date()
   112→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   113→}
   114→
   115→function matchesDateRange(dateStr: string, range: DateRange): boolean {
   116→  switch (range) {
   117→    case 'aujourdhui':
   118→      return isToday(dateStr)
   119→    case 'semaine':
   120→      return isThisWeek(dateStr)
   121→    case 'mois':
   122→      return isThisMonth(dateStr)
   123→    case 'tous':
   124→    default:
   125→      return true
   126→  }
   127→}
   128→
   129→function formatDate(dateStr: string): string {
   130→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   131→    day: '2-digit',
   132→    month: '2-digit',
   133→    year: 'numeric',
   134→    hour: '2-digit',
   135→    minute: '2-digit',
   136→  })
   137→}

   542→                      onClick={() => setCurrentPage(item)}
   543→                    >
   544→                      {item}
   545→                    </Button>
   546→                  )
   547→                )}
   548→            </div>
   549→            <Button
   550→              variant="outline"
   551→              size="sm"
   552→              disabled={currentPage >= totalPages}
   553→              onClick={() => setCurrentPage((p) => p + 1)}
   554→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   555→            >
   556→              Suivant
   557→              <ChevronRight className="ml-1 h-4 w-4" />
   558→            </Button>
   559→          </div>
   560→        </div>
   561→      )}
   562→
   563→      {/* ===== REJECT DIALOG ===== */}
   564→      <Dialog open={rejectDialogOpen} onOpenChange={setRejectDialogOpen}>
   565→        <DialogContent className="sm:max-w-md">
   566→          <DialogHeader>
   567→            <DialogTitle className={isDark ? 'text-slate-100' : 'text-[#333333]'}>
   568→              Rejeter l&apos;enrôlement
   569→            </DialogTitle>
   570→            <DialogDescription>
   571→              {rejectTarget && (
   572→                <>
   573→                  <span className={`font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   574→                    {rejectTarget.actorName}
   575→                  </span>{' '}
   576→                  — {rejectTarget.dossierId}
   577→                </>
   578→              )}
   579→            </DialogDescription>
   580→          </DialogHeader>
   581→
   582→          <div className="flex flex-col gap-4">
   583→            {/* Predefined reasons */}
   584→            <div className="flex flex-col gap-2">
   585→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   586→                Raison prédéfinie
   587→              </Label>
   588→              <div className="flex flex-wrap gap-2">
   589→                {PREDEFINED_REASONS.map((reason) => (
   590→                  <Button
   591→                    key={reason}
   592→                    variant={selectedPreset === reason ? 'default' : 'outline'}
   593→                    size="sm"
   594→                    onClick={() => handlePresetReason(reason)}
   595→                    className={
   596→                      selectedPreset === reason
   597→                        ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   598→                        : isDark ? 'text-slate-100' : 'text-[#333333]'
   599→                    }
   600→                  >
   601→                    {reason}
   602→                  </Button>
   603→                ))}
   604→              </div>
   605→            </div>
   606→
   607→            {/* Custom reason */}
   608→            <div className="flex flex-col gap-2">
   609→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   610→                Raison du rejet <span className="text-red-500">*</span>
   611→              </Label>
   612→              <Textarea
   613→                value={rejectReason}
   614→                onChange={(e) => setRejectReason(e.target.value)}
   615→                placeholder="Décrivez la raison du rejet..."
   616→                className="min-h-[100px] resize-none"
   617→              />
   618→            </div>
   619→          </div>
   620→
   621→          <DialogFooter>
   622→            <Button
   623→              variant="outline"
   624→              onClick={() => setRejectDialogOpen(false)}
   625→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   626→            >
   627→              Annuler
   628→            </Button>
   629→            <Button
   630→              variant="destructive"
   631→              onClick={handleConfirmReject}
   632→              disabled={!rejectReason.trim()}
   633→            >
   634→              <XCircle className="mr-1.5 h-4 w-4" />
   635→              Confirmer le rejet
   636→            </Button>
   637→          </DialogFooter>
   638→        </DialogContent>
   639→      </Dialog>
   640→        </>
   641→      )}
   642→    </div>
   643→  )
