    80→function formatDateTime(iso: string) {
    81→  const d = new Date(iso)
    82→  return d.toLocaleDateString('fr-FR', {
    83→    day: '2-digit',
    84→    month: '2-digit',
    85→    year: 'numeric',
    86→    hour: '2-digit',
    87→    minute: '2-digit',
    88→  })
    89→}
    90→
    91→function getTimeAgo(iso: string) {
    92→  const now = Date.now()
    93→  const then = new Date(iso).getTime()
    94→  const diff = now - then
    95→  const minutes = Math.floor(diff / 60000)
    96→  if (minutes < 1) return 'à l\'instant'
    97→  if (minutes < 60) return `il y a ${minutes}min`
    98→  const hours = Math.floor(minutes / 60)
    99→  if (hours < 24) return `il y a ${hours}h`
   100→  const days = Math.floor(hours / 24)
   101→  if (days < 7) return `il y a ${days}j`
   102→  return formatDateTime(iso)
   103→}

   105→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   106→  const now = new Date()
   107→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   108→  let start: Date
   109→
   110→  switch (dateRange) {
   111→    case 'aujourdhui':
   112→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   113→      break
   114→    case 'semaine': {
   115→      const dayOfWeek = now.getDay() || 7
   116→      start = new Date(now)
   117→      start.setDate(now.getDate() - dayOfWeek + 1)
   118→      start.setHours(0, 0, 0, 0)
   119→      break
   120→    }
   121→    case 'mois':
   122→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   123→      break
   124→    case 'trimestre': {
   125→      const quarter = Math.floor(now.getMonth() / 3)
   126→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   127→      break
   128→    }
   129→    default:
   130→      start = new Date(2000, 0, 1)
   131→  }
   132→
   133→  return { start, end }
   134→}

   136→function truncateString(str: string, maxLen: number) {
   137→  if (!str) return '—'
   138→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   139→}

   141→function formatJsonDetails(details: string | undefined) {
   142→  if (!details) return null
   143→  try {
   144→    const parsed = JSON.parse(details)
   145→    return JSON.stringify(parsed, null, 2)
   146→  } catch {
   147→    return details
   148→  }
   149→}

   153→function downloadCsv(entries: AuditEntry[], filename: string) {
   154→  const headers = [
   155→    'Date/Heure',
   156→    'Utilisateur',
   157→    'Email',
   158→    'Action',
   159→    'Module',
   160→    'Adresse IP',
   161→    'User-Agent',
   162→    'Détails',
   163→  ]
   164→  const rows = entries.map((e) => [
   165→    formatDateTime(e.timestamp),
   166→    e.userName,
   167→    e.userEmail,
   168→    e.action,
   169→    e.module,
   170→    e.ipAddress ?? '',
   171→    e.userAgent ?? '',
   172→    e.details ?? '',
   173→  ])
   174→
   175→  const csvContent = [
   176→    headers.join(';'),
   177→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(';')),
   178→  ].join('\n')
   179→
   180→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   181→  const url = URL.createObjectURL(blob)
   182→  const link = document.createElement('a')
   183→  link.href = url
   184→  link.download = filename
   185→  document.body.appendChild(link)
   186→  link.click()
   187→  document.body.removeChild(link)
   188→  URL.revokeObjectURL(url)
   189→}

   595→                  paginatedEntries.map((entry, idx) => {
   596→                    const isExpanded = expandedRowId === entry.id
   597→                    return (
   598→                      <Collapsible
   599→                        key={entry.id}
   600→                        open={isExpanded}
   601→                        onOpenChange={() => toggleRowExpand(entry.id)}
   602→                      >
   603→                        <TableRow
   604→                          className={`cursor-pointer select-none transition-colors hover:bg-gray-50 ${
   605→                            idx % 2 === 0 ? 'bg-white' : 'bg-gray-50/40'
   606→                          } ${isExpanded ? 'bg-gray-50' : ''}`}
   607→                          onClick={() => toggleRowExpand(entry.id)}
   608→                        >
   609→                          {/* Expand indicator */}
   610→                          <TableCell className="w-8 p-2">
   611→                            <div className="flex h-5 w-5 items-center justify-center rounded text-gray-400">
   612→                              {isExpanded ? (
   613→                                <ChevronUp className="h-3.5 w-3.5" />
   614→                              ) : (
   615→                                <ChevronDown className="h-3.5 w-3.5" />
   616→                              )}
   617→                            </div>
   618→                          </TableCell>
   619→
   620→                          {/* Date/Time */}
   621→                          <TableCell className="text-sm">
   622→                            <div>
   623→                              <div className="text-gray-700" style={{ color: BO_COLOR }}>
   624→                                {formatDateTime(entry.timestamp)}
   625→                              </div>
   626→                              <div className="text-[11px] text-gray-400">
   627→                                {getTimeAgo(entry.timestamp)}
   628→                              </div>
   629→                            </div>
   630→                          </TableCell>
   631→
   632→                          {/* User */}
   633→                          <TableCell>
   634→                            <div>
   635→                              <div className="text-sm font-medium" style={{ color: BO_COLOR }}>
   636→                                {entry.userName}
   637→                              </div>
   638→                              <div className="text-[11px] text-gray-400">
   639→                                {entry.userEmail}
   640→                              </div>
   641→                            </div>
   642→                          </TableCell>
   643→
   644→                          {/* Action */}
   645→                          <TableCell>
   646→                            <Badge
   647→                              variant="outline"
   648→                              className={`whitespace-nowrap text-[11px] font-medium ${
   649→                                ACTION_BADGE_COLORS[entry.action] ??
   650→                                'bg-gray-100 text-gray-700 border-gray-200'
   651→                              }`}
   652→                            >
   653→                              {entry.action}
   654→                            </Badge>
   655→                          </TableCell>
   656→
   657→                          {/* Module */}
   658→                          <TableCell className="text-sm text-gray-600">
   659→                            {entry.module}
   660→                          </TableCell>
   661→
   662→                          {/* IP */}
   663→                          <TableCell className="font-mono text-xs text-gray-500">
   664→                            {entry.ipAddress ?? '—'}
   665→                          </TableCell>
   666→
   667→                          {/* User-Agent (truncated) */}
   668→                          <TableCell className="text-xs text-gray-400">
   669→                            {truncateString(entry.userAgent ?? '', 28)}
   670→                          </TableCell>
   671→
   672→                          {/* Details button */}
   673→                          <TableCell>
   674→                            {entry.details ? (
   675→                              <Button
   676→                                variant="ghost"
   677→                                size="sm"
   678→                                className="h-7 text-[11px] text-gray-500 hover:text-gray-700"
   679→                                onClick={(e) => {
   680→                                  e.stopPropagation()
   681→                                  toggleRowExpand(entry.id)
   682→                                }}
   683→                              >
   684→                                Voir
   685→                              </Button>
   686→                            ) : (
   687→                              <span className="text-xs text-gray-300">—</span>
   688→                            )}
   689→                          </TableCell>
   690→                        </TableRow>
   691→
   692→                        {/* Expanded details */}
   693→                        {isExpanded && (
   694→                          <TableRow>
   695→                            <TableCell
   696→                              colSpan={8}
   697→                              className="p-0"
   698→                            >
   699→                              <ExpandedDetails entry={entry} />
   700→                            </TableCell>
   701→                          </TableRow>
   702→                        )}
   703→                      </Collapsible>
   704→                    )
   705→                  })
   706→                )}
   707→              </TableBody>
   708→            </Table>
   709→          </div>
   710→        </CardContent>
   711→      </Card>
   712→
   713→      {/* ===== PAGINATION ===== */}
   714→      <div className="flex items-center justify-between">
   715→        <p className="text-xs text-gray-400">
   716→          Affichage {Math.min((currentPage - 1) * ITEMS_PER_PAGE + 1, filteredEntries.length)}
   717→          {'–'}
   718→          {Math.min(currentPage * ITEMS_PER_PAGE, filteredEntries.length)}
   719→          {' '}sur {filteredEntries.length}
   720→        </p>
   721→
   722→        <div className="flex items-center gap-1">
   723→          <Button
   724→            variant="outline"
   725→            size="sm"
   726→            className="h-8 w-8 p-0"
   727→            disabled={currentPage <= 1}
   728→            onClick={() => setCurrentPage((p) => p - 1)}
   729→          >
   730→            <ChevronLeft className="h-4 w-4" />
   731→          </Button>
   732→
   733→          {/* Page numbers */}
   734→          {Array.from({ length: totalPages }, (_, i) => i + 1)
   735→            .filter((page) => {
   736→              // Show first, last, current, and neighbors
   737→              if (page === 1 || page === totalPages) return true
   738→              if (Math.abs(page - currentPage) <= 1) return true
   739→              return false
   740→            })
   741→            .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   742→              if (idx > 0) {
   743→                const prev = arr[idx - 1]
   744→                if (page - prev > 1) {
   745→                  acc.push('ellipsis')
   746→                }
   747→              }
   748→              acc.push(page)
   749→              return acc
   750→            }, [])
   751→            .map((item, idx) =>
   752→              item === 'ellipsis' ? (
   753→                <span
   754→                  key={`ellipsis-${idx}`}
   755→                  className="flex h-8 w-8 items-center justify-center text-xs text-gray-400"
   756→                >
   757→                  …
   758→                </span>
   759→              ) : (
   760→                <Button
   761→                  key={item}
   762→                  variant={currentPage === item ? 'default' : 'outline'}
   763→                  size="sm"
   764→                  className={`h-8 w-8 p-0 text-xs ${
   765→                    currentPage === item
   766→                      ? 'text-white'
   767→                      : ''
   768→                  }`}
   769→                  style={
   770→                    currentPage === item
   771→                      ? { backgroundColor: BO_COLOR, color: '#fff' }
   772→                      : undefined
   773→                  }
   774→                  onClick={() => setCurrentPage(item)}
   775→                >
   776→                  {item}
   777→                </Button>
   778→              )
   779→            )}
   780→
   781→          <Button
   782→            variant="outline"
   783→            size="sm"
   784→            className="h-8 w-8 p-0"
   785→            disabled={currentPage >= totalPages}
   786→            onClick={() => setCurrentPage((p) => p + 1)}
   787→          >
   788→            <ChevronRight className="h-4 w-4" />
   789→          </Button>
   790→        </div>
   791→      </div>
   792→    </div>
   793→  )
   794→}
