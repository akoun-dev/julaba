   403→function mapDashboardFromApi(d: Record<string, unknown>): DashboardData {
   404→  const actorCountsByRegionRaw = d.actorCountsByRegion as Record<string, number> | undefined
   405→  const actorCountsByRegion = actorCountsByRegionRaw
   406→    ? Object.entries(actorCountsByRegionRaw).map(([name, count]) => ({ name, count }))
   407→    : []
   408→
   409→  const dailyEnrolmentTrendRaw = d.dailyEnrolmentTrend as { date?: string; day?: string; count: number }[] | undefined
   410→  const dailyEnrolmentTrend = (dailyEnrolmentTrendRaw || []).map((item) => ({
   411→    day: item.day || item.date || '',
   412→    count: item.count,
   413→  }))
   414→
   415→  const topIdentificateursRaw = d.topIdentificateurs as { name: string; zone?: string; count: number }[] | undefined
   416→  const topIdentificateurs = (topIdentificateursRaw || []).map((item) => ({
   417→    name: item.name,
   418→    zone: item.zone || '',
   419→    count: item.count,
   420→  }))
   421→
   422→  const systemHealthRaw = d.systemHealth as Record<string, unknown> | undefined
   423→  const systemHealth = systemHealthRaw
   424→    ? [{
   425→        name: 'Plateforme',
   426→        status: (systemHealthRaw.status as string) || 'operationnel',
   427→        latency: (systemHealthRaw.responseTime as number) || 0,
   428→      }]
   429→    : []
   430→
   431→  const unacknowledgedAlertsRaw = d.unacknowledgedAlerts
   432→  const unacknowledgedAlerts = Array.isArray(unacknowledgedAlertsRaw)
   433→    ? unacknowledgedAlertsRaw.length
   434→    : (unacknowledgedAlertsRaw as number) || 0
   435→
   436→  return {
   437→    totalActors: (d.totalActors as number) || 0,
   438→    activeActors: (d.activeActors as number) || 0,
   439→    suspendedActors: (d.suspendedActors as number) || 0,
   440→    totalEnrolments: (d.totalEnrolments as number) || 0,
   441→    pendingEnrolments: (d.pendingEnrolments as number) || 0,
   442→    totalZones: (d.totalZones as number) || 0,
   443→    totalMissions: (d.totalMissions as number) || 0,
   444→    activeMissions: (d.activeMissions as number) || 0,
   445→    actorCountsByRegion,
   446→    dailyEnrolmentTrend,
   447→    topIdentificateurs,
   448→    dataQuality: { photos: 0, gps: 0, phones: 0 },
   449→    systemHealth,
   450→    unacknowledgedAlerts,
   451→  }
   452→}

   575→          const data = await res.json()
   576→          const auditLog: AuditEntry[] = (data.logs || []).map(mapAuditEntryFromApi)
   577→          set({ auditLog })
   578→        } catch (err) {
   579→          set({ error: err instanceof Error ? err.message : 'Erreur de chargement du journal d\'audit' })
   580→        } finally {
   581→          set({ loading: false })
   582→        }
   583→      },
   584→
   585→      fetchAlerts: async () => {
   586→        set({ loading: true, error: null })
   587→        try {
   588→          const res = await fetch('/api/backoffice/alerts')
   589→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   590→          const data = await res.json()
   591→          const alerts: BoAlert[] = (Array.isArray(data) ? data : []).map(mapAlertFromApi)
   592→          set({ alerts })
   593→        } catch (err) {
   594→          set({ error: err instanceof Error ? err.message : 'Erreur de chargement des alertes' })
   595→        } finally {
   596→          set({ loading: false })
   597→        }
   598→      },
   599→
   600→      fetchDashboard: async () => {
   601→        set({ loading: true, error: null })
   602→        try {
   603→          const res = await fetch('/api/backoffice')
   604→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   605→          const data = await res.json()
   606→          if (data.erreur) throw new Error(data.erreur)
   607→          const dashboard = mapDashboardFromApi(data)
   608→          set({
   609→            dashboard,
   610→            ticker: {
   611→              transactionsPerMin: 0,
   612→              enrolmentsPerHour: data.pendingEnrolments || 0,
   613→              uptime: data.systemHealth?.uptime || 0,
   614→              activeUsers: data.activeActors || 0,
   615→            },
   616→          })
   617→          return dashboard
   618→        } catch (err) {
   619→          set({ error: err instanceof Error ? err.message : 'Erreur de chargement du tableau de bord' })
   620→          return null
   621→        } finally {
   622→          set({ loading: false })
   623→        }
   624→      },
   625→
   626→      fetchAllData: async () => {
   627→        set({ loading: true, error: null })
   628→        const store = get()
   629→        const results = await Promise.allSettled([
   630→          store.fetchUsers(),
   631→          store.fetchActors(),
   632→          store.fetchEnrolments(),
   633→          store.fetchZones(),
   634→          store.fetchMissions(),
   635→          store.fetchAuditLog(),
   636→          store.fetchAlerts(),
   637→          store.fetchDashboard(),
   638→        ])
   639→        const errors = results
   640→          .filter((r): r is PromiseRejectedResult => r.status === 'rejected')
   641→          .map((r) => r.reason?.message || r.reason || 'Erreur inconnue')
   642→        if (errors.length > 0) {
   643→          set({ error: errors.join('; ') })
   644→        }
   645→        set({ loading: false })
   646→      },
   647→
   648→      // ============== MUTATION ACTIONS ==============
   649→
   650→      updateActorStatus: async (actorId, status) => {
   651→        // Optimistic update
   652→        set((s) => ({
   653→          actors: s.actors.map((a) => (a.id === actorId ? { ...a, status } : a)),
   654→        }))
   655→        try {
   656→          const res = await fetch('/api/backoffice/actors', {
   657→            method: 'PATCH',
   658→            headers: { 'Content-Type': 'application/json' },
   659→            body: JSON.stringify({ id: actorId, status }),
   660→          })
   661→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   662→        } catch (err) {
   663→          // Rollback on error
   664→          set((s) => ({
   665→            actors: s.actors.map((a) => (a.id === actorId ? { ...a, status: 'actif' as const } : a)),
   666→            error: err instanceof Error ? err.message : 'Erreur de mise à jour du statut',
   667→          }))
   668→        }
   669→      },
   670→
   671→      validateEnrolment: async (enrolmentId, userId) => {
   672→        const now = new Date().toISOString()
   673→        // Optimistic update
   674→        set((s) => ({
   675→          enrolments: s.enrolments.map((e) =>
   676→            e.id === enrolmentId
   677→              ? { ...e, status: 'valide' as const, validatedBy: userId, validatedAt: now }
   678→              : e
   679→          ),
   680→        }))
   681→        try {
   682→          const res = await fetch('/api/backoffice/enrolments', {
   683→            method: 'PATCH',
   684→            headers: { 'Content-Type': 'application/json' },
   685→            body: JSON.stringify({ id: enrolmentId, action: 'valider', validatedBy: userId }),
   686→          })
   687→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   688→        } catch (err) {
   689→          // Rollback
   690→          set((s) => ({
   691→            enrolments: s.enrolments.map((e) =>
   692→              e.id === enrolmentId
   693→                ? { ...e, status: 'en_attente' as const, validatedBy: undefined, validatedAt: undefined }
   694→                : e
   695→            ),
   696→            error: err instanceof Error ? err.message : 'Erreur de validation',
   697→          }))
   698→        }
   699→      },
   700→
   701→      rejectEnrolment: async (enrolmentId, reason, userId) => {
   702→        const now = new Date().toISOString()
   703→        // Optimistic update
   704→        set((s) => ({
   705→          enrolments: s.enrolments.map((e) =>
   706→            e.id === enrolmentId
   707→              ? { ...e, status: 'rejete' as const, validatedBy: userId, validatedAt: now, rejectReason: reason }
   708→              : e
   709→          ),
   710→        }))
   711→        try {
   712→          const res = await fetch('/api/backoffice/enrolments', {
   713→            method: 'PATCH',
   714→            headers: { 'Content-Type': 'application/json' },
   715→            body: JSON.stringify({ id: enrolmentId, action: 'rejeter', validatedBy: userId, rejectReason: reason }),
   716→          })
   717→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   718→        } catch (err) {
   719→          // Rollback
   720→          set((s) => ({
   721→            enrolments: s.enrolments.map((e) =>
   722→              e.id === enrolmentId
   723→                ? { ...e, status: 'en_attente' as const, validatedBy: undefined, validatedAt: undefined, rejectReason: undefined }
   724→                : e
   725→            ),
   726→            error: err instanceof Error ? err.message : 'Erreur de rejet',
   727→          }))
   728→        }
   729→      },
   730→
   731→      acknowledgeAlert: async (alertId) => {
   732→        // Optimistic update
   733→        set((s) => ({
   734→          alerts: s.alerts.map((a) => (a.id === alertId ? { ...a, acknowledged: true } : a)),
   735→        }))
   736→        try {
   737→          const res = await fetch('/api/backoffice/alerts', {
   738→            method: 'PATCH',
   739→            headers: { 'Content-Type': 'application/json' },
   740→            body: JSON.stringify({ id: alertId, acknowledged: true }),
   741→          })
   742→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   743→        } catch (err) {
   744→          // Rollback
   745→          set((s) => ({
   746→            alerts: s.alerts.map((a) => (a.id === alertId ? { ...a, acknowledged: false } : a)),
   747→            error: err instanceof Error ? err.message : 'Erreur d\'acquittement',
   748→          }))
   749→        }
   750→      },
   751→
   752→      addAuditEntry: (entry) =>
   753→        set((s) => ({
   754→          auditLog: [
   755→            { ...entry, id: `a-${Date.now()}`, timestamp: new Date().toISOString() },
   756→            ...s.auditLog,
   757→          ],
   758→        })),
   759→
   760→      updateUser: async (userId, updates) => {
   761→        const previous = get().users.find((u) => u.id === userId)
   762→        // Optimistic update
   763→        set((s) => ({
   764→          users: s.users.map((u) => (u.id === userId ? { ...u, ...updates } : u)),
   765→        }))
   766→        try {
   767→          const body: Record<string, unknown> = { id: userId }
   768→          if (updates.role) body.role = updates.role
   769→          if (updates.isActive !== undefined) body.isActive = updates.isActive
   770→          if (updates.zone !== undefined) body.zone = updates.zone
   771→          if (updates.name) body.name = updates.name
   772→          const res = await fetch('/api/backoffice/users', {
   773→            method: 'PATCH',
   774→            headers: { 'Content-Type': 'application/json' },
   775→            body: JSON.stringify(body),
   776→          })
   777→          if (!res.ok) throw new Error(`Erreur ${res.status}`)
   778→        } catch (err) {
   779→          // Rollback
   780→          if (previous) {
   781→            set((s) => ({
   782→              users: s.users.map((u) => (u.id === userId ? previous : u)),
   783→            }))
   784→          }
   785→          set({ error: err instanceof Error ? err.message : 'Erreur de mise à jour de l\'utilisateur' })
   786→        }
   787→      },
   788→
   789→      createUser: async (user) => {
   790→        try {
   791→          const res = await fetch('/api/backoffice/users', {
   792→            method: 'POST',
   793→            headers: { 'Content-Type': 'application/json' },
   794→            body: JSON.stringify({
   795→              email: user.email,
   796→              name: user.name,
   797→              role: user.role,
   798→              zone: user.zone || null,
   799→              passwordHash: 'admin123',
   800→            }),
   801→          })
   802→          if (!res.ok) {
   803→            const data = await res.json().catch(() => ({}))
   804→            throw new Error((data as Record<string, string>).erreur || `Erreur ${res.status}`)
   805→          }
   806→          const created = await res.json()
   807→          const newUser = mapUserFromApi(created)
   808→          set((s) => ({ users: [...s.users, newUser] }))
   809→        } catch (err) {
   810→          set({ error: err instanceof Error ? err.message : 'Erreur de création de l\'utilisateur' })
   811→        }
   812→      },
   813→
   814→      // Theme
   815→      boTheme: 'light' as const,
   816→      toggleBoTheme: () => set((s) => ({ boTheme: s.boTheme === 'light' ? 'dark' : 'light' })),
   817→
   818→      // Search
   819→      searchQuery: '',
   820→      setSearchQuery: (q) => set({ searchQuery: q }),
   821→    }),
   822→    {
   823→      name: 'julaba-backoffice-store',
   824→      partialize: (state) => ({
   825→        boUserRole: state.boUserRole,
   826→        sidebarCollapsed: state.sidebarCollapsed,
   827→        boCurrentScreen: state.boCurrentScreen,
   828→        boTheme: state.boTheme,
   829→      }),
   830→      onRehydrateStorage: () => {
   831→        return (state, error) => {
   832→          if (!error && state) {
   833→            // Auto-fetch data from API after rehydration
   834→            state.fetchAllData()
   835→          }
   836→        }
   837→      },
   838→    }
   839→  )
