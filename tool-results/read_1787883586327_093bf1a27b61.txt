    75→function formatDateTime(iso: string) {
    76→  const d = new Date(iso)
    77→  return d.toLocaleDateString('fr-FR', {
    78→    day: '2-digit',
    79→    month: '2-digit',
    80→    year: 'numeric',
    81→    hour: '2-digit',
    82→    minute: '2-digit',
    83→  })
    84→}
    85→
    86→function getTimeAgo(iso: string) {
    87→  const now = Date.now()
    88→  const then = new Date(iso).getTime()
    89→  const diff = now - then
    90→  const minutes = Math.floor(diff / 60000)
    91→  if (minutes < 1) return 'à l\'instant'
    92→  if (minutes < 60) return `il y a ${minutes}min`
    93→  const hours = Math.floor(minutes / 60)
    94→  if (hours < 24) return `il y a ${hours}h`
    95→  const days = Math.floor(hours / 24)
    96→  if (days < 7) return `il y a ${days}j`
    97→  return formatDateTime(iso)
    98→}

   100→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   101→  const now = new Date()
   102→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   103→  let start: Date
   104→
   105→  switch (dateRange) {
   106→    case 'aujourdhui':
   107→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   108→      break
   109→    case 'semaine': {
   110→      const dayOfWeek = now.getDay() || 7
   111→      start = new Date(now)
   112→      start.setDate(now.getDate() - dayOfWeek + 1)
   113→      start.setHours(0, 0, 0, 0)
   114→      break
   115→    }
   116→    case 'mois':
   117→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   118→      break
   119→    case 'trimestre': {
   120→      const quarter = Math.floor(now.getMonth() / 3)
   121→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   122→      break
   123→    }
   124→    default:
   125→      start = new Date(2000, 0, 1)
   126→  }
   127→
   128→  return { start, end }
   129→}

   131→function truncateString(str: string, maxLen: number) {
   132→  if (!str) return '—'
   133→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   134→}

   136→function formatJsonDetails(details: string | undefined) {
   137→  if (!details) return null
   138→  try {
   139→    const parsed = JSON.parse(details)
   140→    return JSON.stringify(parsed, null, 2)
   141→  } catch {
   142→    return details
   143→  }
   144→}

   148→function downloadCsv(entries: AuditEntry[], filename: string) {
   149→  const headers = [
   150→    'Date/Heure',
   151→    'Utilisateur',
   152→    'Email',
   153→    'Action',
   154→    'Module',
   155→    'Adresse IP',
   156→    'User-Agent',
   157→    'Détails',
   158→  ]
   159→  const rows = entries.map((e) => [
   160→    formatDateTime(e.timestamp),
   161→    e.userName,
   162→    e.userEmail,
   163→    e.action,
   164→    e.module,
   165→    e.ipAddress ?? '',
   166→    e.userAgent ?? '',
   167→    e.details ?? '',
   168→  ])
   169→
   170→  const csvContent = [
   171→    headers.join(';'),
   172→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}”`).join(';')),
   173→  ].join('\n')
   174→
   175→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   176→  const url = URL.createObjectURL(blob)
   177→  const link = document.createElement('a')
   178→  link.href = url
   179→  link.download = filename
   180→  document.body.appendChild(link)
   181→  link.click()
   182→  document.body.removeChild(link)
   183→  URL.revokeObjectURL(url)
   184→}

   540→                  >
   541→                    Date/Heure
   542→                  </TableHead>
   543→                  <TableHead
   544→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   545→                    style={{ minWidth: 140 }}
   546→                  >
   547→                    Utilisateur
   548→                  </TableHead>
   549→                  <TableHead
   550→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   551→                    style={{ minWidth: 120 }}
   552→                  >
   553→                    Action
   554→                  </TableHead>
   555→                  <TableHead
   556→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   557→                    style={{ minWidth: 100 }}
   558→                  >
   559→                    Module
   560→                  </TableHead>
   561→                  <TableHead
   562→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   563→                    style={{ minWidth: 120 }}
   564→                  >
   565→                    IP Address
   566→                  </TableHead>
   567→                  <TableHead
   568→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   569→                    style={{ minWidth: 160 }}
   570→                  >
   571→                    User-Agent
   572→                  </TableHead>
   573→                  <TableHead
   574→                    className={`font-semibold text-xs ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   575→                    style={{ minWidth: 80 }}
   576→                  >
   577→                    Détails
   578→                  </TableHead>
   579→                </TableRow>
   580→              </TableHeader>
   581→              <TableBody>
   582→                {paginatedEntries.length === 0 ? (
   583→                  <TableRow>
   584→                    <TableCell
   585→                      colSpan={8}
   586→                      className={`py-12 text-center text-sm ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   587→                    >
   588→                      Aucune entrée d\'audit trouvée pour les filtres sélectionnés.
   589→                    </TableCell>
   590→                  </TableRow>
   591→                ) : (
   592→                  paginatedEntries.map((entry, idx) => {
   593→                    const isExpanded = expandedRowId === entry.id
   594→                    return (
   595→                      <Fragment key={entry.id}>
   596→                        <TableRow
   597→                          className={`cursor-pointer select-none transition-colors ${isDark ? 'hover:bg-slate-700' : 'hover:bg-gray-50'} ${
   598→                            idx % 2 === 0
   599→                              ? isDark ? 'bg-slate-800' : 'bg-white'
   600→                              : isDark ? 'bg-slate-800/70' : 'bg-gray-50/40'
   601→                          } ${isExpanded ? (isDark ? 'bg-slate-700/50' : 'bg-gray-50') : ''}`}
   602→                          onClick={() => toggleRowExpand(entry.id)}
   603→                        >
   604→                          {/* Expand indicator */}
   605→                          <TableCell className="w-8 p-2">
   606→                            <div className={`flex h-5 w-5 items-center justify-center rounded ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   607→                              {isExpanded ? (
   608→                                <ChevronUp className="h-3.5 w-3.5" />
   609→                              ) : (
   610→                                <ChevronDown className="h-3.5 w-3.5" />
   611→                              )}
   612→                            </div>
   613→                          </TableCell>
   614→
   615→                          {/* Date/Time */}
   616→                          <TableCell className="text-sm">
   617→                            <div>
   618→                              <div className={isDark ? 'text-slate-100' : 'text-slate-900'}>
   619→                                {formatDateTime(entry.timestamp)}
   620→                              </div>
   621→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   622→                                {getTimeAgo(entry.timestamp)}
   623→                              </div>
   624→                            </div>
   625→                          </TableCell>
   626→
   627→                          {/* User */}
   628→                          <TableCell>
   629→                            <div>
   630→                              <div className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   631→                                {entry.userName}
   632→                              </div>
   633→                              <div className={`text-[11px] ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   634→                                {entry.userEmail}
   635→                              </div>
   636→                            </div>
   637→                          </TableCell>
   638→
   639→                          {/* Action */}
   640→                          <TableCell>
   641→                            <Badge
   642→                              variant="outline"
   643→                              className={`whitespace-nowrap text-[11px] font-medium ${
   644→                                ACTION_BADGE_COLORS[entry.action] ??
   645→                                (isDark ? 'bg-slate-700 text-slate-300 border-slate-600' : 'bg-gray-100 text-gray-700 border-gray-200')
   646→                              }`}
   647→                            >
   648→                              {entry.action}
   649→                            </Badge>
   650→                          </TableCell>
   651→
   652→                          {/* Module */}
   653→                          <TableCell className={`text-sm ${isDark ? 'text-slate-300' : 'text-gray-600'}`}>

   657→                          {/* IP */}
   658→                          <TableCell className={`font-mono text-xs ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>
   659→                            {entry.ipAddress ?? '—'}
   660→                          </TableCell>
   661→
   662→                          {/* User-Agent (truncated) */}
   663→                          <TableCell className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   664→                            {truncateString(entry.userAgent ?? '', 28)}
   665→                          </TableCell>
   666→
   667→                          {/* Details button */}
   668→                          <TableCell>
   669→                            {entry.details ? (
   670→                              <Button
   671→                                variant="ghost"
   672→                                size="sm"
   673→                                className={`h-7 text-[11px] ${isDark ? 'text-slate-400 hover:text-slate-300' : 'text-gray-500 hover:text-gray-700'}`}
   674→                                onClick={(e) => {
   675→                                  e.stopPropagation()
   676→                                  toggleRowExpand(entry.id)
   677→                                }}
   678→                              >
   679→                                Voir
   680→                              </Button>
   681→                            ) : (
   682→                              <span className={`text-xs ${isDark ? 'text-slate-600' : 'text-gray-300'}`}>—</span>
   683→                            )}
   684→                          </TableCell>
   685→                        </TableRow>
   686→
   687→                        {/* Expanded details */}
   688→                        {isExpanded && (
   689→                          <TableRow>
   690→                            <TableCell
   691→                              colSpan={8}
   692→                              className="p-0"
   693→                            >
   694→                              <ExpandedDetails entry={entry} />
   695→                            </TableCell>
   696→                          </TableRow>
   697→                        )}
   698→                      </Fragment>
   699→                    )
   700→                  })
   701→                )}
   702→              </TableBody>
   703→            </Table>
   704→          </div>
   705→        </CardContent>
   706→      </Card>
   707→
   708→      {/* ===== PAGINATION ===== */}
   709→      <div className="flex items-center justify-between">
   710→        <p className={`text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}>
   711→          Affichage {Math.min((currentPage - 1) * ITEMS_PER_PAGE + 1, filteredEntries.length)}
   712→          {'–'}
   713→          {Math.min(currentPage * ITEMS_PER_PAGE, filteredEntries.length)}
   714→          {' '}sur {filteredEntries.length}
   715→        </p>
   716→
   717→        <div className="flex items-center gap-1">
   718→          <Button
   719→            variant="outline"
   720→            size="sm"
   721→            className="h-8 w-8 p-0"
   722→            disabled={currentPage <= 1}
   723→            onClick={() => setCurrentPage((p) => p - 1)}
   724→          >
   725→            <ChevronLeft className="h-4 w-4" />
   726→          </Button>
   727→
   728→          {/* Page numbers */}
   729→          {Array.from({ length: totalPages }, (_, i) => i + 1)
   730→            .filter((page) => {
   731→              // Show first, last, current, and neighbors
   732→              if (page === 1 || page === totalPages) return true
   733→              if (Math.abs(page - currentPage) <= 1) return true
   734→              return false
   735→            })
   736→            .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   737→              if (idx > 0) {
   738→                const prev = arr[idx - 1]
   739→                if (page - prev > 1) {
   740→                  acc.push('ellipsis')
   741→                }
   742→              }
   743→              acc.push(page)
   744→              return acc
   745→            }, [])
   746→            .map((item, idx) =>
   747→              item === 'ellipsis' ? (
   748→                <span
   749→                  key={`ellipsis-${idx}`}
   750→                  className={`flex h-8 w-8 items-center justify-center text-xs ${isDark ? 'text-slate-500' : 'text-gray-400'}`}
   751→                >
   752→                  …
   753→                </span>
   754→              ) : (
   755→                <Button
   756→                  key={item}
   757→                  variant={currentPage === item ? 'default' : 'outline'}
   758→                  size="sm"
   759→                  className={`h-8 w-8 p-0 text-xs ${
   760→                    currentPage === item
   761→                      ? 'text-white'
   762→                      : ''
   763→                  }`}
   764→                  style={
   765→                    currentPage === item
   766→                      ? { backgroundColor: isDark ? '#3B82F6' : '#0F172A', color: '#fff' }
   767→                      : undefined
   768→                  }
   769→                  onClick={() => setCurrentPage(item)}
   770→                >
   771→                  {item}
   772→                </Button>
   773→              )
   774→            )}
   775→
   776→          <Button
   777→            variant="outline"
   778→            size="sm"
   779→            className="h-8 w-8 p-0"
   780→            disabled={currentPage >= totalPages}
   781→            onClick={() => setCurrentPage((p) => p + 1)}
   782→          >
   783→            <ChevronRight className="h-4 w-4" />
   784→          </Button>
   785→        </div>
   786→      </div>
   787→    </div>
   788→  )
   789→}
