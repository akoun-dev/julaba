    89→    89→
    90→    90→function isToday(dateStr: string): boolean {
    91→    91→  const d = new Date(dateStr)
    92→    92→  const now = new Date()
    93→    93→  return (
    94→    94→    d.getFullYear() === now.getFullYear() &&
    95→    95→    d.getMonth() === now.getMonth() &&
    96→    96→    d.getDate() === now.getDate()
    97→    97→  )
    98→    98→}
    99→    99→
   100→   100→function isThisWeek(dateStr: string): boolean {
   101→   101→  const d = new Date(dateStr)
   102→   102→  const now = new Date()
   103→   103→  const startOfWeek = new Date(now)
   104→   104→  startOfWeek.setDate(now.getDate() - now.getDay() + 1)
   105→   105→  startOfWeek.setHours(0, 0, 0, 0)
   106→   106→  return d >= startOfWeek
   107→   107→}
   108→   108→
   109→   109→function isThisMonth(dateStr: string): boolean {
   110→   110→  const d = new Date(dateStr)
   111→   111→  const now = new Date()
   112→   112→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   113→   113→}
   114→   114→
   115→   115→function matchesDateRange(dateStr: string, range: DateRange): boolean {
   116→   116→  switch (range) {
   117→   117→    case 'aujourdhui':
   118→   118→      return isToday(dateStr)
   119→   119→    case 'semaine':
   120→   120→      return isThisWeek(dateStr)
   121→   121→    case 'mois':
   122→   122→      return isThisMonth(dateStr)
   123→   123→    case 'tous':
   124→   124→    default:
   125→   125→      return true
   126→   126→  }
   127→   127→}
   128→   128→
   129→   129→function formatDate(dateStr: string): string {
   130→   130→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   131→   131→    day: '2-digit',
   132→   132→    month: '2-digit',
   133→   133→    year: 'numeric',
   134→   134→    hour: '2-digit',
   135→   135→    minute: '2-digit',
   136→   136→  })
   137→   137→}

   485→   485→      </div>
   486→   486→
   487→   487→      {/* ===== PAGINATION ===== */}
   488→   488→      {filteredEnrolments.length > ITEMS_PER_PAGE && (

   510→   510→                  if (page === 1 || page === totalPages) return true
   511→   511→                  if (Math.abs(page - currentPage) <= 1) return true
   512→   512→                  return false

   514→   514→                .reduce<(number | 'ellipsis')[]>((acc, page, idx, arr) => {
   515→   515→                  if (idx > 0) {
   516→   516→                    const prev = arr[idx - 1]
   517→   517→                    if (page - prev > 1) {
   518→   518→                      acc.push('ellipsis')
   519→   519→                    }
   520→   520→                  }
   521→   521→                  acc.push(page)
   522→   522→                  return acc
   523→   523→                }, [])
   524→   524→                .map((item, idx) =>
   525→   525→                  item === 'ellipsis' ? (
   526→   526→                    <span
   527→   527→                      key={`ellipsis-${idx}`}
   528→   528→                      className={isDark ? 'px-1 text-slate-500' : 'px-1 text-[#333333]/40'}
   529→   529→                    >
   530→   530→                      …
   531→   531→                    </span>
   532→   532→                  ) : (
   533→   533→                    <Button
   534→   534→                      key={item}
   535→   535→                      variant={currentPage === item ? 'default' : 'outline'}

   537→   537→                      className={
   538→   538→                        currentPage === item
   539→   539→                          ? 'h-8 w-8 bg-[#333333] text-white hover:bg-[#333333]/90 p-0'
   540→   540→                          : `h-8 w-8 p-0 ${isDark ? 'text-slate-100' : 'text-[#333333]'}`
   541→   541→                      }
   542→   542→                      onClick={() => setCurrentPage(item)}
   543→   543→                    >
   544→   544→                      {item}
   545→   545→                    </Button>
   546→   546→                  )
   547→   547→                )}
   548→   548→            </div>
   549→   549→            <Button
   550→   550→              variant="outline"
   551→   551→              size="sm"
   552→   552→              disabled={currentPage >= totalPages}
   553→   553→              onClick={() => setCurrentPage((p) => p + 1)}
   554→   554→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   555→   555→            >
   556→   556→              Suivant
   557→   557→              <ChevronRight className="ml-1 h-4 w-4" />
   558→   558→            </Button>
   559→   559→          </div>
   560→   560→        </div>
   561→   561→      )}
   562→   562→
   563→   563→      {/* ===== REJECT DIALOG ===== */}
   564→   564→      <Dialog open={rejectDialogOpen} onOpenChange={setRejectDialogOpen}>

   567→   567→            <DialogTitle className={isDark ? 'text-slate-100' : 'text-[#333333]'}>
   568→   568→              Rejeter l&apos;enrôlement
   569→   569→            </DialogTitle>
   570→   570→            <DialogDescription>
   571→   571→              {rejectTarget && (
   572→   572→                <>
   573→   573→                  <span className={`font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>

   575→   575→                  </span>{' '}
   576→   576→                  — {rejectTarget.dossierId}
   577→   577→                </>
   578→   578→              )}
   579→   579→            </DialogDescription>
   580→   580→          </DialogHeader>
   581→   581→
   582→   582→          <div className="flex flex-col gap-4">
   583→   583→            {/* Predefined reasons */}
   584→   584→            <div className="flex flex-col gap-2">
   585→   585→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   586→   586→                Raison prédéfinie
   587→   587→              </Label>
   588→   588→              <div className="flex flex-wrap gap-2">
   589→   589→                {PREDEFINED_REASONS.map((reason) => (
   590→   590→                  <Button
   591→   591→                    key={reason}
   592→   592→                    variant={selectedPreset === reason ? 'default' : 'outline'}
   593→   593→                    size="sm"
   594→   594→                    onClick={() => handlePresetReason(reason)}
   595→   595→                    className={
   596→   596→                      selectedPreset === reason
   597→   597→                        ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   598→   598→                        : isDark ? 'text-slate-100' : 'text-[#333333]'
   599→   599→                    }
   600→   600→                  >
   601→   601→                    {reason}
   602→   602→                  </Button>
   603→   603→                ))}
   604→   604→              </div>
   605→   605→            </div>
   606→   606→
   607→   607→            {/* Custom reason */}
   608→   608→            <div className="flex flex-col gap-2">
   609→   609→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   610→   610→                Raison du rejet <span className="text-red-500">*</span>
   611→   611→              </Label>
   612→   612→              <Textarea
   613→   613→                value={rejectReason}
   614→   614→                onChange={(e) => setRejectReason(e.target.value)}
   615→   615→                placeholder="Décrivez la raison du rejet..."
   616→   616→                className="min-h-[100px] resize-none"
   617→   617→              />
   618→   618→            </div>
   619→   619→          </div>
   620→   620→
   621→   621→          <DialogFooter>
   622→   622→            <Button
   623→   623→              variant="outline"
   624→   624→              onClick={() => setRejectDialogOpen(false)}
   625→   625→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   626→   626→            >
   627→   627→              Annuler
   628→   628→            </Button>
   629→   629→            <Button
   630→   630→              variant="destructive"
   631→   631→              onClick={handleConfirmReject}
   632→   632→              disabled={!rejectReason.trim()}
   633→   633→            >
   634→   634→              <XCircle className="mr-1.5 h-4 w-4" />
   635→   635→              Confirmer le rejet
   636→   636→            </Button>
   637→   637→          </DialogFooter>
   638→   638→        </DialogContent>
   639→   639→      </Dialog>
   640→   640→    </div>
   641→   641→  )
