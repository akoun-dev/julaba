   387→   387→export function BoUtilisateursScreen() {
   388→   388→  const { users, createUser, updateUser, boTheme } = useBackofficeStore()
   389→   389→  const isDark = boTheme === 'dark'
   390→   390→
   391→   391→  // Local state
   392→   392→  const [search, setSearch] = useState('')
   393→   393→  const [roleFilter, setRoleFilter] = useState<RoleFilter>('tous')
   394→   394→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('tous')
   395→   395→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   396→   396→  const [editingUser, setEditingUser] = useState<BoUser | null>(null)
   397→   397→
   398→   398→  // Filtered users
   399→   399→  const filteredUsers = useMemo(() => {
   400→   400→    return users.filter((u) => {
   401→   401→      if (search) {
   402→   402→        const q = search.toLowerCase()
   403→   403→        if (
   404→   404→          !u.name.toLowerCase().includes(q) &&
   405→   405→          !u.email.toLowerCase().includes(q) &&
   406→   406→          !(u.zone ?? '').toLowerCase().includes(q)
   407→   407→        ) {
   408→   408→          return false
   409→   409→        }
   410→   410→      }
   411→   411→      if (roleFilter !== 'tous' && u.role !== roleFilter) return false
   412→   412→      if (statusFilter === 'actif' && !u.isActive) return false
   413→   413→      if (statusFilter === 'inactif' && u.isActive) return false
   414→   414→      return true
   415→   415→    })
   416→   416→  }, [users, search, roleFilter, statusFilter])
   417→   417→
   418→   418→  // Stats
   419→   419→  const stats = useMemo(() => {
   420→   420→    const actifs = users.filter((u) => u.isActive).length
   421→   421→    const inactifs = users.length - actifs
   422→   422→    return { total: users.length, actifs, inactifs }
   423→   423→  }, [users])
   424→   424→
   425→   425→  // Handlers
   426→   426→  const handleCreate = (data: UserFormState) => {
   427→   427→    createUser({
   428→   428→      name: data.name,
   429→   429→      email: data.email,
   430→   430→      role: data.role,
   431→   431→      zone: data.zone || undefined,
   432→   432→      isActive: data.isActive,
   433→   433→    })
   434→   434→  }
   435→   435→
   436→   436→  const handleEdit = (data: UserFormState) => {
   437→   437→    if (!editingUser) return
   438→   438→    updateUser(editingUser.id, {
   439→   439→      name: data.name,
   440→   440→      email: data.email,
   441→   441→      role: data.role,
   442→   442→      zone: data.zone || undefined,
   443→   443→      isActive: data.isActive,
   444→   444→    })
   445→   445→    setEditingUser(null)
   446→   446→  }
   447→   447→
   448→   448→  const handleToggleActive = (user: BoUser) => {
   449→   449→    updateUser(user.id, { isActive: !user.isActive })
   450→   450→  }
   451→   451→
   452→   452→  return (
   453→   453→    <div className={'space-y-6 p-6 ' + (isDark ? 'bg-slate-900' : 'bg-[#F8FAFC]')}>
   454→   454→      {/* ===== HEADER ===== */}
   455→   455→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   456→   456→        <div>
   457→   457→          <h1
   458→   458→            className={`text-2xl font-bold tracking-tight ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   459→   459→          >
   460→   460→            UTILISATEURS BACKOFFICE
   461→   461→          </h1>
   462→   462→          <p className={`mt-1 text-sm ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>
   463→   463→            Gestion des comptes et permissions d\'accès au backoffice
   464→   464→          </p>
   465→   465→        </div>
   466→   466→        <Button
   467→   467→          onClick={() => setShowCreateDialog(true)}
   468→   468→          className="gap-2 text-white"
   469→   469→        >
   470→   470→          <Plus className="h-4 w-4" />
   471→   471→          Créer utilisateur
   472→   472→        </Button>
   473→   473→      </div>
   474→   474→
   475→   475→      {/* ===== STATS CARDS ===== */}
   476→   476→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   477→   477→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   478→   478→          <CardContent className="flex items-center gap-3 p-4">
   479→   479→            <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-lg ${isDark ? 'bg-slate-700' : 'bg-gray-100'}`}>
   480→   480→              <Users className={`h-5 w-5 ${isDark ? 'text-slate-100' : 'text-slate-900'}`} />
