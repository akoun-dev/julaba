     1→'use client'
     2→
     3→import { useState, useMemo, useCallback } from 'react'
     4→import {
     5→  Search,
     6→  Eye,
     7→  Pause,
     8→  PlayCircle,
     9→  Download,
    10→  Bell,
    11→  ChevronLeft,
    12→  ChevronRight,
    13→  MapPin,
    14→  Phone,
    15→  User,
    16→  Calendar,
    17→  ShieldCheck,
    18→  StickyNote,
    19→  X,
    20→} from 'lucide-react'
    21→import { Card, CardContent } from '@/components/ui/card'
    22→import { Button } from '@/components/ui/button'
    23→import { Input } from '@/components/ui/input'
    24→import { Checkbox } from '@/components/ui/checkbox'
    25→import { Separator } from '@/components/ui/separator'
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select'
    33→import {
    34→  Dialog,
    35→  DialogContent,
    36→  DialogHeader,
    37→  DialogTitle,
    38→  DialogDescription,
    39→} from '@/components/ui/dialog'
    40→import {
    41→  useBackofficeStore,
    42→  ACTOR_TYPE_LABELS,
    43→  ACTOR_TYPE_ICONS,
    44→  STATUS_LABELS,
    45→  STATUS_COLORS,
    46→  type BoActor,
    47→} from '@/lib/stores/backoffice-store'
    48→
    49→// ============== CONSTANTS ==============
    50→const ITEMS_PER_PAGE = 15
    51→
    52→type ActorTypeFilter = 'tous' | 'marchand' | 'producteur' | 'cooperatif'
    53→type ActorStatusFilter = 'tous' | 'actif' | 'suspendu' | 'en_attente' | 'rejete'
    54→
    55→// ============== MAIN COMPONENT ==============
    56→
    57→export function BoActeursScreen() {
    58→  const { actors, updateActorStatus, searchQuery, setSearchQuery } =
    59→    useBackofficeStore()
    60→
    61→  // Local state
    62→  const [typeFilter, setTypeFilter] = useState<ActorTypeFilter>('tous')
    63→  const [statusFilter, setStatusFilter] = useState<ActorStatusFilter>('tous')
    64→  const [zoneFilter, setZoneFilter] = useState<string>('tous')
    65→  const [currentPage, setCurrentPage] = useState(1)
    66→  const [selectedActors, setSelectedActors] = useState<Set<string>>(new Set())
    67→  const [showDetailModal, setShowDetailModal] = useState(false)
    68→  const [detailActor, setDetailActor] = useState<BoActor | null>(null)
    69→
    70→  // Unique zones from data
    71→  const zones = useMemo(() => {
    72→    const zoneSet = new Set(actors.map((a) => a.zone))
    73→    return Array.from(zoneSet).sort()
    74→  }, [actors])
    75→
    76→  // Filtered actors
    77→  const filteredActors = useMemo(() => {
    78→    return actors.filter((actor) => {
    79→      // Search filter
    80→      if (searchQuery) {
    81→        const q = searchQuery.toLowerCase()
    82→        const matchesSearch =
    83→          actor.firstName.toLowerCase().includes(q) ||
    84→          actor.lastName.toLowerCase().includes(q) ||
    85→          actor.actorId.toLowerCase().includes(q) ||
    86→          actor.phone.includes(q) ||
    87→          actor.zone.toLowerCase().includes(q)
    88→        if (!matchesSearch) return false
    89→      }
    90→
    91→      // Type filter
    92→      if (typeFilter !== 'tous' && actor.type !== typeFilter) return false
    93→
    94→      // Status filter
    95→      if (statusFilter !== 'tous' && actor.status !== statusFilter) return false
    96→
    97→      // Zone filter
    98→      if (zoneFilter !== 'tous' && actor.zone !== zoneFilter) return false
    99→
   100→      return true
   101→    })
   102→  }, [actors, searchQuery, typeFilter, statusFilter, zoneFilter])
   103→
   104→  // Counters
   105→  const counts = useMemo(() => {
   106→    return {
   107→      total: actors.length,
   108→      marchands: actors.filter((a) => a.type === 'marchand').length,
   109→      producteurs: actors.filter((a) => a.type === 'producteur').length,
   110→      cooperatives: actors.filter((a) => a.type === 'cooperatif').length,
   111→    }
   112→  }, [actors])
   113→
   114→  // Pagination
   115→  const totalPages = Math.max(1, Math.ceil(filteredActors.length / ITEMS_PER_PAGE))
   116→  const paginatedActors = useMemo(() => {
   117→    const start = (currentPage - 1) * ITEMS_PER_PAGE
   118→    return filteredActors.slice(start, start + ITEMS_PER_PAGE)
   119→  }, [filteredActors, currentPage])
   120→
   121→  // Reset page when filters change
   122→  const handleTypeFilterChange = useCallback((val: string) => {
   123→    setTypeFilter(val as ActorTypeFilter)
   124→    setCurrentPage(1)
   125→    setSelectedActors(new Set())
   126→  }, [])
   127→
   128→  const handleStatusFilterChange = useCallback((val: string) => {
   129→    setStatusFilter(val as ActorStatusFilter)
   130→    setCurrentPage(1)
   131→    setSelectedActors(new Set())
   132→  }, [])
   133→
   134→  const handleZoneFilterChange = useCallback((val: string) => {
   135→    setZoneFilter(val)
   136→    setCurrentPage(1)
   137→    setSelectedActors(new Set())
   138→  }, [])
   139→
   140→  const handleSearchChange = useCallback(
   141→    (e: React.ChangeEvent<HTMLInputElement>) => {
   142→      setSearchQuery(e.target.value)
   143→      setCurrentPage(1)
   144→      setSelectedActors(new Set())
   145→    },
   146→    [setSearchQuery]
   147→  )
   148→
   149→  // Selection
   150→  const toggleSelectAll = useCallback(() => {
   151→    if (selectedActors.size === paginatedActors.length) {
   152→      setSelectedActors(new Set())
   153→    } else {
   154→      setSelectedActors(new Set(paginatedActors.map((a) => a.id)))
   155→    }
   156→  }, [selectedActors, paginatedActors])
   157→
   158→  const toggleSelectActor = useCallback((id: string) => {
   159→    setSelectedActors((prev) => {
   160→      const next = new Set(prev)
   161→      if (next.has(id)) {
   162→        next.delete(id)
   163→      } else {
   164→        next.add(id)
   165→      }
   166→      return next
   167→    })
   168→  }, [])
   169→
   170→  const isAllSelected =
   171→    paginatedActors.length > 0 && selectedActors.size === paginatedActors.length
   172→
   173→  // Actions
   174→  const handleViewActor = useCallback((actor: BoActor) => {
   175→    setDetailActor(actor)
   176→    setShowDetailModal(true)
   177→  }, [])
   178→
   179→  const handleSuspend = useCallback(
   180→    (actor: BoActor) => {
   181→      updateActorStatus(actor.id, 'suspendu')
   182→    },
   183→    [updateActorStatus]
   184→  )
   185→
   186→  const handleReactivate = useCallback(
   187→    (actor: BoActor) => {
   188→      updateActorStatus(actor.id, 'actif')
   189→    },
   190→    [updateActorStatus]
   191→  )
   192→
   193→  // CSV Export
   194→  const handleExportCSV = useCallback(() => {
   195→    const selected = actors.filter((a) => selectedActors.has(a.id))
   196→    if (selected.length === 0) return
   197→
   198→    const headers = [
   199→      'ID Acteur',
   200→      'Prénom',
   201→      'Nom',
   202→      'Type',
   203→      'Téléphone',
   204→      'Zone',
   205→      'Statut',
   206→      'Identificateur',
   207→      'Validé par',
   208→      'Date validation',
   209→      'Date création',
   210→    ]
   211→    const rows = selected.map((a) => [
   212→      a.actorId,
   213→      a.firstName,
   214→      a.lastName,
   215→      ACTOR_TYPE_LABELS[a.type],
   216→      a.phone,
   217→      a.zone,
   218→      STATUS_LABELS[a.status],
   219→      a.identificateurName || '',
   220→      a.validatedBy || '',
   221→      a.validatedAt ? new Date(a.validatedAt).toLocaleDateString('fr-FR') : '',
   222→      new Date(a.createdAt).toLocaleDateString('fr-FR'),
   223→    ])
   224→    const csv = [headers.join(';'), ...rows.map((r) => r.join(';'))].join('\n')
   225→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   226→    const url = URL.createObjectURL(blob)
   227→    const link = document.createElement('a')
   228→    link.href = url
   229→    link.download = `acteurs_export_${new Date().toISOString().slice(0, 10)}.csv`
   230→    link.click()
   231→    URL.revokeObjectURL(url)
   232→  }, [actors, selectedActors])
   233→
   234→  // Reset filters
   235→  const resetFilters = useCallback(() => {
   236→    setTypeFilter('tous')
   237→    setStatusFilter('tous')
   238→    setZoneFilter('tous')
   239→    setSearchQuery('')
   240→    setCurrentPage(1)
   241→    setSelectedActors(new Set())
   242→  }, [setSearchQuery])
   243→
   244→  const hasActiveFilters =
   245→    typeFilter !== 'tous' || statusFilter !== 'tous' || zoneFilter !== 'tous' || searchQuery !== ''
   246→
   247→  // ============== RENDER ==============
   248→
   249→  return (
   250→    <div className="space-y-4 p-6">
   251→      {/* ===== HEADER ===== */}
   252→      <div>
   253→        <h1
   254→          className="text-2xl font-bold tracking-tight"
   255→          style={{ color: '#333333' }}
   256→        >
   257→          👥 GESTION DES ACTEURS
   258→        </h1>
   259→        <p className="text-sm text-muted-foreground mt-1">
   260→          Consultez, filtrez et gérez l\'ensemble des acteurs enregistrés sur la
   261→          plateforme Jùlaba.
   262→        </p>
   263→      </div>
   264→
   265→      {/* ===== SEARCH ===== */}
   266→      <Card>
   267→        <CardContent className="p-4">
   268→          <div className="relative max-w-md">
   269→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   270→            <Input
   271→              placeholder="Rechercher par nom, téléphone, ID, zone..."
   272→              value={searchQuery}
   273→              onChange={handleSearchChange}
   274→              className="pl-9"
   275→            />
   276→          </div>
   277→        </CardContent>
   278→      </Card>
   279→
   280→      {/* ===== FILTERS ===== */}
   281→      <Card>
   282→        <CardContent className="p-4">
   283→          <div className="flex flex-wrap items-center gap-3">
   284→            <span className="text-sm font-medium" style={{ color: '#333333' }}>
   285→              Filtres :
   286→            </span>
   287→
   288→            <Select value={typeFilter} onValueChange={handleTypeFilterChange}>
   289→              <SelectTrigger className="w-[170px]">
   290→                <SelectValue placeholder="Type" />
   291→              </SelectTrigger>
   292→              <SelectContent>
   293→                <SelectItem value="tous">Tous les types</SelectItem>
   294→                <SelectItem value="marchand">
   295→                  🏪 Marchand(e)s
   296→                </SelectItem>
   297→                <SelectItem value="producteur">
   298→                  🌾 Producteur(rice)s
   299→                </SelectItem>
   300→                <SelectItem value="cooperatif">
   301→                  🤝 Coopératives
   302→                </SelectItem>
   303→              </SelectContent>
   304→            </Select>
   305→
   306→            <Select
   307→              value={statusFilter}
   308→              onValueChange={handleStatusFilterChange}
   309→            >
   310→              <SelectTrigger className="w-[170px]">
   311→                <SelectValue placeholder="Statut" />
   312→              </SelectTrigger>
   313→              <SelectContent>
   314→                <SelectItem value="tous">Tous les statuts</SelectItem>
   315→                <SelectItem value="actif">✅ Actif</SelectItem>
   316→                <SelectItem value="suspendu">⏸️ Suspendu</SelectItem>
   317→                <SelectItem value="en_attente">⏳ En attente</SelectItem>
   318→                <SelectItem value="rejete">❌ Rejeté</SelectItem>
   319→              </SelectContent>
   320→            </Select>
   321→
   322→            <Select value={zoneFilter} onValueChange={handleZoneFilterChange}>
   323→              <SelectTrigger className="w-[180px]">
   324→                <SelectValue placeholder="Zone" />
   325→              </SelectTrigger>
   326→              <SelectContent>
   327→                <SelectItem value="tous">Toutes les zones</SelectItem>
   328→                {zones.map((z) => (
   329→                  <SelectItem key={z} value={z}>
   330→                    📍 {z}
   331→                  </SelectItem>
   332→                ))}
   333→              </SelectContent>
   334→            </Select>
   335→
   336→            {hasActiveFilters && (
   337→              <Button
   338→                variant="ghost"
   339→                size="sm"
   340→                onClick={resetFilters}
   341→                className="text-muted-foreground hover:text-foreground"
   342→              >
   343→                <X className="size-3.5 mr-1" />
   344→                Réinitialiser
   345→              </Button>
   346→            )}
   347→          </div>
   348→        </CardContent>
   349→      </Card>
   350→
   351→      {/* ===== COUNTER BAR ===== */}
   352→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   353→        <Card>
   354→          <CardContent className="p-4 flex items-center gap-3">
   355→            <div className="flex items-center justify-center size-10 rounded-lg bg-gray-100 text-lg">
   356→              👥
   357→            </div>
   358→            <div>
   359→              <p className="text-2xl font-bold" style={{ color: '#333333' }}>
   360→                {counts.total}
   361→              </p>
   362→              <p className="text-xs text-muted-foreground">Total acteurs</p>
   363→            </div>
   364→          </CardContent>
   365→        </Card>
   366→        <Card>
   367→          <CardContent className="p-4 flex items-center gap-3">
   368→            <div className="flex items-center justify-center size-10 rounded-lg bg-orange-50 text-lg">
   369→              🏪
   370→            </div>
   371→            <div>
   372→              <p className="text-2xl font-bold" style={{ color: '#333333' }}>
   373→                {counts.marchands}
   374→              </p>
   375→              <p className="text-xs text-muted-foreground">Marchands</p>
   376→            </div>
   377→          </CardContent>
   378→        </Card>
   379→        <Card>
   380→          <CardContent className="p-4 flex items-center gap-3">
   381→            <div className="flex items-center justify-center size-10 rounded-lg bg-green-50 text-lg">
   382→              🌾
   383→            </div>
   384→            <div>
   385→              <p className="text-2xl font-bold" style={{ color: '#333333' }}>
   386→                {counts.producteurs}
   387→              </p>
   388→              <p className="text-xs text-muted-foreground">Producteurs</p>
   389→            </div>
   390→          </CardContent>
   391→        </Card>
   392→        <Card>
   393→          <CardContent className="p-4 flex items-center gap-3">
   394→            <div className="flex items-center justify-center size-10 rounded-lg bg-amber-50 text-lg">
   395→              🤝
   396→            </div>
   397→            <div>
   398→              <p className="text-2xl font-bold" style={{ color: '#333333' }}>
   399→                {counts.cooperatives}
   400→              </p>
   401→              <p className="text-xs text-muted-foreground">Coopératives</p>
   402→            </div>
   403→          </CardContent>
   404→        </Card>
   405→      </div>
   406→
   407→      {/* ===== BULK ACTIONS ===== */}
   408→      {selectedActors.size > 0 && (
   409→        <Card className="border-l-4 border-l-amber-500">
   410→          <CardContent className="p-4 flex items-center justify-between">
   411→            <span className="text-sm font-medium" style={{ color: '#333333' }}>
   412→              <strong>{selectedActors.size}</strong> acteur(s) sélectionné(s)
   413→            </span>
   414→            <div className="flex items-center gap-2">
   415→              <Button size="sm" variant="outline" onClick={handleExportCSV}>
   416→                <Download className="size-3.5 mr-1.5" />
   417→                Exporter CSV
   418→              </Button>
   419→              <Button size="sm" variant="outline">
   420→                <Bell className="size-3.5 mr-1.5" />
   421→                Notifier
   422→              </Button>
   423→            </div>
   424→          </CardContent>
   425→        </Card>
   426→      )}
   427→
   428→      {/* ===== DATA TABLE ===== */}
   429→      <Card>
   430→        <CardContent className="p-0">
   431→          <div className="max-h-[620px] overflow-y-auto">
   432→            <table className="w-full text-sm">
   433→              <thead className="sticky top-0 z-10 bg-gray-50 border-b">
   434→                <tr>
   435→                  <th className="w-10 px-3 py-3 text-left">
   436→                    <Checkbox
   437→                      checked={isAllSelected}
   438→                      onCheckedChange={toggleSelectAll}
   439→                      aria-label="Sélectionner tout"
   440→                    />
   441→                  </th>
   442→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   443→                    ID
   444→                  </th>
   445→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   446→                    Nom complet
   447→                  </th>
   448→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   449→                    Type
   450→                  </th>
   451→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   452→                    Zone
   453→                  </th>
   454→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   455→                    Téléphone
   456→                  </th>
   457→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   458→                    Identificateur
   459→                  </th>
   460→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   461→                    Statut
   462→                  </th>
   463→                  <th className="px-3 py-3 text-right font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   464→                    Actions
   465→                  </th>
   466→                </tr>
   467→              </thead>
   468→              <tbody className="divide-y">
   469→                {paginatedActors.length === 0 ? (
   470→                  <tr>
   471→                    <td colSpan={9} className="px-3 py-12 text-center text-muted-foreground">
   472→                      <div className="flex flex-col items-center gap-2">
   473→                        <User className="size-8 opacity-30" />
   474→                        <p>Aucun acteur trouvé</p>
   475→                        {hasActiveFilters && (
   476→                          <Button
   477→                            variant="link"
   478→                            size="sm"
   479→                            onClick={resetFilters}
   480→                          >
   481→                            Réinitialiser les filtres
   482→                          </Button>
   483→                        )}
   484→                      </div>
   485→                    </td>
   486→                  </tr>
   487→                ) : (
   488→                  paginatedActors.map((actor) => {
   489→                    const isSelected = selectedActors.has(actor.id)
   490→                    return (
   491→                      <tr
   492→                        key={actor.id}
   493→                        className={`${
   494→                          isSelected
   495→                            ? 'bg-amber-50/60'
   496→                            : 'hover:bg-gray-50/80'
   497→                        } transition-colors`}
   498→                      >
   499→                        <td className="px-3 py-3">
   500→                          <Checkbox
   501→                            checked={isSelected}
   502→                            onCheckedChange={() => toggleSelectActor(actor.id)}
   503→                            aria-label={`Sélectionner ${actor.firstName} ${actor.lastName}`}
   504→                          />
   505→                        </td>
   506→                        <td className="px-3 py-3">
   507→                          <span className="font-mono text-xs font-semibold" style={{ color: '#333333' }}>
   508→                            {actor.actorId}
   509→                          </span>
   510→                        </td>
   511→                        <td className="px-3 py-3">
   512→                          <div className="flex items-center gap-2">
   513→                            <div className="flex items-center justify-center size-7 rounded-full bg-gray-200 text-xs font-semibold text-gray-600 shrink-0">
   514→                              {actor.firstName.charAt(0)}
   515→                              {actor.lastName.charAt(0)}
   516→                            </div>
   517→                            <span className="font-medium" style={{ color: '#333333' }}>
   518→                              {actor.firstName} {actor.lastName}
   519→                            </span>
   520→                          </div>
   521→                        </td>
   522→                        <td className="px-3 py-3">
   523→                          <span className="inline-flex items-center gap-1.5">
   524→                            <span>{ACTOR_TYPE_ICONS[actor.type]}</span>
   525→                            <span className="text-xs text-muted-foreground">
   526→                              {ACTOR_TYPE_LABELS[actor.type]}
   527→                            </span>
   528→                          </span>
   529→                        </td>
   530→                        <td className="px-3 py-3">
   531→                          <span className="text-xs text-muted-foreground">
   532→                            {actor.zone}
   533→                          </span>
   534→                        </td>
   535→                        <td className="px-3 py-3">
   536→                          <span className="text-xs font-mono text-muted-foreground">
   537→                            {actor.phone}
   538→                          </span>
   539→                        </td>
   540→                        <td className="px-3 py-3">
   541→                          <span className="text-xs text-muted-foreground">
   542→                            {actor.identificateurName || '—'}
   543→                          </span>
   544→                        </td>
   545→                        <td className="px-3 py-3">
   546→                          <span
   547→                            className={`inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium ${
   548→                              STATUS_COLORS[actor.status] || 'bg-gray-100 text-gray-800'
   549→                            }`}
   550→                          >
   551→                            {STATUS_LABELS[actor.status]}
   552→                          </span>
   553→                        </td>
   554→                        <td className="px-3 py-3 text-right">
   555→                          <div className="flex items-center justify-end gap-1">
   556→                            <Button
   557→                              size="sm"
   558→                              variant="ghost"
   559→                              className="size-7 p-0 text-muted-foreground hover:text-foreground"
   560→                              onClick={() => handleViewActor(actor)}
   561→                              title="Voir les détails"
   562→                            >
   563→                              <Eye className="size-3.5" />
   564→                            </Button>
   565→                            {actor.status === 'actif' && (
   566→                              <Button
   567→                                size="sm"
   568→                                variant="ghost"
   569→                                className="size-7 p-0 text-amber-600 hover:text-amber-700 hover:bg-amber-50"
   570→                                onClick={() => handleSuspend(actor)}
   571→                                title="Suspendre cet acteur"
   572→                              >
   573→                                <Pause className="size-3.5" />
   574→                              </Button>
   575→                            )}
   576→                            {actor.status === 'suspendu' && (
   577→                              <Button
