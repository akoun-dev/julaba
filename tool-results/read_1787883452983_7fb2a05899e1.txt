   376→   376→function mapDashboardFromApi(d: Record<string, unknown>): DashboardData {
   377→   377→  const actorCountsByRegionRaw = d.actorCountsByRegion as Record<string, number> | undefined
   378→   378→  const actorCountsByRegion = actorCountsByRegionRaw
   379→   379→    ? Object.entries(actorCountsByRegionRaw).map(([name, count]) => ({ name, count }))
   380→   380→    : []
   381→   381→
   382→   382→  const dailyEnrolmentTrendRaw = d.dailyEnrolmentTrend as { date?: string; day?: string; count: number }[] | undefined
   383→   383→  const dailyEnrolmentTrend = (dailyEnrolmentTrendRaw || []).map((item) => ({
   384→   384→    day: item.day || item.date || '',
   385→   385→    count: item.count,
   386→   386→  }))
   387→   387→
   388→   388→  const topIdentificateursRaw = d.topIdentificateurs as { name: string; zone?: string; count: number }[] | undefined
   389→   389→  const topIdentificateurs = (topIdentificateursRaw || []).map((item) => ({
   390→   390→    name: item.name,
   391→   391→    zone: item.zone || '',
   392→   392→    count: item.count,
   393→   393→  }))
   394→   394→
   395→   395→  const systemHealthRaw = d.systemHealth as Record<string, unknown> | undefined
   396→   396→  const systemHealth = systemHealthRaw
   397→   397→    ? [{
   398→   398→        name: 'Plateforme',
   399→   399→        status: (systemHealthRaw.status as string) || 'operationnel',
   400→   400→        latency: (systemHealthRaw.responseTime as number) || 0,
   401→   401→      }]
   402→   402→    : []
   403→   403→
   404→   404→  const unacknowledgedAlertsRaw = d.unacknowledgedAlerts
   405→   405→  const unacknowledgedAlerts = Array.isArray(unacknowledgedAlertsRaw)
   406→   406→    ? unacknowledgedAlertsRaw.length
   407→   407→    : (unacknowledgedAlertsRaw as number) || 0
   408→   408→
   409→   409→  return {
   410→   410→    totalActors: (d.totalActors as number) || 0,
   411→   411→    activeActors: (d.activeActors as number) || 0,
   412→   412→    suspendedActors: (d.suspendedActors as number) || 0,
   413→   413→    totalEnrolments: (d.totalEnrolments as number) || 0,
   414→   414→    pendingEnrolments: (d.pendingEnrolments as number) || 0,
   415→   415→    totalZones: (d.totalZones as number) || 0,
   416→   416→    totalMissions: (d.totalMissions as number) || 0,
   417→   417→    activeMissions: (d.activeMissions as number) || 0,
   418→   418→    actorCountsByRegion,
   419→   419→    dailyEnrolmentTrend,
   420→   420→    topIdentificateurs,
   421→   421→    dataQuality: { photos: 0, gps: 0, phones: 0 },
   422→   422→    systemHealth,
   423→   423→    unacknowledgedAlerts,
   424→   424→  }
   425→   425→}

   442→   442→      toggleSidebar: () => set({ sidebarCollapsed: !get().sidebarCollapsed }),

   447→   447→      setLoading: (v) => set({ loading: v }),
   448→   448→      setError: (e) => set({ error: e }),
   449→   449→
   450→   450→      // Data - empty initial state
   451→   451→      users: [],

   453→   453→      enrolments: [],

   462→   462→        activeUsers: 0,
   463→   463→      },
   464→   464→      dashboard: null,
   465→   465→
   466→   466→      // ============== FETCH FUNCTIONS ==============
   467→   467→
   468→   468→      fetchUsers: async () => {
   469→   469→        set({ loading: true, error: null })
   470→   470→        try {
   471→   471→          const res = await fetch('/api/backoffice/users')

   638→   638→            actors: s.actors.map((a) => (a.id === actorId ? { ...a, status: 'actif' as const } : a)),

   695→   695→              e.id === enrolmentId
