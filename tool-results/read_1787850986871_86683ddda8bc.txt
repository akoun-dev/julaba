    76→    76→function formatDateTime(iso: string) {
    77→    77→  const d = new Date(iso)
    78→    78→  return d.toLocaleDateString('fr-FR', {
    79→    79→    day: '2-digit',
    80→    80→    month: '2-digit',
    81→    81→    year: 'numeric',
    82→    82→    hour: '2-digit',
    83→    83→    minute: '2-digit',
    84→    84→  })
    85→    85→}
    86→    86→
    87→    87→function getTimeAgo(iso: string) {
    88→    88→  const now = Date.now()
    89→    89→  const then = new Date(iso).getTime()
    90→    90→  const diff = now - then
    91→    91→  const minutes = Math.floor(diff / 60000)
    92→    92→  if (minutes < 1) return 'à l\'instant'
    93→    93→  if (minutes < 60) return `il y a ${minutes}min`
    94→    94→  const hours = Math.floor(minutes / 60)
    95→    95→  if (hours < 24) return `il y a ${hours}h`
    96→    96→  const days = Math.floor(hours / 24)
    97→    97→  if (days < 7) return `il y a ${days}j`
    98→    98→  return formatDateTime(iso)
    99→    99→}

   101→   101→function getDateRange(dateRange: DateRangeFilter): { start: Date; end: Date } {
   102→   102→  const now = new Date()
   103→   103→  const end = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59, 999)
   104→   104→  let start: Date
   105→   105→
   106→   106→  switch (dateRange) {
   107→   107→    case 'aujourdhui':
   108→   108→      start = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 0, 0, 0, 0)
   109→   109→      break
   110→   110→    case 'semaine': {
   111→   111→      const dayOfWeek = now.getDay() || 7
   112→   112→      start = new Date(now)
   113→   113→      start.setDate(now.getDate() - dayOfWeek + 1)
   114→   114→      start.setHours(0, 0, 0, 0)
   115→   115→      break
   116→   116→    }
   117→   117→    case 'mois':
   118→   118→      start = new Date(now.getFullYear(), now.getMonth(), 1, 0, 0, 0, 0)
   119→   119→      break
   120→   120→    case 'trimestre': {
   121→   121→      const quarter = Math.floor(now.getMonth() / 3)
   122→   122→      start = new Date(now.getFullYear(), quarter * 3, 1, 0, 0, 0, 0)
   123→   123→      break
   124→   124→    }
   125→   125→    default:
   126→   126→      start = new Date(2000, 0, 1)
   127→   127→  }
   128→   128→
   129→   129→  return { start, end }
   130→   130→}

   132→   132→function truncateString(str: string, maxLen: number) {
   133→   133→  if (!str) return '—'
   134→   134→  return str.length > maxLen ? str.slice(0, maxLen) + '…' : str
   135→   135→}

   137→   137→function formatJsonDetails(details: string | undefined) {
   138→   138→  if (!details) return null
   139→   139→  try {
   140→   140→    const parsed = JSON.parse(details)
   141→   141→    return JSON.stringify(parsed, null, 2)
   142→   142→  } catch {
   143→   143→    return details
   144→   144→  }
   145→   145→}

   149→   149→function downloadCsv(entries: AuditEntry[], filename: string) {
   150→   150→  const headers = [
   151→   151→    'Date/Heure',
   152→   152→    'Utilisateur',
   153→   153→    'Email',
   154→   154→    'Action',
   155→   155→    'Module',
   156→   156→    'Adresse IP',
   157→   157→    'User-Agent',
   158→   158→    'Détails',
   159→   159→  ]
   160→   160→  const rows = entries.map((e) => [
   161→   161→    formatDateTime(e.timestamp),
   162→   162→    e.userName,
   163→   163→    e.userEmail,
   164→   164→    e.action,
   165→   165→    e.module,
   166→   166→    e.ipAddress ?? '',
   167→   167→    e.userAgent ?? '',
   168→   168→    e.details ?? '',
   169→   169→  ])
   170→   170→
   171→   171→  const csvContent = [
   172→   172→    headers.join(';'),
   173→   173→    ...rows.map((r) => r.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(';')),
   174→   174→  ].join('\n')
   175→   175→
   176→   176→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   177→   177→  const url = URL.createObjectURL(blob)
   178→   178→  const link = document.createElement('a')
   179→   179→  link.href = url
   180→   180→  link.download = filename
   181→   181→  document.body.appendChild(link)
   182→   182→  link.click()
   183→   183→  document.body.removeChild(link)
   184→   184→  URL.revokeObjectURL(url)
   185→   185→}
