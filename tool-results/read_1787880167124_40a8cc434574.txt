    86→// ============== HELPERS ==============
    87→
    88→function isToday(dateStr: string): boolean {
    89→  const d = new Date(dateStr)
    90→  const now = new Date()
    91→  return (
    92→    d.getFullYear() === now.getFullYear() &&
    93→    d.getMonth() === now.getMonth() &&
    94→    d.getDate() === now.getDate()
    95→  )
    96→}
    97→
    98→function isThisWeek(dateStr: string): boolean {
    99→  const d = new Date(dateStr)
   100→  const now = new Date()
   101→  const startOfWeek = new Date(now)
   102→  startOfWeek.setDate(now.getDate() - now.getDay() + 1)
   103→  startOfWeek.setHours(0, 0, 0, 0)
   104→  return d >= startOfWeek
   105→}
   106→
   107→function isThisMonth(dateStr: string): boolean {
   108→  const d = new Date(dateStr)
   109→  const now = new Date()
   110→  return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   111→}
   112→
   113→function matchesDateRange(dateStr: string, range: DateRange): boolean {
   114→  switch (range) {
   115→    case 'aujourdhui':
   116→      return isToday(dateStr)
   117→    case 'semaine':
   118→      return isThisWeek(dateStr)
   119→    case 'mois':
   120→      return isThisMonth(dateStr)
   121→    case 'tous':
   122→    default:
   123→      return true
   124→  }
   125→}
   126→
   127→function formatDate(dateStr: string): string {
   128→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   129→    day: '2-digit',
   130→    month: '2-digit',
   131→    year: 'numeric',
   132→    hour: '2-digit',
   133→    minute: '2-digit',
   134→  })
   135→}

   554→      )}
   555→
   556→      {/* ===== REJECT DIALOG ===== */}
   557→      <Dialog open={rejectDialogOpen} onOpenChange={setRejectDialogOpen}>
   558→        <DialogContent className="sm:max-w-md">
   559→          <DialogHeader>
   560→            <DialogTitle className={isDark ? 'text-slate-100' : 'text-[#333333]'}>
   561→              Rejeter l&apos;enrôlement
   562→            </DialogTitle>
   563→            <DialogDescription>
   564→              {rejectTarget && (
   565→                <>
   566→                  <span className={`font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   567→                    {rejectTarget.actorName}
   568→                  </span>{' '}
   569→                  — {rejectTarget.dossierId}
   570→                </>
   571→              )}
   572→            </DialogDescription>
   573→          </DialogHeader>
   574→
   575→          <div className="flex flex-col gap-4">
   576→            {/* Predefined reasons */}
   577→            <div className="flex flex-col gap-2">
   578→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   579→                Raison prédéfinie
   580→              </Label>
   581→              <div className="flex flex-wrap gap-2">
   582→                {PREDEFINED_REASONS.map((reason) => (
   583→                  <Button
   584→                    key={reason}
   585→                    variant={selectedPreset === reason ? 'default' : 'outline'}
   586→                    size="sm"
   587→                    onClick={() => handlePresetReason(reason)}
   588→                    className={
   589→                      selectedPreset === reason
   590→                        ? 'bg-[#333333] text-white hover:bg-[#333333]/90'
   591→                        : isDark ? 'text-slate-100' : 'text-[#333333]'
   592→                    }
   593→                  >
   594→                    {reason}
   595→                  </Button>
   596→                ))}
   597→              </div>
   598→            </div>
   599→
   600→            {/* Custom reason */}
   601→            <div className="flex flex-col gap-2">
   602→              <Label className={`text-sm font-medium ${isDark ? 'text-slate-100' : 'text-[#333333]'}`}>
   603→                Raison du rejet <span className="text-red-500">*</span>
   604→              </Label>
   605→              <Textarea
   606→                value={rejectReason}
   607→                onChange={(e) => setRejectReason(e.target.value)}
   608→                placeholder="Décrivez la raison du rejet..."
   609→                className="min-h-[100px] resize-none"
   610→              />
   611→            </div>
   612→          </div>
   613→
   614→          <DialogFooter>
   615→            <Button
   616→              variant="outline"
   617→              onClick={() => setRejectDialogOpen(false)}
   618→              className={isDark ? 'text-slate-100' : 'text-[#333333]'}
   619→            >
   620→              Annuler
   621→            </Button>
   622→            <Button
   623→              variant="destructive"
   624→              onClick={handleConfirmReject}
   625→              disabled={!rejectReason.trim()}
   626→            >
   627→              <XCircle className="mr-1.5 h-4 w-4" />
   628→              Confirmer le rejet
   629→            </Button>
   630→          </DialogFooter>
   631→        </DialogContent>
   632→      </Dialog>
   633→    </div>
   634→  )
