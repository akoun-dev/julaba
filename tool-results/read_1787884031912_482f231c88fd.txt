   105→      if (statusFilter !== 'tous' && actor.status !== statusFilter) return false
   106→

   108→      if (zoneFilter !== 'tous' && actor.zone !== zoneFilter) return false

   350→              <Button

   352→                size="sm"

   373→                {counts.total}
   374→              </p>
   375→              <p className="text-xs text-muted-foreground">Total acteurs</p>
   376→            </div>
   377→          </CardContent>
   378→        </Card>
   379→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
