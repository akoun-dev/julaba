    27→    27→const simpleHash = (str: string) => {
    28→    28→  let hash = 0
    29→    29→  for (let i = 0; i < str.length; i++) {
    30→    30→    const char = str.charCodeAt(i)
    31→    31→    hash = ((hash << 5) - hash) + char
    32→    32→    hash |= 0
    33→    33→  }
    34→    34→  return hash.toString()
    35→    35→}
