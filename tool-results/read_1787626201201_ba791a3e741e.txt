    97→function loadMerchantProfile(phone: string): MerchantProfile {
    98→  if (typeof window === 'undefined') return { ...defaultProfile }
    99→  const raw = localStorage.getItem(`julaba-profile-${phone}`)
   100→  return raw ? { ...defaultProfile, ...JSON.parse(raw) } : { ...defaultProfile }
   101→}

   103→function saveMerchantProfile(phone: string, profile: MerchantProfile) {
   104→  localStorage.setItem(`julaba-profile-${phone}`, JSON.stringify(profile))
   105→}

   369→function SecuriteSubScreen({
   370→  profile,
   371→  setProfile,
   372→  phone,
   373→  soleilMode,
   374→  onBack,
   375→}: {
   376→  profile: MerchantProfile
   377→  setProfile: (p: MerchantProfile) => void
   378→  phone: string
   379→  soleilMode: boolean
   380→  onBack: () => void
   381→}) {
   382→  const [pinStep, setPinStep] = useState<'idle' | 'old' | 'new' | 'confirm'>('idle')
   383→  const [pin, setPin] = useState('')
   384→  const [newPin, setNewPin] = useState('')
   385→  const [confirmPinVal, setConfirmPinVal] = useState('')
   386→  const [error, setError] = useState('')
   387→
   388→  const authData = loadMerchantAuthData(phone)
   389→  const authMethod = authData?.authMethod || 'pin'
   390→
   391→  const handlePinDigit = (digit: string) => {
   392→    if (pinStep === 'old') {
   393→      if (pin.length < 4) {
   394→        const next = pin + digit
   395→        setPin(next)
   396→        if (next.length === 4) {
   397→          // Verify old PIN
   398→          const merchantRaw = localStorage.getItem(`julaba-merchant-${phone}`)
   399→          if (merchantRaw) {
   400→            const merchant = JSON.parse(merchantRaw)
   401→            const simpleHash = (str: string) => {
   402→              let hash = 0
   403→              for (let i = 0; i < str.length; i++) {
   404→                const char = str.charCodeAt(i)
   405→                hash = ((hash << 5) - hash) + char
   406→                hash |= 0
   407→              }
   408→              return hash.toString()
   409→            }
   410→            if (simpleHash(next) === merchant.pinHash) {
   411→              setPinStep('new')
   412→              setError('')
   413→            } else {
   414→              setError('Code incorrect')
   415→              setPin('')
   416→              tataSpeak('Code incorrect')
   417→              haptic('error')
   418→            }
   419→          }
   420→        }
   421→      }
   422→    } else if (pinStep === 'new') {
   423→      if (newPin.length < 4) {
   424→        const next = newPin + digit
   425→        setNewPin(next)
   426→        if (next.length === 4) {
   427→          setPinStep('confirm')
   428→        }
   429→      }
   430→    } else if (pinStep === 'confirm') {
   431→      if (confirmPinVal.length < 4) {
   432→        const next = confirmPinVal + digit
   433→        setConfirmPinVal(next)
   434→        if (next.length === 4) {
   435→          if (next === newPin) {
   436→            // Save new PIN
   437→            const merchantRaw = localStorage.getItem(`julaba-merchant-${phone}`)
   438→            if (merchantRaw) {
   439→              const merchant = JSON.parse(merchantRaw)
   440→              const simpleHash = (str: string) => {
   441→                let hash = 0
   442→                for (let i = 0; i < str.length; i++) {
   443→                  const char = str.charCodeAt(i)
   444→                  hash = ((hash << 5) - hash) + char
   445→                  hash |= 0
   446→                }
   447→                return hash.toString()
   448→              }
   449→              merchant.pinHash = simpleHash(next)
   450→              localStorage.setItem(`julaba-merchant-${phone}`, JSON.stringify(merchant))
   451→              // Add to connection history
   452→              const updatedProfile = {
   453→                ...profile,
   454→                connectionHistory: [
   455→                  { date: new Date().toISOString(), method: 'Changement PIN' },
   456→                  ...profile.connectionHistory,
   457→                ].slice(0, 5),
   458→              }
   459→              setProfile(updatedProfile)
   460→              tataSpeak('Ton nouveau code est enregistré !')
   461→              haptic('success')
   462→              setPinStep('idle')
   463→              setPin('')
   464→              setNewPin('')
   465→              setConfirmPinVal('')
   466→              setError('')
   467→            }
   468→          } else {
   469→            setError('Les codes ne correspondent pas')
   470→            setConfirmPinVal('')
   471→            setPinStep('new')
   472→            setNewPin('')
   473→            tataSpeak('Les codes ne correspondent pas')
   474→            haptic('error')
   475→          }
   476→        }
   477→      }
   478→    }
   479→  }
   480→
   481→  const handlePinDelete = () => {
   482→    if (pinStep === 'old') setPin((p) => p.slice(0, -1))
   483→    else if (pinStep === 'new') setNewPin((p) => p.slice(0, -1))
   484→    else if (pinStep === 'confirm') setConfirmPinVal((p) => p.slice(0, -1))
   485→  }
   486→
   487→  const currentPin = pinStep === 'old' ? pin : pinStep === 'new' ? newPin : confirmPinVal
   488→
   489→  const pinLabels: Record<string, string> = {
   490→    idle: '',
   491→    old: 'Entrez votre code actuel',
   492→    new: 'Entrez le nouveau code',
   493→    confirm: 'Confirmez le nouveau code',
   494→  }
   495→
   496→  const tc = soleilMode ? 'text-black' : ''
   497→
   498→  return (
   499→    <div className="screen-enter pb-24">
   500→      <div className="sticky top-0 z-40 bg-background border-b px-4 py-3">
   501→        <div className="flex items-center gap-2">
   502→          <Button variant="ghost" size="icon" onClick={() => { haptic('light'); onBack() }} className="h-9 w-9">
   503→            <ArrowLeft className="w-5 h-5" />
   504→          </Button>
   505→          <h1 className={soleilMode ? 'text-xl font-bold text-black' : 'text-lg font-bold'}>Sécurité & Connexion</h1>
   506→        </div>
   507→      </div>
   508→
   509→      <div className="px-4 mt-4 space-y-4">
   510→        {/* Current auth method */}
   511→        <Card>
   512→          <CardContent className="p-4">
   513→            <p className={cn('text-xs text-muted-foreground mb-1', tc)}>Méthode de connexion</p>
   514→            <div className="flex items-center gap-2">
   515→              <Badge className="bg-[#C66A2C]/10 text-[#C66A2C] border-0">
   516→                {authMethod === 'pin' ? 'Code PIN' : authMethod === 'pattern' ? 'Motif' : 'PIN + Motif'}
   517→              </Badge>
   518→            </div>
   519→          </CardContent>
   520→        </Card>
   521→
   522→        {/* Change PIN */}
   523→        <Card>
   524→          <CardContent className="p-4">
   525→            <div className="flex items-center justify-between">
   526→              <div className="flex items-center gap-2">
   527→                <Lock className="w-4 h-4 text-muted-foreground" />
   528→                <span className={cn('text-sm font-medium', tc)}>Changer mon code PIN</span>
   529→              </div>
   530→              {pinStep === 'idle' ? (
   531→                <Button
   532→                  variant="outline"
   533→                  size="sm"
   534→                  onClick={() => { haptic('light'); setPinStep('old'); setError(''); setPin(''); setNewPin(''); setConfirmPinVal('') }}
   535→                >
   536→                  Modifier
   537→                </Button>
   538→              ) : (
   539→                <Button
   540→                  variant="ghost"
   541→                  size="sm"
   542→                  onClick={() => { haptic('light'); setPinStep('idle'); setPin(''); setNewPin(''); setConfirmPinVal(''); setError('') }}
   543→                >
   544→                  Annuler
   545→                </Button>
   546→              )}
   547→            </div>
   548→
   549→            {pinStep !== 'idle' && (
   550→              <div className="mt-4 space-y-4">
   551→                <p className={cn('text-sm text-center font-medium', tc)}>{pinLabels[pinStep]}</p>
   552→
   553→                {/* PIN dots */}
   554→                <div className="flex justify-center gap-3">
   555→                  {[0, 1, 2, 3].map((i) => (
   556→                    <div
   557→                      key={i}
   558→                      className={cn(
   559→                        'w-10 h-10 rounded-full border-2 flex items-center justify-center transition-all',
   560→                        i < currentPin.length
   561→                          ? 'bg-[#C66A2C] border-[#C66A2C]'
   562→                          : 'border-muted-foreground/30'
   563→                      )}
   564→                    >
   565→                      {i < currentPin.length && <div className="w-3 h-3 rounded-full bg-white" />}
   566→                    </div>
   567→                  ))}
   568→                </div>
   569→
   570→                {error && (
   571→                  <p className="text-destructive text-sm text-center">{error}</p>
   572→                )}
   573→
   574→                {/* Numeric keypad */}
   575→                <div className="grid grid-cols-3 gap-2 max-w-xs mx-auto">
   576→                  {['1', '2', '3', '4', '5', '6', '7', '8', '9'].map((d) => (
   577→                    <Button
   578→                      key={d}
   579→                      variant="outline"
   580→                      className={cn('h-12 text-lg font-semibold', soleilMode && 'text-base')}
   581→                      onClick={() => handlePinDigit(d)}
   582→                    >
   583→                      {d}
   584→                    </Button>
   585→                  ))}
   586→                  <Button variant="ghost" className="h-12" onClick={handlePinDelete}>
   587→                    ←
   588→                  </Button>
   589→                  <Button
   590→                    variant="outline"
   591→                    className="h-12 text-lg font-semibold"
   592→                    onClick={() => handlePinDigit('0')}
   593→                  >
   594→                    0
   595→                  </Button>
   596→                  <div className="h-12" />
   597→                </div>
   598→              </div>
   599→            )}
   600→          </CardContent>
