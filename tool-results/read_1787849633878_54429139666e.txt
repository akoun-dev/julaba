   102→function formatDate(iso: string) {
   103→  const d = new Date(iso)
   104→  return d.toLocaleDateString('fr-FR', {
   105→    day: '2-digit', month: '2-digit', year: 'numeric',
   106→  })
   107→}
   108→
   109→function formatDateTime(iso: string) {
   110→  const d = new Date(iso)
   111→  return d.toLocaleDateString('fr-FR', {
   112→    day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit',
   113→  })
   114→}

   134→function UserFormDialog({
   135→  open,
   136→  onOpenChange,
   137→  initialData,
   138→  onSubmit,
   139→}: {
   140→  open: boolean
   141→  onOpenChange: (v: boolean) => void
   142→  initialData?: BoUser | null
   143→  onSubmit: (data: UserFormState) => void
   144→}) {
   145→  const [form, setForm] = useState<UserFormState>(emptyForm)
   146→  const isEdit = !!initialData
   147→
   148→  // Reset form when dialog opens
   149→  const handleOpenChange = (v: boolean) => {
   150→    if (v) {
   151→      setForm(
   152→        initialData
   153→          ? {
   154→              name: initialData.name,
   155→              email: initialData.email,
   156→              role: initialData.role,
   157→              zone: initialData.zone ?? '',
   158→              isActive: initialData.isActive,
   159→            }
   160→          : { ...emptyForm }
   161→      )
   162→    }
   163→    onOpenChange(v)
   164→  }
   165→
   166→  const handleSubmit = () => {
   167→    if (!form.name.trim() || !form.email.trim()) return
   168→    onSubmit(form)
   169→    onOpenChange(false)
   170→  }
   171→
   172→  return (
   173→    <Dialog open={open} onOpenChange={handleOpenChange}>
   174→      <DialogContent className="sm:max-w-[480px]">
   175→        <DialogHeader>
   176→          <DialogTitle style={{ color: BO_COLOR }}>
   177→            {isEdit ? 'Modifier l\'utilisateur' : 'Créer un utilisateur'}
   178→          </DialogTitle>
   179→          <DialogDescription>
   180→            {isEdit
   181→              ? 'Modifiez les informations de l\'utilisateur backoffice.'
   182→              : 'Remplissez les informations pour créer un nouvel utilisateur backoffice.'}
   183→          </DialogDescription>
   184→        </DialogHeader>
   185→
   186→        <div className="space-y-4 py-2">
   187→          <div className="space-y-2">
   188→            <Label htmlFor="user-name">Nom complet *</Label>
   189→            <Input
   190→              id="user-name"
   191→              placeholder="Ex: Aminata KONÉ"
   192→              value={form.name}
   193→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   194→            />
   195→          </div>
   196→
   197→          <div className="space-y-2">
   198→            <Label htmlFor="user-email">Email *</Label>
   199→            <Input
   200→              id="user-email"
   201→              type="email"
   202→              placeholder="Ex: aminata@julaba.ci"
   203→              value={form.email}
   204→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   205→            />
   206→          </div>
   207→
   208→          <div className="space-y-2">
   209→            <Label htmlFor="user-role">Rôle *</Label>
   210→            <Select
   211→              value={form.role}
   212→              onValueChange={(v) => setForm({ ...form, role: v as BoRole })}
   213→            >
   214→              <SelectTrigger id="user-role" className="w-full">
   215→                <SelectValue placeholder="Sélectionner un rôle" />
   216→              </SelectTrigger>
   217→              <SelectContent>
   218→                {ALL_ROLES.map((role) => (
   219→                  <SelectItem key={role} value={role}>
   220→                    {ROLE_LABELS[role]}
   221→                  </SelectItem>
   222→                ))}
   223→              </SelectContent>
   224→            </Select>
   225→          </div>
   226→
   227→          <div className="space-y-2">
   228→            <Label htmlFor="user-zone">Zone (optionnel)</Label>
   229→            <Select
   230→              value={form.zone}
   231→              onValueChange={(v) => setForm({ ...form, zone: v })}
   232→            >
   233→              <SelectTrigger id="user-zone" className="w-full">
   234→                <SelectValue placeholder="Aucune zone" />
   235→              </SelectTrigger>
   236→              <SelectContent>
   237→                <SelectItem value="">Aucune zone</SelectItem>
   238→                {ZONES_LIST.map((z) => (
   239→                  <SelectItem key={z} value={z}>
   240→                    {z}
   241→                  </SelectItem>
   242→                ))}
   243→              </SelectContent>
   244→            </Select>
   245→          </div>
   246→        </div>
   247→
   248→        <DialogFooter className="gap-2 pt-2">
   249→          <Button
   250→            variant="outline"
   251→            onClick={() => onOpenChange(false)}
   252→          >
   253→            Annuler
   254→          </Button>
   255→          <Button
   256→            disabled={!form.name.trim() || !form.email.trim()}
   257→            onClick={handleSubmit}
   258→            style={{ backgroundColor: BO_COLOR, color: '#fff' }}
   259→          >
   260→            {isEdit ? 'Enregistrer' : 'Créer'}
   261→          </Button>
   262→        </DialogFooter>
   263→      </DialogContent>
   264→    </Dialog>
   265→  )
   266→}

   384→export function BoUtilisateursScreen() {
   385→  const { users, createUser, updateUser } = useBackofficeStore()
   386→
   387→  // Local state
   388→  const [search, setSearch] = useState('')
   389→  const [roleFilter, setRoleFilter] = useState<RoleFilter>('tous')
   390→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('tous')
   391→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   392→  const [editingUser, setEditingUser] = useState<BoUser | null>(null)
   393→
   394→  // Filtered users
   395→  const filteredUsers = useMemo(() => {
   396→    return users.filter((u) => {
   397→      if (search) {
   398→        const q = search.toLowerCase()
   399→        if (
   400→          !u.name.toLowerCase().includes(q) &&
   401→          !u.email.toLowerCase().includes(q) &&
   402→          !(u.zone ?? '').toLowerCase().includes(q)
   403→        ) {
   404→          return false
   405→        }
   406→      }
   407→      if (roleFilter !== 'tous' && u.role !== roleFilter) return false
   408→      if (statusFilter === 'actif' && !u.isActive) return false
   409→      if (statusFilter === 'inactif' && u.isActive) return false
   410→      return true
   411→    })
   412→  }, [users, search, roleFilter, statusFilter])
   413→
   414→  // Stats
   415→  const stats = useMemo(() => {
   416→    const actifs = users.filter((u) => u.isActive).length
   417→    const inactifs = users.length - actifs
   418→    return { total: users.length, actifs, inactifs }
   419→  }, [users])
   420→
   421→  // Handlers
   422→  const handleCreate = (data: UserFormState) => {
   423→    createUser({
   424→      name: data.name,
   425→      email: data.email,
   426→      role: data.role,
   427→      zone: data.zone || undefined,
   428→      isActive: data.isActive,
   429→    })
   430→  }
   431→
   432→  const handleEdit = (data: UserFormState) => {
   433→    if (!editingUser) return
   434→    updateUser(editingUser.id, {
   435→      name: data.name,
   436→      email: data.email,
   437→      role: data.role,
   438→      zone: data.zone || undefined,
   439→      isActive: data.isActive,
   440→    })
   441→    setEditingUser(null)
   442→  }
   443→
   444→  const handleToggleActive = (user: BoUser) => {
   445→    updateUser(user.id, { isActive: !user.isActive })
   446→  }
   447→
   448→  return (
   449→    <div className="space-y-6 p-6">
   450→      {/* ===== HEADER ===== */}
   451→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   452→        <div>
   453→          <h1
   454→            className="text-2xl font-bold tracking-tight"
   455→            style={{ color: BO_COLOR }}
   456→          >
   457→            UTILISATEURS BACKOFFICE
   458→          </h1>
   459→          <p className="mt-1 text-sm text-gray-500">
   460→            Gestion des comptes et permissions d\'accès au backoffice
   461→          </p>
   462→        </div>
   463→        <Button
   464→          onClick={() => setShowCreateDialog(true)}
   465→          style={{ backgroundColor: BO_COLOR, color: '#fff' }}
   466→          className="gap-2"
   467→        >
   468→          <Plus className="h-4 w-4" />
   469→          Créer utilisateur
   470→        </Button>
   471→      </div>
   472→
   473→      {/* ===== STATS CARDS ===== */}
   474→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   475→        <Card>
   476→          <CardContent className="flex items-center gap-3 p-4">
   477→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-gray-100">
   478→              <Users className="h-5 w-5" style={{ color: BO_COLOR }} />
   479→            </div>
   480→            <div>
   481→              <p className="text-2xl font-bold" style={{ color: BO_COLOR }}>
   482→                {stats.total}
   483→              </p>
   484→              <p className="text-xs text-gray-500">Total utilisateurs</p>
   485→            </div>
   486→          </CardContent>
   487→        </Card>
   488→        <Card>
   489→          <CardContent className="flex items-center gap-3 p-4">
   490→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-50">
   491→              <Check className="h-5 w-5 text-emerald-600" />
   492→            </div>
   493→            <div>
   494→              <p className="text-2xl font-bold text-emerald-700">
   495→                {stats.actifs}
   496→              </p>
   497→              <p className="text-xs text-gray-500">Actifs</p>
