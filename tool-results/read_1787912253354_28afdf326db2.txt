     1→     1→'use client'
     2→     2→
     3→     3→import { useState, useCallback, useEffect } from 'react'
     4→     4→import { useAppStore } from '@/lib/stores/app-store'
     5→     5→import { useBackofficeStore, type BoRole, ROLE_LABELS } from '@/lib/stores/backoffice-store'
     6→     6→import { OrbitOtp } from './orbit-otp'
     7→     7→import { ArrowLeft, Shield, Fingerprint, CheckCircle2, Lock, KeyRound } from 'lucide-react'
     8→     8→
     9→     9→// Types for demo accounts from API (no password exposed)
    10→    10→interface DemoAccount {
    11→    11→  email: string
    12→    12→  name: string
    13→    13→  role: string
    14→    14→  zone: string | null
    15→    15→}
    16→    16→
    17→    17→// Type for authenticated user from login API
    18→    18→interface AuthenticatedUser {
    19→    19→  id: string
    20→    20→  email: string
    21→    21→  name: string
    22→    22→  role: string
    23→    23→  zone: string | null
    24→    24→  isActive: boolean
    25→    25→  lastLogin: string | null
    26→    26→  createdAt: string
    27→    27→}
    28→    28→
    29→    29→type Step = 'credentials' | 'mfa' | 'success'
    30→    30→
    31→    31→export function BoAuthScreen() {
    32→    32→  const { setUserRole, navigate, setAuth } = useAppStore()
    33→    33→  const { setBoAuth } = useBackofficeStore()
    34→    34→
    35→    35→  const [step, setStep] = useState<Step>('credentials')
    36→    36→  const [email, setEmail] = useState('')
    37→    37→  const [password, setPassword] = useState('')
    38→    38→  const [error, setError] = useState('')
    39→    39→  const [loading, setLoading] = useState(false)
    40→    40→  const [verifying, setVerifying] = useState(false)
    41→    41→  const [matchedUser, setMatchedUser] = useState<AuthenticatedUser | null>(null)
    42→    42→  const [showDemo, setShowDemo] = useState(false)

    46→    46→
    47→    47→  // Fetch demo accounts from DB on mount
    48→    48→  useEffect(() => {
    49→    49→    fetch('/api/backoffice/demo-accounts')
    50→    50→      .then((r) => (r.ok ? r.json() : []))

    55→    55→
    56→    56→  const handleLogin = useCallback(() => {
    57→    57→    setError('')

    59→    59→      setError('Veuillez remplir tous les champs')
    60→    60→      return
    61→    61→    }
    62→    62→
    63→    63→    setLoading(true)
    64→    64→    fetch('/api/backoffice/login', {
    65→    65→      method: 'POST',
    66→    66→      headers: { 'Content-Type': 'application/json' },
    67→    67→      body: JSON.stringify({ email, password }),
    68→    68→    })
    69→    69→      .then((res) => {
    70→    70→        if (!res.ok) return res.json().then((d) => { throw new Error(d.erreur || 'Erreur') })
    71→    71→        return res.json()
    72→    72→      })
    73→    73→      .then((user: AuthenticatedUser) => {
    74→    74→        setMatchedUser(user)
    75→    75→        setEmail(user.email)
    76→    76→        setStep('mfa')
    77→    77→      })

    79→    79→        setError(err.message || 'Email ou mot de passe incorrect')
    80→    80→      })

    82→    82→  }, [email, password])
    83→    83→

    98→    98→            id: matchedUser.id,
    99→    99→            email: matchedUser.email,
   100→   100→            name: matchedUser.name,
   101→   101→            role: matchedUser.role as BoRole,
   102→   102→            zone: matchedUser.zone || undefined,
   103→   103→            isActive: matchedUser.isActive,
   104→   104→            lastLogin: matchedUser.lastLogin || undefined,
   105→   105→            createdAt: matchedUser.createdAt,
   106→   106→          })

   112→   112→  )
   113→   113→

   115→   115→    setError('')
   116→   116→    setLoading(true)
   117→   117→    // Authenticate against DB with the demo account
   118→   118→    fetch('/api/backoffice/login', {
   119→   119→      method: 'POST',
   120→   120→      headers: { 'Content-Type': 'application/json' },
   121→   121→      body: JSON.stringify({ email: account.email, password: 'admin123' }),
   122→   122→    })
   123→   123→      .then((res) => {
   124→   124→        if (!res.ok) return res.json().then((d) => { throw new Error(d.erreur || 'Erreur') })
   125→   125→        return res.json()
   126→   126→      })
   127→   127→      .then((user: AuthenticatedUser) => {

   135→   135→      })
   136→   136→      .finally(() => setLoading(false))
   137→   137→  }, [])
   138→   138→
   139→   139→  const handleBack = useCallback(() => {

   141→   141→      setStep('credentials')

   143→   143→      setVerifying(false)
   144→   144→      setOtpResetKey((k) => k + 1)

   146→   146→      navigate('auth')

   148→   148→  }, [step, navigate])
   149→   149→

   172→   172→            Interface d&rsquo;administration sécurisée pour la gestion des acteurs

   263→   263→
   264→   264→              <button
   265→   265→                className="bo-auth-submit-btn"
   266→   266→                onClick={handleLogin}
   267→   267→                disabled={loading}
   268→   268→              >
   269→   269→                {loading ? (
   270→   270→                  <span className="bo-auth-submit-loading">
   271→   271→                    <span className="bo-auth-spinner" />
   272→   272→                    Vérification…
   273→   273→                  </span>
   274→   274→                ) : (
   275→   275→                  'Se connecter'

   278→   278→
   279→   279→              {/* MFA notice */}
   280→   280→              <div className="bo-auth-mfa-notice">

   303→   303→                      Chargement des comptes…

   306→   306→                    <div style={{ textAlign: 'center', padding: '12px', color: 'rgba(255,255,255,0.4)', fontSize: '13px' }}>
   307→   307→                      Aucun compte disponible

   309→   309→                  ) : (

   311→   311→                      <button
   312→   312→                        key={account.email}
   313→   313→                        className="bo-auth-demo-item"
   314→   314→                        onClick={() => handleDemoLogin(account)}
   315→   315→                      >
   316→   316→                        <div className="bo-auth-demo-info">
   317→   317→                          <span className="bo-auth-demo-email">{account.name} — {account.email}</span>
   318→   318→                          <span className="bo-auth-demo-role">{ROLE_LABELS[account.role as BoRole] || account.role}{account.zone ? ` · ${account.zone}` : ''}</span>
   319→   319→                        </div>
   320→   320→                        <span className="bo-auth-demo-badge">{ROLE_LABELS[account.role as BoRole] || account.role}</span>
   321→   321→                      </button>
   322→   322→                    ))
   323→   323→                  )}
   324→   324→                </div>
   325→   325→              )}
   326→   326→            </div>
   327→   327→          )}
   328→   328→
   329→   329→          {/* ====== STEP: MFA ====== */}
   330→   330→          {step === 'mfa' && (
   331→   331→            <div className="bo-auth-card">
   332→   332→              {/* Fingerprint icon */}
   333→   333→              <div className="bo-auth-mfa-icon-wrapper">

   349→   349→                resetKey={otpResetKey}
   350→   350→                onResend={() => {
   351→   351→                  setOtpResetKey((k) => k + 1)
   352→   352→                  setError('')
   353→   353→                }}
   354→   354→              />
   355→   355→
   356→   356→              <p className="bo-auth-mfa-demo-hint">
   357→   357→                Démo : entrez n&rsquo;importe quel code à 6 chiffres
   358→   358→              </p>
   359→   359→
   360→   360→              {/* Back to credentials */}
   361→   361→              <button
   362→   362→                className="bo-auth-mfa-back-btn"
   363→   363→                onClick={() => {
   364→   364→                  setStep('credentials')
   365→   365→                  setError('')
   366→   366→                  setVerifying(false)
   367→   367→                  setOtpResetKey((k) => k + 1)
   368→   368→                }}
   369→   369→              >
   370→   370→                <ArrowLeft size={14} />
   371→   371→                Retour aux identifiants
   372→   372→              </button>
   373→   373→            </div>
   374→   374→          )}
   375→   375→
   376→   376→          {/* ====== STEP: SUCCESS ====== */}
   377→   377→          {step === 'success' && (
   378→   378→            <div className="bo-auth-success">

   380→   380→                <CheckCircle2 className="bo-auth-success-check" size={40} />
