   101→function formatDate(iso: string) {
   102→  const d = new Date(iso)
   103→  return d.toLocaleDateString('fr-FR', {
   104→    day: '2-digit', month: '2-digit', year: 'numeric',
   105→  })
   106→}
   107→
   108→function formatDateTime(iso: string) {
   109→  const d = new Date(iso)
   110→  return d.toLocaleDateString('fr-FR', {
   111→    day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit',
   112→  })

   133→function UserFormDialog({
   134→  open,
   135→  onOpenChange,
   136→  initialData,
   137→  onSubmit,
   138→}: {
   139→  open: boolean
   140→  onOpenChange: (v: boolean) => void
   141→  initialData?: BoUser | null
   142→  onSubmit: (data: UserFormState) => void
   143→}) {
   144→  const { boTheme } = useBackofficeStore()
   145→  const isDark = boTheme === 'dark'
   146→  const [form, setForm] = useState<UserFormState>(emptyForm)
   147→  const isEdit = !!initialData
   148→
   149→  // Reset form when dialog opens
   150→  const handleOpenChange = (v: boolean) => {
   151→    if (v) {
   152→      setForm(
   153→        initialData
   154→          ? {
   155→              name: initialData.name,
   156→              email: initialData.email,
   157→              role: initialData.role,
   158→              zone: initialData.zone ?? '',
   159→              isActive: initialData.isActive,
   160→            }
   161→          : { ...emptyForm }
   162→      )
   163→    }
   164→    onOpenChange(v)
   165→  }
   166→
   167→  const handleSubmit = () => {
   168→    if (!form.name.trim() || !form.email.trim()) return
   169→    onSubmit(form)
   170→    onOpenChange(false)
   171→  }
   172→
   173→  return (
   174→    <Dialog open={open} onOpenChange={handleOpenChange}>
   175→      <DialogContent className="sm:max-w-[480px]">
   176→        <DialogHeader>
   177→          <DialogTitle className={isDark ? 'text-slate-100' : 'text-slate-900'}>
   178→            {isEdit ? 'Modifier l\'utilisateur' : 'Créer un utilisateur'}
   179→          </DialogTitle>
   180→          <DialogDescription>
   181→            {isEdit
   182→              ? 'Modifiez les informations de l\'utilisateur backoffice.'
   183→              : 'Remplissez les informations pour créer un nouvel utilisateur backoffice.'}
   184→          </DialogDescription>
   185→        </DialogHeader>
   186→
   187→        <div className="space-y-4 py-2">
   188→          <div className="space-y-2">
   189→            <Label htmlFor="user-name">Nom complet *</Label>
   190→            <Input
   191→              id="user-name"
   192→              placeholder="Ex: Aminata KONÉ"
   193→              value={form.name}
   194→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   195→            />
   196→          </div>
   197→
   198→          <div className="space-y-2">
   199→            <Label htmlFor="user-email">Email *</Label>
   200→            <Input
   201→              id="user-email"
   202→              type="email"
   203→              placeholder="Ex: aminata@julaba.ci"
   204→              value={form.email}
   205→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   206→            />
   207→          </div>
   208→
   209→          <div className="space-y-2">
   210→            <Label htmlFor="user-role">Rôle *</Label>
   211→            <Select
   212→              value={form.role}
   213→              onValueChange={(v) => setForm({ ...form, role: v as BoRole })}
   214→            >
   215→              <SelectTrigger id="user-role" className="w-full">
   216→                <SelectValue placeholder="Sélectionner un rôle" />
   217→              </SelectTrigger>
   218→              <SelectContent>
   219→                {ALL_ROLES.map((role) => (
   220→                  <SelectItem key={role} value={role}>
   221→                    {ROLE_LABELS[role]}
   222→                  </SelectItem>
   223→                ))}
   224→              </SelectContent>
   225→            </Select>
   226→          </div>
   227→
   228→          <div className="space-y-2">
   229→            <Label htmlFor="user-zone">Zone (optionnel)</Label>
   230→            <Select
   231→              value={form.zone}
   232→              onValueChange={(v) => setForm({ ...form, zone: v })}
   233→            >
   234→              <SelectTrigger id="user-zone" className="w-full">
   235→                <SelectValue placeholder="Aucune zone" />
   236→              </SelectTrigger>
   237→              <SelectContent>
   238→                <SelectItem value="">Aucune zone</SelectItem>
   239→                {ZONES_LIST.map((z) => (
   240→                  <SelectItem key={z} value={z}>
   241→                    {z}
   242→                  </SelectItem>
   243→                ))}
   244→              </SelectContent>
   245→            </Select>
   246→          </div>
   247→        </div>
   248→
   249→        <DialogFooter className="gap-2 pt-2">
   250→          <Button
   251→            variant="outline"
   252→            onClick={() => onOpenChange(false)}
   253→          >
   254→            Annuler
   255→          </Button>
   256→          <Button
   257→            disabled={!form.name.trim() || !form.email.trim()}
   258→            onClick={handleSubmit}
   259→            className="text-white"
   260→          >
   261→            {isEdit ? 'Enregistrer' : 'Créer'}
   262→          </Button>
   263→        </DialogFooter>
   264→      </DialogContent>
   265→    </Dialog>
   266→  )
   267→}

   387→export function BoUtilisateursScreen() {
   388→  const { users, createUser, updateUser, boTheme } = useBackofficeStore()
   389→  const isDark = boTheme === 'dark'
   390→
   391→  // Local state
   392→  const [search, setSearch] = useState('')
   393→  const [roleFilter, setRoleFilter] = useState<RoleFilter>('tous')
   394→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('tous')
   395→  const [showCreateDialog, setShowCreateDialog] = useState(false)
   396→  const [editingUser, setEditingUser] = useState<BoUser | null>(null)
   397→
   398→  // Filtered users
   399→  const filteredUsers = useMemo(() => {
   400→    return users.filter((u) => {
   401→      if (search) {
   402→        const q = search.toLowerCase()
   403→        if (
   404→          !u.name.toLowerCase().includes(q) &&
   405→          !u.email.toLowerCase().includes(q) &&
   406→          !(u.zone ?? '').toLowerCase().includes(q)
   407→        ) {
   408→          return false
   409→        }
   410→      }
   411→      if (roleFilter !== 'tous' && u.role !== roleFilter) return false
   412→      if (statusFilter === 'actif' && !u.isActive) return false
   413→      if (statusFilter === 'inactif' && u.isActive) return false
   414→      return true
   415→    })
   416→  }, [users, search, roleFilter, statusFilter])
   417→
   418→  // Stats
   419→  const stats = useMemo(() => {
   420→    const actifs = users.filter((u) => u.isActive).length
   421→    const inactifs = users.length - actifs
   422→    return { total: users.length, actifs, inactifs }
   423→  }, [users])
   424→
   425→  // Handlers
   426→  const handleCreate = (data: UserFormState) => {
   427→    createUser({
   428→      name: data.name,
   429→      email: data.email,
   430→      role: data.role,
   431→      zone: data.zone || undefined,
   432→      isActive: data.isActive,
   433→    })
   434→  }
   435→
   436→  const handleEdit = (data: UserFormState) => {
   437→    if (!editingUser) return
   438→    updateUser(editingUser.id, {
   439→      name: data.name,
   440→      email: data.email,
   441→      role: data.role,
   442→      zone: data.zone || undefined,
   443→      isActive: data.isActive,
   444→    })
   445→    setEditingUser(null)
   446→  }
   447→
   448→  const handleToggleActive = (user: BoUser) => {
   449→    updateUser(user.id, { isActive: !user.isActive })
   450→  }
   451→
   452→  return (
   453→    <div className={'space-y-6 p-6 ' + (isDark ? 'bg-slate-900' : 'bg-[#F8FAFC]')}>
   454→      {/* ===== HEADER ===== */}
   455→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   456→        <div>
   457→          <h1
   458→            className={`text-2xl font-bold tracking-tight ${isDark ? 'text-slate-100' : 'text-slate-900'}`}
   459→          >
   460→            UTILISATEURS BACKOFFICE
   461→          </h1>
   462→          <p className={`mt-1 text-sm ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>
   463→            Gestion des comptes et permissions d\'accès au backoffice
   464→          </p>
   465→        </div>
   466→        <Button
   467→          onClick={() => setShowCreateDialog(true)}
   468→          className="gap-2 text-white"
   469→        >
   470→          <Plus className="h-4 w-4" />
   471→          Créer utilisateur
   472→        </Button>
   473→      </div>
   474→
   475→      {/* ===== STATS CARDS ===== */}
   476→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   477→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   478→          <CardContent className="flex items-center gap-3 p-4">
   479→            <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-lg ${isDark ? 'bg-slate-700' : 'bg-gray-100'}`}>
   480→              <Users className={`h-5 w-5 ${isDark ? 'text-slate-100' : 'text-slate-900'}`} />
   481→            </div>
   482→            <div>
   483→              <p className={`text-2xl font-bold ${isDark ? 'text-slate-100' : 'text-slate-900'}`}>
   484→                {stats.total}
   485→              </p>
   486→              <p className={`text-xs ${isDark ? 'text-slate-400' : 'text-gray-500'}`}>Total utilisateurs</p>
   487→            </div>
   488→          </CardContent>
   489→        </Card>
   490→        <Card className={isDark ? 'bg-slate-800 border-slate-700' : ''}>
   491→          <CardContent className="flex items-center gap-3 p-4">
   492→            <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-lg ${isDark ? 'bg-emerald-500/10' : 'bg-emerald-50'}`}>
   493→              <Check className="h-5 w-5 text-emerald-600" />

   536→                Rôle :

   538→              <Select
   539→                value={roleFilter}
   540→                onValueChange={(v) => setRoleFilter(v as RoleFilter)}
   541→              >
   542→                <SelectTrigger className="h-8 w-[170px] text-xs">
