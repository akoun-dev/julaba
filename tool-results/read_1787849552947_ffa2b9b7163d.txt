     1→'use client'
     2→
     3→import { useState, useMemo, useCallback } from 'react'
     4→import {
     5→  Search,
     6→  Eye,
     7→  Pause,
     8→  PlayCircle,
     9→  Download,
    10→  Bell,
    11→  ChevronLeft,
    12→  ChevronRight,
    13→  MapPin,
    14→  Phone,
    15→  User,
    16→  Users,
    17→  Store,
    18→  Wheat,
    19→  Handshake,
    20→  Calendar,
    21→  ShieldCheck,
    22→  StickyNote,
    23→  X,
    24→  Hourglass,
    25→} from 'lucide-react'
    26→import { Card, CardContent } from '@/components/ui/card'
    27→import { Button } from '@/components/ui/button'
    28→import { Input } from '@/components/ui/input'
    29→import { Checkbox } from '@/components/ui/checkbox'
    30→import { Separator } from '@/components/ui/separator'
    31→import {
    32→  Select,
    33→  SelectContent,
    34→  SelectItem,
    35→  SelectTrigger,
    36→  SelectValue,
    37→} from '@/components/ui/select'
    38→import {
    39→  Dialog,
    40→  DialogContent,
    41→  DialogHeader,
    42→  DialogTitle,
    43→  DialogDescription,
    44→} from '@/components/ui/dialog'
    45→import {
    46→  useBackofficeStore,
    47→  ACTOR_TYPE_LABELS,
    48→  ACTOR_TYPE_ICONS,
    49→  STATUS_LABELS,
    50→  STATUS_COLORS,
    51→  type BoActor,
    52→} from '@/lib/stores/backoffice-store'
    53→
    54→// ============== CONSTANTS ==============
    55→const ITEMS_PER_PAGE = 15
    56→
    57→type ActorTypeFilter = 'tous' | 'marchand' | 'producteur' | 'cooperatif'
    58→type ActorStatusFilter = 'tous' | 'actif' | 'suspendu' | 'en_attente' | 'rejete'
    59→
    60→// ============== MAIN COMPONENT ==============
    61→
    62→export function BoActeursScreen() {
    63→  const { actors, updateActorStatus, searchQuery, setSearchQuery } =
    64→    useBackofficeStore()
    65→
    66→  // Local state
    67→  const [typeFilter, setTypeFilter] = useState<ActorTypeFilter>('tous')
    68→  const [statusFilter, setStatusFilter] = useState<ActorStatusFilter>('tous')
    69→  const [zoneFilter, setZoneFilter] = useState<string>('tous')
    70→  const [currentPage, setCurrentPage] = useState(1)
    71→  const [selectedActors, setSelectedActors] = useState<Set<string>>(new Set())
    72→  const [showDetailModal, setShowDetailModal] = useState(false)
    73→  const [detailActor, setDetailActor] = useState<BoActor | null>(null)
    74→
    75→  // Unique zones from data
    76→  const zones = useMemo(() => {
    77→    const zoneSet = new Set(actors.map((a) => a.zone))
    78→    return Array.from(zoneSet).sort()
    79→  }, [actors])
    80→
    81→  // Filtered actors
    82→  const filteredActors = useMemo(() => {
    83→    return actors.filter((actor) => {
    84→      // Search filter
    85→      if (searchQuery) {
    86→        const q = searchQuery.toLowerCase()
    87→        const matchesSearch =
    88→          actor.firstName.toLowerCase().includes(q) ||
    89→          actor.lastName.toLowerCase().includes(q) ||
    90→          actor.actorId.toLowerCase().includes(q) ||
    91→          actor.phone.includes(q) ||
    92→          actor.zone.toLowerCase().includes(q)
    93→        if (!matchesSearch) return false
    94→      }
    95→
    96→      // Type filter
    97→      if (typeFilter !== 'tous' && actor.type !== typeFilter) return false
    98→
    99→      // Status filter
   100→      if (statusFilter !== 'tous' && actor.status !== statusFilter) return false
   101→
   102→      // Zone filter
   103→      if (zoneFilter !== 'tous' && actor.zone !== zoneFilter) return false
   104→
   105→      return true
   106→    })
   107→  }, [actors, searchQuery, typeFilter, statusFilter, zoneFilter])
   108→
   109→  // Counters
   110→  const counts = useMemo(() => {
   111→    return {
   112→      total: actors.length,
   113→      marchands: actors.filter((a) => a.type === 'marchand').length,
   114→      producteurs: actors.filter((a) => a.type === 'producteur').length,
   115→      cooperatives: actors.filter((a) => a.type === 'cooperatif').length,
   116→    }
   117→  }, [actors])
   118→
   119→  // Pagination
   120→  const totalPages = Math.max(1, Math.ceil(filteredActors.length / ITEMS_PER_PAGE))
   121→  const paginatedActors = useMemo(() => {
   122→    const start = (currentPage - 1) * ITEMS_PER_PAGE
   123→    return filteredActors.slice(start, start + ITEMS_PER_PAGE)
   124→  }, [filteredActors, currentPage])
   125→
   126→  // Reset page when filters change
   127→  const handleTypeFilterChange = useCallback((val: string) => {
   128→    setTypeFilter(val as ActorTypeFilter)
   129→    setCurrentPage(1)
   130→    setSelectedActors(new Set())
   131→  }, [])
   132→
   133→  const handleStatusFilterChange = useCallback((val: string) => {
   134→    setStatusFilter(val as ActorStatusFilter)
   135→    setCurrentPage(1)
   136→    setSelectedActors(new Set())
   137→  }, [])
   138→
   139→  const handleZoneFilterChange = useCallback((val: string) => {
   140→    setZoneFilter(val)
   141→    setCurrentPage(1)
   142→    setSelectedActors(new Set())
   143→  }, [])
   144→
   145→  const handleSearchChange = useCallback(
   146→    (e: React.ChangeEvent<HTMLInputElement>) => {
   147→      setSearchQuery(e.target.value)
   148→      setCurrentPage(1)
   149→      setSelectedActors(new Set())
   150→    },
   151→    [setSearchQuery]
   152→  )
   153→
   154→  // Selection
   155→  const toggleSelectAll = useCallback(() => {
   156→    if (selectedActors.size === paginatedActors.length) {
   157→      setSelectedActors(new Set())
   158→    } else {
   159→      setSelectedActors(new Set(paginatedActors.map((a) => a.id)))
   160→    }
   161→  }, [selectedActors, paginatedActors])
   162→
   163→  const toggleSelectActor = useCallback((id: string) => {
   164→    setSelectedActors((prev) => {
   165→      const next = new Set(prev)
   166→      if (next.has(id)) {
   167→        next.delete(id)
   168→      } else {
   169→        next.add(id)
   170→      }
   171→      return next
   172→    })
   173→  }, [])
   174→
   175→  const isAllSelected =
   176→    paginatedActors.length > 0 && selectedActors.size === paginatedActors.length
   177→
   178→  // Actions
   179→  const handleViewActor = useCallback((actor: BoActor) => {
   180→    setDetailActor(actor)
   181→    setShowDetailModal(true)
   182→  }, [])
   183→
   184→  const handleSuspend = useCallback(
   185→    (actor: BoActor) => {
   186→      updateActorStatus(actor.id, 'suspendu')
   187→    },
   188→    [updateActorStatus]
   189→  )
   190→
   191→  const handleReactivate = useCallback(
   192→    (actor: BoActor) => {
   193→      updateActorStatus(actor.id, 'actif')
   194→    },
   195→    [updateActorStatus]
   196→  )
   197→
   198→  // CSV Export
   199→  const handleExportCSV = useCallback(() => {
   200→    const selected = actors.filter((a) => selectedActors.has(a.id))
   201→    if (selected.length === 0) return
   202→
   203→    const headers = [
   204→      'ID Acteur',
   205→      'Prénom',
   206→      'Nom',
   207→      'Type',
   208→      'Téléphone',
   209→      'Zone',
   210→      'Statut',
   211→      'Identificateur',
   212→      'Validé par',
   213→      'Date validation',
   214→      'Date création',
   215→    ]
   216→    const rows = selected.map((a) => [
   217→      a.actorId,
   218→      a.firstName,
   219→      a.lastName,
   220→      ACTOR_TYPE_LABELS[a.type],
   221→      a.phone,
   222→      a.zone,
   223→      STATUS_LABELS[a.status],
   224→      a.identificateurName || '',
   225→      a.validatedBy || '',
   226→      a.validatedAt ? new Date(a.validatedAt).toLocaleDateString('fr-FR') : '',
   227→      new Date(a.createdAt).toLocaleDateString('fr-FR'),
   228→    ])
   229→    const csv = [headers.join(';'), ...rows.map((r) => r.join(';'))].join('\n')
   230→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' })
   231→    const url = URL.createObjectURL(blob)
   232→    const link = document.createElement('a')
   233→    link.href = url
   234→    link.download = `acteurs_export_${new Date().toISOString().slice(0, 10)}.csv`
   235→    link.click()
   236→    URL.revokeObjectURL(url)
   237→  }, [actors, selectedActors])
   238→
   239→  // Reset filters
   240→  const resetFilters = useCallback(() => {
   241→    setTypeFilter('tous')
   242→    setStatusFilter('tous')
   243→    setZoneFilter('tous')
   244→    setSearchQuery('')
   245→    setCurrentPage(1)
   246→    setSelectedActors(new Set())
   247→  }, [setSearchQuery])
   248→
   249→  const hasActiveFilters =
   250→    typeFilter !== 'tous' || statusFilter !== 'tous' || zoneFilter !== 'tous' || searchQuery !== ''
   251→
   252→  // ============== RENDER ==============
   253→
   254→  return (
   255→    <div className="space-y-4 p-6">
   256→      {/* ===== HEADER ===== */}
   257→      <div>
   258→        <h1
   259→          className="text-2xl font-bold tracking-tight text-slate-900"
   260→        >
   261→          GESTION DES ACTEURS
   262→        </h1>
   263→        <p className="text-sm text-muted-foreground mt-1">
   264→          Consultez, filtrez et gérez l\'ensemble des acteurs enregistrés sur la
   265→          plateforme Jùlaba.
   266→        </p>
   267→      </div>
   268→
   269→      {/* ===== SEARCH ===== */}
   270→      <Card>
   271→        <CardContent className="p-4">
   272→          <div className="relative max-w-md">
   273→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   274→            <Input
   275→              placeholder="Rechercher par nom, téléphone, ID, zone..."
   276→              value={searchQuery}
   277→              onChange={handleSearchChange}
   278→              className="pl-9"
   279→            />
   280→          </div>
   281→        </CardContent>
   282→      </Card>
   283→
   284→      {/* ===== FILTERS ===== */}
   285→      <Card>
   286→        <CardContent className="p-4">
   287→          <div className="flex flex-wrap items-center gap-3">
   288→            <span className="text-sm font-medium text-slate-900">
   289→              Filtres :
   290→            </span>
   291→
   292→            <Select value={typeFilter} onValueChange={handleTypeFilterChange}>
   293→              <SelectTrigger className="w-[170px]">
   294→                <SelectValue placeholder="Type" />
   295→              </SelectTrigger>
   296→              <SelectContent>
   297→                <SelectItem value="tous">Tous les types</SelectItem>
   298→                <SelectItem value="marchand">
   299→                  Marchand(e)s
   300→                </SelectItem>
   301→                <SelectItem value="producteur">
   302→                   Producteur(rice)s
   303→                </SelectItem>
   304→                <SelectItem value="cooperatif">
   305→                   Coopératives
   306→                </SelectItem>
   307→              </SelectContent>
   308→            </Select>
   309→
   310→            <Select
   311→              value={statusFilter}
   312→              onValueChange={handleStatusFilterChange}
   313→            >
   314→              <SelectTrigger className="w-[170px]">
   315→                <SelectValue placeholder="Statut" />
   316→              </SelectTrigger>
   317→              <SelectContent>
   318→                <SelectItem value="tous">Tous les statuts</SelectItem>
   319→                <SelectItem value="actif">Actif</SelectItem>
   320→                <SelectItem value="suspendu">Suspendu</SelectItem>
   321→                <SelectItem value="en_attente"><span className="flex items-center gap-1.5"><Hourglass className="w-3.5 h-3.5" /> En attente</span></SelectItem>
   322→                <SelectItem value="rejete">Rejeté</SelectItem>
   323→              </SelectContent>
   324→            </Select>
   325→
   326→            <Select value={zoneFilter} onValueChange={handleZoneFilterChange}>
   327→              <SelectTrigger className="w-[180px]">
   328→                <SelectValue placeholder="Zone" />
   329→              </SelectTrigger>
   330→              <SelectContent>
   331→                <SelectItem value="tous">Toutes les zones</SelectItem>
   332→                {zones.map((z) => (
   333→                  <SelectItem key={z} value={z}>
   334→                    {z}
   335→                  </SelectItem>
   336→                ))}
   337→              </SelectContent>
   338→            </Select>
   339→
   340→            {hasActiveFilters && (
   341→              <Button
   342→                variant="ghost"
   343→                size="sm"
   344→                onClick={resetFilters}
   345→                className="text-muted-foreground hover:text-foreground"
   346→              >
   347→                <X className="size-3.5 mr-1" />
   348→                Réinitialiser
   349→              </Button>
   350→            )}
   351→          </div>
   352→        </CardContent>
   353→      </Card>
   354→
   355→      {/* ===== COUNTER BAR ===== */}
   356→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   357→        <Card>
   358→          <CardContent className="p-4 flex items-center gap-3">
   359→            <div className="flex items-center justify-center size-10 rounded-lg bg-gray-100">
   360→              <Users className="size-5 text-gray-600" />
   361→            </div>
   362→            <div>
   363→              <p className="text-2xl font-bold text-slate-900">
   364→                {counts.total}
   365→              </p>
   366→              <p className="text-xs text-muted-foreground">Total acteurs</p>
   367→            </div>
   368→          </CardContent>
   369→        </Card>
   370→        <Card>
   371→          <CardContent className="p-4 flex items-center gap-3">
   372→            <div className="flex items-center justify-center size-10 rounded-lg bg-orange-50">
   373→              <Store className="size-5 text-orange-600" />
   374→            </div>
   375→            <div>
   376→              <p className="text-2xl font-bold text-slate-900">
   377→                {counts.marchands}
   378→              </p>
   379→              <p className="text-xs text-muted-foreground">Marchands</p>
   380→            </div>
   381→          </CardContent>
   382→        </Card>
   383→        <Card>
   384→          <CardContent className="p-4 flex items-center gap-3">
   385→            <div className="flex items-center justify-center size-10 rounded-lg bg-green-50">
   386→              <Wheat className="size-5 text-green-600" />
   387→            </div>
   388→            <div>
   389→              <p className="text-2xl font-bold text-slate-900">
   390→                {counts.producteurs}
   391→              </p>
   392→              <p className="text-xs text-muted-foreground">Producteurs</p>
   393→            </div>
   394→          </CardContent>
   395→        </Card>
   396→        <Card>
   397→          <CardContent className="p-4 flex items-center gap-3">
   398→            <div className="flex items-center justify-center size-10 rounded-lg bg-amber-50">
   399→              <Handshake className="size-5 text-amber-600" />
   400→            </div>
   401→            <div>
   402→              <p className="text-2xl font-bold text-slate-900">
   403→                {counts.cooperatives}
   404→              </p>
   405→              <p className="text-xs text-muted-foreground">Coopératives</p>
   406→            </div>
   407→          </CardContent>
   408→        </Card>
   409→      </div>
   410→
   411→      {/* ===== BULK ACTIONS ===== */}
   412→      {selectedActors.size > 0 && (
   413→        <Card className="border-l-4 border-l-amber-500">
   414→          <CardContent className="p-4 flex items-center justify-between">
   415→            <span className="text-sm font-medium text-slate-900">
   416→              <strong>{selectedActors.size}</strong> acteur(s) sélectionné(s)
   417→            </span>
   418→            <div className="flex items-center gap-2">
   419→              <Button size="sm" variant="outline" onClick={handleExportCSV}>
   420→                <Download className="size-3.5 mr-1.5" />
   421→                Exporter CSV
   422→              </Button>
   423→              <Button size="sm" variant="outline">
   424→                <Bell className="size-3.5 mr-1.5" />
   425→                Notifier
   426→              </Button>
   427→            </div>
   428→          </CardContent>
   429→        </Card>
   430→      )}
   431→
   432→      {/* ===== DATA TABLE ===== */}
   433→      <Card>
   434→        <CardContent className="p-0">
   435→          <div className="max-h-[620px] overflow-y-auto">
   436→            <table className="w-full text-sm">
   437→              <thead className="sticky top-0 z-10 bg-gray-50 border-b">
   438→                <tr>
   439→                  <th className="w-10 px-3 py-3 text-left">
   440→                    <Checkbox
   441→                      checked={isAllSelected}
   442→                      onCheckedChange={toggleSelectAll}
   443→                      aria-label="Sélectionner tout"
   444→                    />
   445→                  </th>
   446→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   447→                    ID
   448→                  </th>
   449→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   450→                    Nom complet
   451→                  </th>
   452→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   453→                    Type
   454→                  </th>
   455→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   456→                    Zone
   457→                  </th>
   458→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   459→                    Téléphone
   460→                  </th>
   461→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   462→                    Identificateur
   463→                  </th>
   464→                  <th className="px-3 py-3 text-left font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   465→                    Statut
   466→                  </th>
   467→                  <th className="px-3 py-3 text-right font-semibold text-xs uppercase tracking-wider text-muted-foreground">
   468→                    Actions
   469→                  </th>
   470→                </tr>
   471→              </thead>
   472→              <tbody className="divide-y">
   473→                {paginatedActors.length === 0 ? (
   474→                  <tr>
   475→                    <td colSpan={9} className="px-3 py-12 text-center text-muted-foreground">
   476→                      <div className="flex flex-col items-center gap-2">
   477→                        <User className="size-8 opacity-30" />
   478→                        <p>Aucun acteur trouvé</p>
   479→                        {hasActiveFilters && (
   480→                          <Button
   481→                            variant="link"
   482→                            size="sm"
   483→                            onClick={resetFilters}
   484→                          >
   485→                            Réinitialiser les filtres
   486→                          </Button>
   487→                        )}
   488→                      </div>
   489→                    </td>
   490→                  </tr>
   491→                ) : (
   492→                  paginatedActors.map((actor) => {
   493→                    const isSelected = selectedActors.has(actor.id)
   494→                    return (
   495→                      <tr
   496→                        key={actor.id}
   497→                        className={`${
   498→                          isSelected
   499→                            ? 'bg-amber-50/60'
   500→                            : 'hover:bg-gray-50/80'
   501→                        } transition-colors`}
   502→                      >
   503→                        <td className="px-3 py-3">
   504→                          <Checkbox
   505→                            checked={isSelected}
   506→                            onCheckedChange={() => toggleSelectActor(actor.id)}
   507→                            aria-label={`Sélectionner ${actor.firstName} ${actor.lastName}`}
   508→                          />
   509→                        </td>
   510→                        <td className="px-3 py-3">
   511→                          <span className="font-mono text-xs font-semibold text-slate-900">
   512→                            {actor.actorId}
   513→                          </span>
   514→                        </td>
   515→                        <td className="px-3 py-3">
   516→                          <div className="flex items-center gap-2">
   517→                            <div className="flex items-center justify-center size-7 rounded-full bg-gray-200 text-xs font-semibold text-gray-600 shrink-0">
   518→                              {actor.firstName.charAt(0)}
   519→                              {actor.lastName.charAt(0)}
   520→                            </div>
   521→                            <span className="font-medium text-slate-900">
   522→                              {actor.firstName} {actor.lastName}
   523→                            </span>
   524→                          </div>
   525→                        </td>
   526→                        <td className="px-3 py-3">
   527→                          <span className="inline-flex items-center gap-1.5">
   528→                            <span>{ACTOR_TYPE_ICONS[actor.type]}</span>
   529→                            <span className="text-xs text-muted-foreground">
   530→                              {ACTOR_TYPE_LABELS[actor.type]}
   531→                            </span>
   532→                          </span>
   533→                        </td>
   534→                        <td className="px-3 py-3">
   535→                          <span className="text-xs text-muted-foreground">
   536→                            {actor.zone}
   537→                          </span>
   538→                        </td>
   539→                        <td className="px-3 py-3">
   540→                          <span className="text-xs font-mono text-muted-foreground">
   541→                            {actor.phone}
   542→                          </span>
   543→                        </td>
   544→                        <td className="px-3 py-3">
   545→                          <span className="text-xs text-muted-foreground">
   546→                            {actor.identificateurName || '—'}
   547→                          </span>
   548→                        </td>
   549→                        <td className="px-3 py-3">
   550→                          <span
   551→                            className={`inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium ${
   552→                              STATUS_COLORS[actor.status] || 'bg-gray-100 text-gray-800'
   553→                            }`}
   554→                          >
   555→                            {STATUS_LABELS[actor.status]}
   556→                          </span>
   557→                        </td>
   558→                        <td className="px-3 py-3 text-right">
   559→                          <div className="flex items-center justify-end gap-1">
   560→                            <Button
   561→                              size="sm"
   562→                              variant="ghost"
   563→                              className="size-7 p-0 text-muted-foreground hover:text-foreground"
   564→                              onClick={() => handleViewActor(actor)}
   565→                              title="Voir les détails"
   566→                            >
   567→                              <Eye className="size-3.5" />
   568→                            </Button>
   569→                            {actor.status === 'actif' && (
   570→                              <Button
   571→                                size="sm"
   572→                                variant="ghost"
   573→                                className="size-7 p-0 text-amber-600 hover:text-amber-700 hover:bg-amber-50"
   574→                                onClick={() => handleSuspend(actor)}
   575→                                title="Suspendre cet acteur"
   576→                              >
   577→                                <Pause className="size-3.5" />
   578→                              </Button>
   579→                            )}
